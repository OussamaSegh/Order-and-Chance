% Native 1049-pixel grain tiles remain >=359 ppi after the 7/6 print scaling.
% The muted tile is preblended RGB, avoiding transparent paper overlays in print.
\newcommand{\coverprintpaper}[3]{%
 \ifdim#1pt<1pt
  \def\paperstyle{Muted}\def\paperfile{tools/assets/print/paper-muted.png}%
 \else
  \def\paperstyle{Plain}\def\paperfile{tools/assets/approved-charcoal-paper.png}%
 \fi
 \ifcsname ocPaperPrepared\paperstyle\endcsname\else
  \expandafter\gdef\csname ocPaperPrepared\paperstyle\endcsname{1}%
  \xsbox{ocPaper\paperstyle}{\includegraphics[width=2.5in]{\paperfile}}%
 \fi
 \pgfmathsetmacro{\tilecols}{int(ceil((#2)/(2.5*1in))-1)}%
 \pgfmathsetmacro{\tilerows}{int(ceil((#3)/(2.5*1500/1049*1in))-1)}%
 \begin{pgfinterruptboundingbox}% Tiles outside the clip must not move the cover canvas.
 \foreach \xx in {0,...,\tilecols}{%
  \foreach \yy in {0,...,\tilerows}{%
   \node[anchor=south west,inner sep=0pt] at ({\xx*2.5in},{\yy*2.5*1500/1049*1in})
    {\xusebox{ocPaper\paperstyle}};%
  }%
 }%
 \end{pgfinterruptboundingbox}%
}
% Matching charcoal texture for backs and variable-content fronts.
\newcommand{\coverpaper}[3][1]{%
 \pgfmathsetlengthmacro{\paperw}{#2}
 \pgfmathsetlengthmacro{\paperh}{#3}
 \begin{scope}
  \fill[ocBlack] (0,0) rectangle ({#2},{#3});
  \clip (0,0) rectangle ({#2},{#3});
  \ifcoverprint
   \coverprintpaper{#1}{#2}{#3}%
  \else
   \node[anchor=south west,inner sep=0pt,opacity=#1] at (0,0)
    {\includegraphics[width=\paperw,height=\paperh]{tools/assets/approved-charcoal-paper.png}};
  \fi
 \end{scope}}
