% Paperback geometry from common/print-layout.json; spine from actual interior pages.
\documentclass[border=0pt]{standalone}
\input{tools/cover-colors}
\coverprinttrue
\input{common/print-layout}
\begin{document}%
\newsavebox{\printfrontbox}%
\pgfmathsetmacro{\frontheight}{6*\PRINTHEIGHT/\PRINTWIDTH}%
\prepareprintcoverassets{\frontheight}%
\sbox{\printfrontbox}{\resizebox{\PRINTWIDTH in}{!}{%
  \begin{tikzpicture}%
    \coordinate (O) at (0,0);%
    \let\CH\frontheight\coverbgfalse%
    \useasboundingbox (0,0) rectangle (6in,{\CH*1in});%
    \input{tools/cover-front}%
  \end{tikzpicture}}}%
\newsavebox{\printbackbox}%
\sbox{\printbackbox}{\resizebox{\PRINTWIDTH in}{!}{%
  \begin{tikzpicture}%
    \coordinate (O) at (0,0);%
    \let\CH\frontheight%
    \useasboundingbox (0,0) rectangle (6in,{\CH*1in});%
    \input{tools/cover-back}%
  \end{tikzpicture}}}%
\begin{tikzpicture}
  \let\B\PRINTBLEED
  \let\TW\PRINTWIDTH\let\TH\PRINTHEIGHT
  \pgfmathsetmacro{\W}{2*\B+2*\TW+\SPINE}
  \useasboundingbox (0,0) rectangle ({\W*1in},{(\TH+2*\B)*1in});
  \coverpaper[.35]{\W*1in}{(\TH+2*\B)*1in}% Same charcoal for all backs/spines/fronts.
  % Front cover, preboxed outside the wrap's TikZ coordinate system.
  \node[anchor=south west,inner sep=0pt,outer sep=0pt] at ({(\B+\TW+\SPINE)*1in},{\B*1in})
    {\usebox{\printfrontbox}};
  % Back cover
  \coordinate (K) at ({\B*1in},{\B*1in});
  \node[anchor=south west,inner sep=0pt,outer sep=0pt] at (K) {\usebox{\printbackbox}};
  % ISBN barcode (EAN-13, made by tools/ean13.py) in KDP's barcode area, lower right of the back
  % cover, when the tome has its ISBN; then untick "add a barcode" in KDP.
  \ifdefined\BARCODE
    \fill[white] ($(K)+({(\TW-2.6)*1in},0.5in)$) rectangle ++(2.05in,1.25in);
    \begin{scope}[shift={($(K)+({(\TW-2.6)*1in},0.5in)$)}]
      \input{\BARCODE}
    \end{scope}
  \fi
  % Spine (text reads top to bottom, as on English-language books)
  \ifdefined\SPINETEXT
    \pgfmathsetmacro{\spfs}{min(10,\SPINE*72*0.42)}
    \coordinate (S) at ({(\B+\TW+\SPINE/2)*1in},{(\B+\TH/2)*1in});
    \node[rotate=-90,ivory,font=\fontsize{\spfs}{\spfs}\selectfont] at (S)
      {\covertracking{12}{ORDER AND CHANCE}\quad{\color{accent}\covertracking{6}{VOLUME \TOMENUM}}\quad\TOMETITLE};
  \fi
\end{tikzpicture}%
\end{document}
