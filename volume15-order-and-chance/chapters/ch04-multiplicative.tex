\chapter{Random multiplicative functions}\label{ch:oc-multiplicative}\label{ch:randommult}

Steinhaus and Rademacher random multiplicative functions are the models. The M\"obius function is the point one would like to prove behaves like them. Chowla's conjecture and Sarnak's conjecture are stated with the comparisons the models make unconditional. Random polynomials are not a separate chapter; their equidistribution is the last section.

\section{Exercises}

\begin{exercise}[$\star$]\label{ex:dirichlet}
(a) Deduce from Theorem~\ref{thm:wintner} and partial summation that, almost surely, the series $\sum_nf(n)n^{-s}$ converges for every $s$ with $\Re s>\frac12$ and defines a holomorphic function there. (b) In the Steinhaus model, show that it equals $\prod_p(1-f(p)p^{-s})^{-1}$ for $\Re s>1$. Why can it not be continued to a function holomorphic on a disc around $\frac12$ for every sample? (Only a heuristic is asked for (b).)
\end{exercise}

\begin{exercise}[$\star$]\label{ex:pz}
Using Proposition~\ref{prop:orth}, Theorem~\ref{thm:fourth} and the Paley--Zygmund inequality $P(Y\ge\frac12\mathbb EY)\ge\frac{(\mathbb EY)^2}{4\mathbb EY^2}$ for $Y\ge0$, show that $P(|S(x)|^2\ge\frac12\lfloor x\rfloor)\gg\frac1{\log x}$ in the Steinhaus model. Is this a good lower bound?
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:notgauss}
Assume Harper's theorem $\mathbb E|S(x)|\ll\frac{\sqrt x}{(\log\log x)^{1/4}}$. Show that $\frac{S(x)}{\sqrt x}\to0$ in probability, although $\mathbb E\frac{|S(x)|^2}x\to1$. Deduce that $\frac{S(x)}{\sqrt x}$ does not converge in law to a complex Gaussian, and that the second moment is carried by events of probability tending to $0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:rademacherclt}
State and prove the analogue of Theorem~\ref{thm:condclt} in the Rademacher model, where the limit is a standard real Gaussian.
\end{exercise}

\begin{exercise}[$\star$]\label{ex:tablelink}
(a) Show that, in the Steinhaus model, $\mathbb E|S(x)|^4\ge\frac{\lfloor x\rfloor^4}{D(\lfloor x\rfloor)}$, where $D(N)$ is the number of distinct entries of the $N\times N$ multiplication table (Section~4.4), and deduce from Theorem~\ref{thm:multtable} that $\mathbb E|S(x)|^4\gg x^2(\log x)^\delta$. (b) Compute $\mathbb E|S(10)|^4$ exactly with the formula of the proof of Theorem~\ref{thm:fourth}, and compare with this lower bound and with the value $2\cdot10^2-10$ for $10$ independent uniform points of the circle.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{ex:varV}
Show that $\mathbb EV(x)^2\ge cx^2\log x$ for $x$ large, so that $\frac{\mathrm{Var}\,V(x)}{(\mathbb EV(x))^2}\to\infty$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:golden}
Let $\varphi=\frac{1+\sqrt5}2$ and $x_n=n\varphi$. Using $\|k\varphi\|\ge\frac1{3k}$ for $k\ge1$ (where $\|\cdot\|$ is the distance to the nearest integer), show that $|S_k|\le\frac{3k}2$, and deduce from Theorem~\ref{thm:erdosturan} that $D_N=O(N^{-1/3})$.
\end{exercise}

\begin{exercise}[$\star$]\label{ex:catalan}
Show that $\int(2\cos\theta)^{2m}\dd\mu_{ST}=C_m=\frac1{m+1}\binom{2m}m$ and that the odd moments vanish.
\end{exercise}

\begin{exercise}[$\star$]\label{ex:squarefreecount}
Show that the number of monic squarefree polynomials of degree $n\ge2$ over $\mathbb F_p$ is exactly $p^n-p^{n-1}$. (Every monic $f$ is uniquely $g^2h$ with $h$ squarefree.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:noroot}
Show that for $n\ge p$ the proportion of monic polynomials of degree $n$ over $\mathbb F_p$ with no root in $\mathbb F_p$ is exactly $(1-\frac1p)^p$. Compare with the probability $\sum_{j=0}^n\frac{(-1)^j}{j!}$ that a random permutation of $S_n$ has no fixed point.
\end{exercise}

\begin{exercise}[$\star$]\label{ex:jordaninfinite}
Show that every matrix of $GL_2(\mathbb C)$ is conjugate to an upper triangular matrix, so that Lemma~\ref{lem:jordan} fails for the infinite group $GL_2(\mathbb C)$ and its proper subgroup of upper triangular matrices.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:snrecognize}
Let $G\le S_n$, $n\ge3$, be transitive, and suppose that $G$ contains a transposition and an $(n-1)$-cycle. Show that $G=S_n$. Deduce that if $f\in\Z[X]$ is monic of degree $n$ and, for three primes $p_1,p_2,p_3$ not dividing $\operatorname{disc}(f)$, $\bar f$ is irreducible mod $p_1$, of type $1\,(n-1)$ mod $p_2$ and of type $1^{n-2}2$ mod $p_3$, then $G_f=S_n$.
\end{exercise}

\section{Hints}

\begin{hint}{ex:dirichlet}
(a) $\sum_{n\le N}f(n)n^{-s}=S(N)N^{-s}+s\int_1^NS(t)t^{-s-1}\dd t$. (b) Near $s=\frac12$, $\log F(s)\approx\sum_pf(p)p^{-s}$, whose variance at $s=\frac12+\sigma$ is $\frac12\sum_pp^{-1-2\sigma}\to\infty$ as $\sigma\to0$.
\end{hint}
\begin{hint}{ex:pz}
$Y=|S(x)|^2$. Compare with the probability $e^{-1/2}$ for a Gaussian.
\end{hint}
\begin{hint}{ex:notgauss}
Markov's inequality for the first point. If $X_x\Rightarrow Z$ and $X_x\to0$ in probability, then $Z=0$.
\end{hint}
\begin{hint}{ex:rademacherclt}
Replace $\xi_j$ by independent signs; $\Re(\overline wW)$ becomes $wW$ with $w$ real. For $p>\sqrt x$ and $m<\sqrt x$, $f(pm)=f(p)f(m)$ since $p\nmid m$.
\end{hint}
\begin{hint}{ex:tablelink}
$\mathbb E|S(x)|^4=\sum_Nr(N)^2$ with $\sum_Nr(N)=\lfloor x\rfloor^2$; Cauchy--Schwarz over the $N$ with $r(N)>0$.
\end{hint}
\begin{hint}{ex:varV}
$\mathbb E|S(y)|^2|S(z)|^2$ counts $ac=bd$ with $a,b\le y$, $c,d\le z$; parametrise as in Theorem~\ref{thm:fourth} and restrict $p,q$ to $(\sqrt x,x^{3/4}]$.
\end{hint}

\begin{hint}{ex:golden}
$|\sum_{n\le N}e(n\alpha)|\le\frac1{|\sin\pi\alpha|}\le\frac1{2\|\alpha\|}$. Then take $K=N^{2/3}$.
\end{hint}
\begin{hint}{ex:catalan}
Count the paths: $(e^{i\theta}+e^{-i\theta})^{2m}$ expanded, then integrate against $\frac2\pi\sin^2\theta=\frac1\pi(1-\cos2\theta)$.
\end{hint}
\begin{hint}{ex:squarefreecount}
Let $s_n$ be the number of squarefree ones. Then $p^n=\sum_ks_{n-2k}p^k$; compare generating functions.
\end{hint}
\begin{hint}{ex:noroot}
Inclusion--exclusion over the set of roots: for a set $R$ of $j\le n$ elements of $\mathbb F_p$, the polynomials vanishing on $R$ are $\prod_{r\in R}(X-r)\cdot g$.
\end{hint}
\begin{hint}{ex:jordaninfinite}
A matrix over $\mathbb C$ has an eigenvector.
\end{hint}
\begin{hint}{ex:snrecognize}
The $(n-1)$-cycle fixes a point $i$ and is transitive on the others; with transitivity, $G$ is $2$-transitive. Conjugate the transposition.
\end{hint}

\section{Solutions}

\begin{solution}{ex:dirichlet}
\idea A function that grows like $t^{1/2+\varepsilon}$ is a harmless weight for $t^{-s-1}$ as soon as $\Re s>\frac12+\varepsilon$.
\writeup (a) Fix a sample in the almost sure event of Theorem~\ref{thm:wintner} for $\varepsilon=\frac1j$, for every $j$ (a countable intersection). By partial summation, $\sum_{n\le N}f(n)n^{-s}=S(N)N^{-s}+s\int_1^NS(t)t^{-s-1}\dd t$. If $\sigma=\Re s>\frac12+\varepsilon$, then $|S(N)N^{-s}|\le C_\varepsilon N^{1/2+\varepsilon-\sigma}\to0$, and the integral converges absolutely and uniformly on $\Re s\ge\frac12+2\varepsilon$, $|s|\le R$, since $|S(t)t^{-s-1}|\le C_\varepsilon t^{-1-\varepsilon}$. So the series converges locally uniformly on $\Re s>\frac12$, and its sum is holomorphic, as a locally uniform limit of entire functions. (b) For $\Re s>1$ the series converges absolutely and $f$ is completely multiplicative, so the Euler product follows as in Volume~6. Near $\frac12$: for $\sigma>0$, $\log F(\frac12+\sigma)=\sum_pf(p)p^{-1/2-\sigma}+O(1)$, a sum of independent terms whose real part has variance $\frac12\sum_pp^{-1-2\sigma}\sim\frac12\log\frac1\sigma$. So $|F(\frac12+\sigma)|$ fluctuates by factors $e^{\pm c\sqrt{\log(1/\sigma)}}$ as $\sigma\to0$, which a function holomorphic at $\frac12$ cannot do unless it vanishes there; a law of the iterated logarithm makes this rigorous, and Exercise~\ref{ex:dirichlet}(b) is only the heuristic.
\end{solution}

\begin{solution}{ex:pz}
\idea Second and fourth moments give a lower bound for large values, and the large fourth moment makes it weak.
\writeup With $Y=|S(x)|^2$, $\mathbb EY=\lfloor x\rfloor$ and $\mathbb EY^2=\mathbb E|S(x)|^4\le Cx^2\log x$ for $x\ge2$ (Theorem~\ref{thm:fourth}). Paley--Zygmund gives $P(Y\ge\frac12\lfloor x\rfloor)\ge\frac{\lfloor x\rfloor^2}{4Cx^2\log x}\ge\frac1{16C\log x}$. For a standard complex Gaussian, $P(|Z|^2\ge\frac12)=e^{-1/2}=0.61$. The truth lies in between: by Exercise~\ref{ex:notgauss} the probability tends to $0$, but numerically it is still about $0.35$ at $x=10^6$ ($300$ samples). The bound $\frac1{\log x}$ is poor, because the fourth moment is dominated by rare large values that say little about typical ones.
\end{solution}

\begin{solution}{ex:notgauss}
\idea A first moment that is $o(\sqrt x)$ forces $S(x)/\sqrt x$ to be small most of the time, and then the second moment, which is exactly $1$, must come from the exceptions.
\writeup By Markov, $P(|S(x)|\ge\eta\sqrt x)\le\frac{\mathbb E|S(x)|}{\eta\sqrt x}\ll\frac1{\eta(\log\log x)^{1/4}}\to0$ for each $\eta>0$. If $X_x=\frac{S(x)}{\sqrt x}$ converged in law to $Z$, then since $X_x\to0$ in probability, also in law, the limits coincide and $Z=0$, which is not a standard complex Gaussian. And $\mathbb E|X_x|^2=\frac{\lfloor x\rfloor}x\to1$: for any $\eta>0$, $\mathbb E[|X_x|^2\mathbf 1\{|X_x|\le\eta\}]\le\eta^2$, so $\mathbb E[|X_x|^2\mathbf 1\{|X_x|>\eta\}]\ge1-\eta^2-o(1)$, while $P(|X_x|>\eta)\to0$. This does not contradict Theorem~\ref{thm:condclt}, whose limit is a Gaussian with the random variance $V(x)$: a mixture of Gaussians can be small with high probability and still have a fixed second moment.
\end{solution}

\begin{solution}{ex:rademacherclt}
\idea The same conditioning works, with signs instead of points of the circle.
\writeup In the Rademacher model, every $n\le x$ with $P^+(n)>\sqrt x$ is $pm$ with $p>\sqrt x>m$, and $f(n)=f(p)f(m)$ if $m$ is squarefree, $0$ otherwise; so $T(x)=\sum_{\sqrt x<p\le x}f(p)S(\frac xp)$ again, with $V(x)=\sum_pS(\frac xp)^2\ge\pi(x)-\pi(\frac x2)$. For independent signs $\xi_j$ and real $a_j$, the proof of Lemma~\ref{lem:circleclt} with $X_j=\frac{wa_j\xi_j}{\sqrt V}$, $w\in\R$, $s_j^2=\frac{w^2a_j^2}V$, gives $|\mathbb Ee^{iwW}-e^{-w^2/2}|\le\frac{|w|^3\mu}6+\frac{w^4\mu^2}8$. Theorem~\ref{thm:wintner} holds in the Rademacher model, so $\mu_x\to0$ almost surely, and the conditioning argument gives $\frac{T(x)}{\sqrt{V(x)}}\Rightarrow\mathcal N(0,1)$.
\end{solution}

\begin{solution}{ex:tablelink}
\idea Few distinct products force many repetitions: the fourth moment is the number of collisions in the multiplication table.
\writeup (a) Let $M=\lfloor x\rfloor$ and $r(N)=\#\{(a,b):a,b\le M,\ ab=N\}$. Then $\mathbb E|S(x)|^4=\#\{ab=cd\}=\sum_Nr(N)^2$, $\sum_Nr(N)=M^2$, and $r(N)>0$ for exactly $D(M)$ values of $N$. By Cauchy--Schwarz, $M^4=(\sum_Nr(N))^2\le D(M)\sum_Nr(N)^2$. With $D(M)\ll\frac{M^2}{(\log M)^\delta}$ this gives $\mathbb E|S(x)|^4\gg M^2(\log M)^\delta\gg x^2(\log x)^\delta$ for $x\ge3$. Theorem~\ref{thm:fourth} is much stronger, $x^2\log x$: the products are not only few, they are hit very unevenly. (b) With $c_1=1$, $c_m=2\varphi(m)$: $\sum_{m\le10}c_m\lfloor\frac{10}m\rfloor^2=100+2\cdot25+4\cdot9+4\cdot4+8\cdot4+4+12+8+12+8=278$. The Cauchy--Schwarz bound gives $\frac{10^4}{42}=238.1$, and independent points would give $190$. Already at $x=10$ the multiplicative structure adds almost $50\%$ to the fourth moment.
\end{solution}

\begin{solution}{ex:varV}
\idea Two partial sums $S(y)$ and $S(z)$ are correlated through the solutions of $ac=bd$, and summing these correlations over pairs of primes gives an extra $\log x$.
\writeup Let $2\le y\le z$. As in the proof of Theorem~\ref{thm:fourth}, $\mathbb E|S(y)|^2|S(z)|^2$ is the number of $(a,b,c,d)$ with $a,b\le y$, $c,d\le z$ and $ac=bd$; writing $a=gv$, $b=gw$ with $g=\gcd(a,b)$ and $\gcd(v,w)=1$, we get $c=wt$, $d=vt$, and the count is $\sum_{\gcd(v,w)=1}\lfloor\frac y{\max(v,w)}\rfloor\lfloor\frac z{\max(v,w)}\rfloor=\sum_{m\le y}c_m\lfloor\frac ym\rfloor\lfloor\frac zm\rfloor$. Using $\lfloor t\rfloor\ge\frac t2$ for $t\ge1$ and $c_m\ge\varphi(m)$, this is at least $\frac{yz}4\sum_{m\le y}\frac{\varphi(m)}{m^2}\ge c_1yz\log y$ by the proof of Theorem~\ref{thm:fourth}, for $y\ge y_0$. Now all terms of $V(x)^2=\sum_{p,q}|S(\frac xp)|^2|S(\frac xq)|^2$ are non-negative, so restricting to $\sqrt x<p,q\le x^{3/4}$, where $\frac xp,\frac xq\ge x^{1/4}$,
\[
\mathbb EV(x)^2\ge c_1\sum_{\sqrt x<p,q\le x^{3/4}}\frac{x^2}{pq}\cdot\frac{\log x}4=\frac{c_1}4x^2\log x\Bigl(\sum_{\sqrt x<p\le x^{3/4}}\frac1p\Bigr)^2,
\]
and the last sum tends to $\log\frac32$ by Mertens. Since $\mathbb EV(x)\sim x\log2$, $\mathrm{Var}\,V(x)=\mathbb EV(x)^2-(\mathbb EV(x))^2\ge c_2x^2\log x$ for $x$ large, and the ratio tends to infinity.
\end{solution}

\begin{solution}{ex:golden}
\idea Badly approximable numbers have small Weyl sums for every $k$.
\writeup $|S_k|=|\sum_{n\le N}e(nk\varphi)|=\frac{|\sin\pi Nk\varphi|}{|\sin\pi k\varphi|}\le\frac1{2\|k\varphi\|}\le\frac{3k}2$, using $|\sin\pi x|\ge2\|x\|$. (The bound $\|k\varphi\|\ge\frac1{3k}$ follows from $|k\varphi-m||k\varphi'-m|=|m^2-km-k^2|\ge1$, where $\varphi'=\frac{1-\sqrt5}2$, and $|k\varphi'-m|\le k(\varphi-\varphi')+\frac12=k\sqrt5+\frac12<3k$ when $|k\varphi-m|\le\frac12$.) Then $\sum_{k<K}\frac{|S_k|}{kN}\le\frac{3K}{2N}$, and Theorem~\ref{thm:erdosturan} gives $D_N\le\frac6{\sqrt K}+\frac{3K}{\pi N}$. With $K=\lceil N^{2/3}\rceil$, $D_N=O(N^{-1/3})$. (The truth is $O(\frac{\log N}N)$.)
\end{solution}

\begin{solution}{ex:catalan}
\idea The moments of a semicircle count balanced bracket sequences.
\writeup $(2\cos\theta)^{2m}=\sum_j\binom{2m}je^{i(2m-2j)\theta}$, whose integral against $\frac1\pi(1-\cos2\theta)\dd\theta$ on $[0,\pi]$ keeps the constant term $\binom{2m}m$ and subtracts half the coefficients of $e^{\pm2i\theta}$, namely $\frac12(\binom{2m}{m-1}+\binom{2m}{m+1})=\binom{2m}{m+1}$. So the moment is $\binom{2m}m-\binom{2m}{m+1}=\frac1{m+1}\binom{2m}m$. The odd moments vanish since $\theta\mapsto\pi-\theta$ preserves $\mu_{ST}$ and changes the sign of $\cos\theta$.
\end{solution}

\begin{solution}{ex:squarefreecount}
\idea Splitting off the largest square factor makes the count a convolution.
\writeup Write $f=g^2h$ with $h$ squarefree, $g$ monic; this is unique (each irreducible $\pi$ with $\pi^e\| f$ goes $\lfloor\frac e2\rfloor$ times into $g$). So $p^n=\sum_{k\ge0}p^ks_{n-2k}$, that is $\frac1{1-pt}=\frac{S(t)}{1-pt^2}$ for the generating function $S(t)=\sum s_nt^n$. Hence $S(t)=\frac{1-pt^2}{1-pt}$, and $s_n=p^n-p^{n-1}$ for $n\ge2$ (and $s_0=1$, $s_1=p$).
\end{solution}

\begin{solution}{ex:noroot}
\idea Prescribing roots is a linear condition, so inclusion--exclusion is exact.
\writeup For a set $R\subseteq\mathbb F_p$ of size $j\le n$, the monic $f$ of degree $n$ vanishing on $R$ are $\prod_{r\in R}(X-r)\,g$ with $g$ monic of degree $n-j$: $p^{n-j}$ of them. Since $n\ge p$, every $j\le p$ is allowed, and the number with no root is $\sum_{j=0}^p(-1)^j\binom pjp^{n-j}=p^n(1-\frac1p)^p$. As $p\to\infty$ this tends to $\frac1e$, and so does $\sum_{j\le n}\frac{(-1)^j}{j!}$ as $n\to\infty$; for $n<p$ the inclusion--exclusion stops at $j=n$ and gives $\sum_{j\le n}(-1)^j\binom pjp^{-j}$, which tends to the derangement probability $\sum_{j\le n}\frac{(-1)^j}{j!}$ as $p\to\infty$, as Theorem~\ref{thm:factortypes} predicts.
\end{solution}

\begin{solution}{ex:jordaninfinite}
\idea Triangularization is a conjugacy statement.
\writeup Let $M\in GL_2(\mathbb C)$ and $v$ an eigenvector, $Mv=\lambda v$. Complete $v$ to a basis $(v,w)$ and let $g$ be the matrix with columns $v,w$. Then $g^{-1}Mg$ has first column $(\lambda,0)$: it is upper triangular. So the conjugates of the subgroup $B$ of upper triangular matrices cover $GL_2(\mathbb C)$, although $B\ne GL_2(\mathbb C)$. The proof of Lemma~\ref{lem:jordan} breaks because there are infinitely many conjugates.
\end{solution}

\begin{solution}{ex:snrecognize}
\idea A $2$-transitive group with one transposition contains them all.
\writeup Let $c$ be the $(n-1)$-cycle, fixing $i$. The stabilizer $G_i$ contains $c$, hence is transitive on the other $n-1$ points; with the transitivity of $G$, $G$ acts transitively on ordered pairs of distinct points. If $(a\,b)\in G$ and $\{x,y\}$ is any pair, take $g\in G$ with $g(a)=x$, $g(b)=y$: then $g(a\,b)g^{-1}=(x\,y)\in G$. So $G$ contains all transpositions, and $G=S_n$. For the polynomial: irreducibility mod $p_1$ implies $f$ irreducible over $\Q$ (Lemma~\ref{lem:missingtype}), so $G_f$ is transitive, and Dedekind's theorem puts an $(n-1)$-cycle and a transposition in $G_f$.
\end{solution}
