\section{Exercises}

\begin{exercise}[$\star$]\label{ex:clcyclic}
Show that under $\mu_{CL}$ the probability that $X$ is cyclic (including trivial) is $c_\infty(1+\frac p{(p-1)^2})$. Evaluate it for $p=3$, and compare with the probability $c_\infty$ that $X$ is trivial.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:clrank}
(a) Show that the number of $n\times n$ matrices over $\mathbb F_p$ of rank $k$ is $\frac{\prod_{i=0}^{k-1}(p^n-p^i)^2}{\prod_{i=0}^{k-1}(p^k-p^i)}$, by counting pairs (image, map). (b) Deduce that the corank $r=n-k$ of a uniform random $n\times n$ matrix over $\mathbb F_p$ converges in law to $\frac{c_\infty p^{-r^2}}{\prod_{i=1}^r(1-p^{-i})^2}$, and that this is the law of the $p$-rank under $\mu_{CL}$.
\end{exercise}

\begin{exercise}[$\star$]\label{ex:clmean}
Show that under $\mu_{CL}$, $\mathbb E|X|=\infty$, but $\mathbb E[p^{\operatorname{rank}X}]=2$ and $\mathbb E[\#X[p]]=2$, where $X[p]$ is the $p$-torsion.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:clsur}
Verify directly, without matrices, that $\sum_{a\ge0}\mu_{CL}(\Z/p^a)\cdot\#\operatorname{Sur}(\Z/p^a,\Z/p)$ is not $1$, and explain why this does not contradict Corollary~\ref{cor:clmeasure}(b).
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:hbuniq}
(The case $e=1$, after Heath-Brown.) Let $K$ be a random non-negative integer, and suppose that $\mathbb E\bigl[\prod_{i=0}^{k-1}(p^K-p^i)\bigr]\le Cp^{\binom k2}$ for all $k$. Show, with the $q$-binomial inversion formula $\mathbf 1_{K=j}=\sum_{k\ge j}(-1)^{k-j}p^{\binom{k-j}2}\binom kj_p\binom Kk_p$, that these moments determine the law of $K$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:symrank}
Let $X=\operatorname{cok}M$ for a uniform random symmetric $n\times n$ matrix over $\Z/p^k$. Use Theorem~\ref{thm:symmoments} with $G=(\Z/p)^t$ to show that $\mathbb P(\operatorname{rank}X\ge t)\le c_\infty^{-1}p^{-t(t+1)/2}$. Compare with the bound $c_\infty^{-1}p^{-t^2}$ of Corollary~\ref{cor:clmeasure}(c) for matrices without symmetry.
\end{exercise}

\section{Hints}

\begin{hint}{ex:clcyclic}
$|\operatorname{Aut}(\Z/p^a)|=p^{a-1}(p-1)$; sum a geometric series.
\end{hint}
\begin{hint}{ex:clrank}
(a) Choose the image, a subspace $V$ of dimension $k=n-r$, then a surjection $\mathbb F_p^n\to V$. (b) Divide by $p^{n^2}$, and use $\#\{V\}=\frac{\prod_{i<k}(p^n-p^i)}{\prod_{i<k}(p^k-p^i)}$.
\end{hint}
\begin{hint}{ex:clmean}
\[\#X[p]=p^{\operatorname{rank}X}=\#\operatorname{Hom}(X,\Z/p)=1+\#\operatorname{Sur}(X,\Z/p).\]
\end{hint}
\begin{hint}{ex:clsur}
Non-cyclic groups also surject onto $\Z/p$.
\end{hint}
\begin{hint}{ex:hbuniq}
$\prod_{i<k}(p^K-p^i)=|GL_k(\mathbb F_p)|\binom Kk_p$, and $\binom kj_p\le c_\infty^{-1}p^{j(k-j)}$. Check that the series of absolute values converges, then exchange the sums.
\end{hint}

\begin{hint}{ex:symrank}
A group of rank at least $t$ has at least $|GL_t(\mathbb F_p)|$ surjections onto $(\Z/p)^t$, and $|\wedge^2(\Z/p)^t|=p^{\binom t2}$.
\end{hint}

\section{Solutions}

\begin{solution}{ex:clcyclic}
\idea Cyclic groups have few automorphisms, so they dominate.
\writeup $\mu_{CL}(\text{cyclic})=c_\infty\bigl(1+\sum_{a\ge1}\frac1{p^{a-1}(p-1)}\bigr)=c_\infty\bigl(1+\frac1{p-1}\cdot\frac1{1-1/p}\bigr)=c_\infty\bigl(1+\frac p{(p-1)^2}\bigr)$. For $p=3$: $0.5601\times1.75=0.9802$. So $98\%$ of the $3$-parts should be cyclic, although $44\%$ are non-trivial.
\end{solution}

\begin{solution}{ex:clrank}
\idea A matrix is its image together with a surjection onto it.
\writeup (a) A matrix of rank $k$ is a surjective linear map from $\mathbb F_p^n$ onto its image $V$, of dimension $k$; for a given $V$ there are $\prod_{i<k}(p^n-p^i)$ surjections (Proposition~\ref{prop:freesur} with $e=1$), and the number of $V$ is $\frac{\prod_{i<k}(p^n-p^i)}{\prod_{i<k}(p^k-p^i)}$ (ordered bases of $V$ in $\mathbb F_p^n$, divided by ordered bases of $\mathbb F_p^k$). The product is $\frac{\prod_{i<k}(p^n-p^i)^2}{\prod_{i<k}(p^k-p^i)}$. (b) Dividing by $p^{n^2}$: $\prod_{i<k}(p^n-p^i)^2=p^{2nk}\prod_{i<k}(1-p^{i-n})^2$ and $\prod_{i<k}(p^k-p^i)=p^{k^2}\prod_{j=1}^k(1-p^{-j})$. So the probability is $p^{2nk-k^2-n^2}\frac{\prod_{j=r+1}^n(1-p^{-j})^2}{\prod_{j=1}^k(1-p^{-j})}=p^{-r^2}\frac{\prod_{j=r+1}^n(1-p^{-j})^2}{\prod_{j=1}^{n-r}(1-p^{-j})}$, which tends to $p^{-r^2}\frac{c_\infty^2/\prod_{j\le r}(1-p^{-j})^2}{c_\infty}$. The $p$-rank of $\operatorname{cok}M$ is the corank of $M\bmod p$, since $\operatorname{cok}M/p\operatorname{cok}M=\operatorname{cok}(M\bmod p)$; by Theorem~\ref{thm:fw} and Corollary~\ref{cor:clmeasure}, the $p$-rank of a large random cokernel tends in law to that under $\mu_{CL}$, so the two laws agree.
\end{solution}

\begin{solution}{ex:clmean}
\idea Moments of size $1$ for every $G$ do not prevent a heavy tail for $|X|$.
\writeup $\mathbb E|X|\ge\sum_{a\ge1}p^a\mu_{CL}(\Z/p^a)=\sum_a\frac{c_\infty p}{p-1}=\infty$. On the other hand $\#X[p]=p^{\operatorname{rank}X}=\#\operatorname{Hom}(X,\Z/p)=1+\#\operatorname{Sur}(X,\Z/p)$, whose expectation is $1+1=2$ by Corollary~\ref{cor:clmeasure}(b).
\end{solution}

\begin{solution}{ex:clsur}
\idea The moment sees all groups, not only the cyclic ones.
\writeup For $a\ge1$, $\#\operatorname{Sur}(\Z/p^a,\Z/p)=p-1$, so the sum is $(p-1)\sum_{a\ge1}\frac{c_\infty}{p^{a-1}(p-1)}=\frac{c_\infty p}{p-1}$, which is $0.840$ for $p=3$, not $1$. The missing $0.160$ comes from the non-cyclic groups: $\Z/p\times\Z/p$ alone has $p^2-1$ surjections onto $\Z/p$ and probability $\frac{c_\infty}{p(p-1)^2(p+1)}$, contributing $\frac{c_\infty}{p(p-1)}=0.093$ for $p=3$, and so on.
\end{solution}

\begin{solution}{ex:hbuniq}
\idea Gaussian binomials of $p^K$ are to $p$-ranks what binomial coefficients are to integers.
\writeup Write $m_k=\mathbb E\binom Kk_p=\frac{\mathbb E\prod_{i<k}(p^K-p^i)}{|GL_k(\mathbb F_p)|}\le\frac{Cp^{\binom k2}}{c_\infty p^{k^2}}=C'p^{-\binom{k+1}2}$. By the inversion formula, $\mathbb P(K=j)=\mathbb E\sum_{k\ge j}(-1)^{k-j}p^{\binom{k-j}2}\binom kj_p\binom Kk_p$, and the exchange of $\mathbb E$ and $\sum$ is justified if $\sum_kp^{\binom{k-j}2}\binom kj_pm_k<\infty$. With $k=j+t$, the term is at most $c_\infty^{-1}C'p^{\binom t2+jt-\binom{j+t+1}2}=c_\infty^{-1}C'p^{-t-\binom{j+1}2}$, summable. So $\mathbb P(K=j)=\sum_k(-1)^{k-j}p^{\binom{k-j}2}\binom kj_pm_k$ is determined by the moments. (The inversion formula itself follows from the $q$-binomial theorem $\sum_{i}(-1)^ip^{\binom i2}\binom ni_p=\mathbf 1_{n=0}$, applied with $n=K-j$ after writing $\binom kj_p\binom Kk_p=\binom Kj_p\binom{K-j}{k-j}_p$.) This is the case $e=1$ of Theorem~\ref{thm:woodunique}, where the lexicographic induction is not needed.
\end{solution}

\begin{solution}{ex:symrank}
\idea Markov's inequality on the moment, as in Corollary~\ref{cor:clmeasure}(c); only the size of the moment changes.
\writeup If $\operatorname{rank}X\ge t$, then $X/pX\cong\mathbb F_p^r$ with $r\ge t$ maps onto $\mathbb F_p^t$ in at least $|GL_t(\mathbb F_p)|$ ways (compose one surjection with the automorphisms of $\mathbb F_p^t$). By Theorem~\ref{thm:symmoments} with $\lambda_1=\dots=\lambda_t=1$, $\mathbb E\#\operatorname{Sur}(X,(\Z/p)^t)\le p^{\binom t2}$. Markov's inequality gives $\mathbb P(\operatorname{rank}X\ge t)\le\frac{p^{\binom t2}}{|GL_t(\mathbb F_p)|}\le\frac{p^{\binom t2}}{c_\infty p^{t^2}}=c_\infty^{-1}p^{-t(t+1)/2}$. Without symmetry the exponent is $t^2$: symmetric cokernels have larger ranks, with tails like $p^{-t^2/2}$ instead of $p^{-t^2}$. The extra freedom is the pairing on the cokernel that a symmetric matrix carries.
\end{solution}

\section{What we proved, and what comes next}

The epigraph asked why $\Z/9$ should be eight times as common as $\Z/3\times\Z/3$ in class groups. The proved part of the answer is a theorem about random matrices: the cokernel of a large random matrix over $\Z/p^e$ is isomorphic to $G$ with probability tending to $\frac{c_\infty}{|\operatorname{Aut}G|}$ (Theorem~\ref{thm:fw}), and the ratio of $|\operatorname{Aut}|$ for the two groups is $48/6=8$. Behind it are the moments: every $\mathbb E\#\operatorname{Sur}(\operatorname{cok}M,G)$ tends to $1$ (Theorem~\ref{thm:cokmoments}), and Wood's theorem says that moments of size at most $|\wedge^2G|$ determine the law (Theorem~\ref{thm:woodunique}). For symmetric matrices the moments are $|\wedge^2G|$ (Theorem~\ref{thm:symmoments}), still within reach of Wood's theorem. The arithmetic part of the answer is a conjecture: the Cohen--Lenstra conjecture for class groups, of which only the case $G=\Z/3$ is a theorem, by Davenport and Heilbronn, quoted here.

The next chapter applies the same philosophy to elliptic curves. Their Selmer groups are modelled by the intersection of two random subspaces, and the model predicts the average rank.
