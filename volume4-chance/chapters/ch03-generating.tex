\chapter{Generating Functions and Random Families}

\begin{center}\itshape
A family name dies out, or it does not. The probability is the smallest root of $s=f(s)$,\\
where $f$ is the generating function of the number of children.
\end{center}

Francis Galton posed the question in 1873, in the \emph{Educational Times}: why do aristocratic surnames disappear? Henry William Watson answered him with the functional equation that this chapter solves. The same function packages a law on the nonnegative integers. For a sum of independent copies it multiplies, which is why it is the right bookkeeping.

\section{The probability generating function}

Let $X$ be a random variable with values in $\{0,1,2,\ldots\}$ and $P(X=k)=p_k$. The \emph{probability generating function} is
\[
 f(s)=\mathbb E[s^X]=\sum_{k\ge0}p_k s^k,\qquad s\in[0,1].
\]
The series converges at $s=1$ to $1$, and on $[0,1)$ it converges absolutely. Termwise differentiation on $[0,r]$ for any $0<r<1$ is legitimate: the differentiated series is uniformly bounded term by term by $k p_k r^{k-1}\le k r^{k-1}$, and $\sum_{k\ge1}kr^{k-1}=1/(1-r)^2$ by differentiating the geometric series of Volume~1 inside its interval of convergence. Thus $f$ is continuous on $[0,1]$, differentiable on $[0,1)$, and
\[
 f'(s)=\sum_{k\ge1}k p_k s^{k-1}.
\]

\begin{proposition}[mean from the derivative]\label{prop:pgfmean}
The limit $\lim_{s\uparrow1}f'(s)$ equals $\mathbb E[X]$, in $[0,\infty]$.
\end{proposition}
\begin{proof}
For $s\in(0,1)$ the terms $k p_k s^{k-1}$ increase to $k p_k$ as $s\uparrow1$. Since every term is nonnegative, $f'(s)\le\sum_{k\ge1}kp_k=\mathbb E[X]$. For every fixed $K$, $f'(s)\ge\sum_{k=1}^Kkp_ks^{k-1}$, whose limit is $\sum_{k=1}^Kkp_k$. Taking the supremum over $K$ gives the matching lower bound, even when the mean is infinite, by the nonnegative-series convention of Volume~1.
\end{proof}

In particular, if $\mathbb E[X]<\infty$, then $f'(1)$ exists as a left derivative and equals the mean. The second factorial moment is the same argument applied to $f''$, when one needs it. Exercise~\ref{ex:randomsumvar} records the variance formula.

\begin{example}[three laws]\label{ex:pgfexamples}
A Bernoulli variable of parameter $p$ has $f(s)=1-p+ps$. A binomial count of $n$ trials has $f(s)=(1-p+ps)^n$, by the binomial theorem. A Poisson variable of parameter $\lambda$ has
\[
 f(s)=\sum_{k\ge0}e^{-\lambda}\frac{\lambda^k}{k!}s^k=e^{\lambda(s-1)}.
\]
Differentiating the Poisson function gives $f'(s)=\lambda e^{\lambda(s-1)}$, so $f'(1)=\lambda$, which matches Volume~1.
\end{example}

\section{Sums}

\begin{proposition}[independent sums]\label{prop:pgfsum}
If $X$ and $Y$ are independent and nonnegative integer valued, the generating function of $X+Y$ is the product of the generating functions. The same holds for any finite independent sum.
\end{proposition}
\begin{proof}
Independence gives $P(X=i,Y=j)=P(X=i)P(Y=j)$. Therefore
\[
 \mathbb E[s^{X+Y}]=\sum_{i,j}P(X=i)P(Y=j)s^{i+j}=\mathbb E[s^X]\,\mathbb E[s^Y].
\]
The double sum is absolutely convergent for $s\in[0,1]$, so the product of the sums is the sum of the products.
\end{proof}

A binomial count is a sum of $n$ independent Bernoulli variables, and $(1-p+ps)^n$ is the product of the factors $1-p+ps$. A sum of independent Poisson variables of parameters $\lambda$ and $\mu$ has generating function $e^{\lambda(s-1)}e^{\mu(s-1)}=e^{(\lambda+\mu)(s-1)}$, which is the Poisson$(\lambda+\mu)$ function. That is the generating-function proof of Exercise~\ref{ex:twopoisson}.

\begin{proposition}[random sums]\label{prop:randomsum}
Let $N$ be nonnegative and integer valued. Conditional on $N=n$, draw $n$ independent, identically distributed nonnegative integer-valued variables with a fixed common law that does not depend on $n$. Let $S$ be their sum, with $S=0$ when $n=0$. If $g$ is the generating function of $N$ and $f$ is the generating function of one summand, then $S$ has generating function $g(f(s))$.
\end{proposition}
\begin{proof}
Condition on $N=n$. For $n=0$, $\mathbb E[s^S\mid N=0]=1$. For $n\ge1$, independence and Proposition~\ref{prop:pgfsum} give $\mathbb E[s^S\mid N=n]=f(s)^n$. Therefore
\[
 \mathbb E[s^S]=\sum_{n\ge0}P(N=n)f(s)^n=g\bigl(f(s)\bigr).
\]
There is no need to assume that a countably infinite Cartesian product is countable (it need not be). Sample $N$ first and then its finite list of summands: the disjoint union over $n\ge0$ of these finite lists is a countable sample space. Its masses are $P(N=n)$ multiplied by the conditional finite-list masses, and their total is $\sum_nP(N=n)=1$. The displayed calculation uses only nonnegative sums on that space.
\end{proof}

\section{Galton--Watson extinction}

Start with one individual. Each individual has a number of children distributed as $X$, independently of everyone else, with $P(X=k)=p_k$ and generating function $f$. Let $Z_n$ be the number of individuals in generation $n$, so $Z_0=1$ and, given $Z_n=m$, the next generation is a sum of $m$ independent copies of $X$. Let $\eta=P(Z_n=0\text{ for some }n)$ be the probability that the line dies out.

\begin{theorem}[extinction]\label{thm:extinction}
The extinction probability $\eta$ is the smallest root of $s=f(s)$ in $[0,1]$. If $\mathbb E[X]\le1$ and $P(X=1)<1$, then $\eta=1$. If $\mathbb E[X]>1$, then $\eta<1$.
\end{theorem}
\begin{proof}
Let $\eta_n=P(Z_n=0)$. Then $\eta_0=0$ and $\eta_1=p_0=f(0)$. Conditioning on the first generation, $Z_n$ is the sum of $Z_1$ independent copies of a process with the same law, so $\eta_{n}=f(\eta_{n-1})$ for $n\ge1$. Since $f$ is increasing on $[0,1]$ and $\eta_1=f(0)\ge0=\eta_0$, induction gives $\eta_n\le\eta_{n+1}\le1$. The limit $L=\lim\eta_n$ satisfies $L=f(L)$ by continuity of $f$.

Extinction by time $n$ depends only on the first $n$ finite generations, whose joint laws are determined by the offspring masses. Let $D_j$ be the event of \emph{first} extinction in generation $j$. These events are disjoint and $\eta_n=\sum_{j=1}^nP(D_j)$. The event of eventual extinction is their countable union, with probability defined by the nonnegative sum $\eta=\sum_{j\ge1}P(D_j)$. Hence $\eta=\lim_n\eta_n=L$. No conclusion about an increasing union is drawn from finite additivity alone.

Now let $r\in[0,1]$ be any root of $s=f(s)$. Then $\eta_0=0\le r$, and if $\eta_{n-1}\le r$ then $\eta_n=f(\eta_{n-1})\le f(r)=r$. So $L\le r$. The limit is the smallest root.

It remains to locate that root. The function $f(s)-s$ has $f(1)-1=0$. Suppose $\mathbb E[X]=m$ exists in $[0,\infty]$. If $m>1$, then the nonnegative-series identity $(1-f(s))/(1-s)=\sum_{k\ge1}p_k(1+s+\cdots+s^{k-1})$ tends to $m$ as $s\uparrow1$ (by the same finite-partial-sum argument as Proposition~\ref{prop:pgfmean}), so $1-f(s)>1-s$ for $s$ close to $1$, that is $f(s)<s$ on a left neighbourhood of $1$. Combined with $f(0)\ge0$, the intermediate-value theorem supplies a root in $[0,1)$. The smallest root is then strictly less than $1$, so $\eta<1$.

If $m<1$, the same expansion gives
\[
 \frac{1-f(s)}{1-s}=\sum_{k\ge1}p_k(1+s+\cdots+s^{k-1})\le\sum_{k\ge1}kp_k=m<1
\]
for $s\in[0,1)$, so $f(s)>s$. There is no root in $[0,1)$, and $\eta=1$.

If $m=1$ and $P(X=1)<1$, then for $s\in[0,1)$,
\[
 \frac{1-f(s)}{1-s}=\sum_{k\ge1}p_k(1+s+\cdots+s^{k-1}).
\]
Each inner sum is at most $k$, and it is strictly smaller than $k$ whenever $k\ge2$. The hypothesis $P(X=1)<1$ together with mean $1$ forces some $p_k>0$ for a $k\ne1$. The case of mass only at $0$ and $1$ with mean $1$ is $p_1=1$, which is excluded. So the displayed average is strictly less than $\sum k p_k=1$, hence $f(s)>s$ on $[0,1)$. There is no root in $[0,1)$, and $\eta=1$.

If $P(X=1)=1$, then $f(s)=s$, every point is a root, the smallest root is $0$, and the population stays at $1$, so $\eta=0$. The hypothesis $P(X=1)<1$ excludes this case from the sentence $\eta=1$.
\end{proof}

Chapter~14 compares a sparse random graph with this process. The subcritical regime is proved there by a first-moment count of isolated trees. The survival probability above is the candidate for the fraction of vertices in a giant component. The only limit proved in that direction is a positive lower bound on the probability that one given vertex has a linear component.

\begin{example}[two children or none]\label{ex:binary}
Suppose $P(X=0)=1-p$ and $P(X=2)=p$, with $0<p\le1$. Then $f(s)=1-p+ps^2$ and $\mathbb E[X]=2p$. The fixed-point equation is
\[
 f(s)-s=1-p+ps^2-s=(1-s)\bigl(1-p(1+s)\bigr).
\]
The roots are $s=1$ and, when $p>0$, $s=(1-p)/p$. If $p\le1/2$, then $(1-p)/p\ge1$, so the smallest root in $[0,1]$ is $1$ and the line dies almost surely. If $p>1/2$, the smallest root is $(1-p)/p<1$. For $p=2/3$ that root is $1/2$.
\end{example}

\section{Catalan numbers}

Let $C_0=1$ and, for $n\ge1$, let $C_n$ be the number of Dyck paths of semilength $n$, from Exercise~\ref{ex:catalan}. A nonempty Dyck path is $X$, then a Dyck path, then $Y$, then a Dyck path: the first return to equal counts splits the path. Therefore
\[
 C_n=\sum_{i=0}^{n-1}C_i C_{n-1-i},\qquad n\ge1.
\]
Since $0\le C_n\le\binom{2n}{n}\le4^n$, the series $c(x)=\sum_{n\ge0}C_nx^n$ converges absolutely for $|x|<1/4$. We may multiply its series and collect terms there. Multiply the recurrence by $x^n$ and sum:
\[
 c(x)-1=x\,c(x)^2.
\]
The quadratic equation $x c^2-c+1=0$ has solutions $(1\pm\sqrt{1-4x})/(2x)$ for $x\ne0$. The solution that stays bounded as $x\downarrow0$ is the one with the minus sign, because $\sqrt{1-4x}=1-2x+O(x^2)$ and $(1-(1-2x))/(2x)=1$. Thus
\[
 c(x)=\frac{1-\sqrt{1-4x}}{2x},\qquad 0<|x|<\frac14,
\]
and $c(0)=1$. Exercise~\ref{ex:binseries} identifies the series of $(1-4x)^{-1/2}$ and recovers $C_n=\binom{2n}{n}/(n+1)$ from this formula, which matches the count in Chapter~1.

The first-step split is the same idea as a random walk's first return. Chapter~8 counts those returns. The generating function here is only the enumeration.

\section{Exercises}

\begin{exercise}[$\star$]\label{ex:thinning}
Each point of a Poisson$(\lambda)$ count is kept with probability $p$, independently. Show that the number kept is Poisson$(\lambda p)$.
\end{exercise}

\begin{exercise}[$\star$]\label{ex:golden}
In Example~\ref{ex:binary} with $p=2/3$, the extinction probability is $1/2$. Check the fixed-point equation numerically at $s=1/2$, and compute the mean number of children.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:lotka}
Take $p_0=1/4$, $p_1=1/2$, $p_2=1/4$. Find the extinction probability and the mean.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:randomsumvar}
Under the hypotheses of Proposition~\ref{prop:randomsum}, assume $\mathbb E[X_1^2]$ and $\mathbb E[N^2]$ are finite. Show that $\mathbb E[S]=\mathbb E[N]\,\mathbb E[X_1]$ and
\[
 \operatorname{Var}S=\mathbb E[N]\,\operatorname{Var}X_1+\operatorname{Var}N\,(\mathbb E[X_1])^2.
\]
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:progeny}
Let $T=Z_0+Z_1+\cdots$ be the total progeny in a Galton--Watson process, including the founder, and let $h(s)=\mathbb E[s^T\mathbf 1_{T<\infty}]$. Show that $h(s)=s f(h(s))$ for $s\in[0,1]$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:firstpassage}
A walk on the integers starts at $0$ and steps $+1$ with probability $p\in(0,1)$ and $-1$ with probability $q=1-p$, independently. Let $\tau$ be the first time the walk reaches $+1$, or $\infty$ if it never does. Let $F(s)=\mathbb E[s^\tau\mathbf 1_{\tau<\infty}]$. Show that $F(s)=ps+qs F(s)^2$ for $s\in[0,1)$, and that $P(\tau<\infty)=\min(1,p/q)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:binseries}
Let $a_n=\binom{2n}{n}$ and $g(x)=\sum_{n\ge0}a_n x^n$ for $|x|<1/4$.
\begin{enumerate}[label=(\alph*)]
\item Show that $a_n=2(2n-1)a_{n-1}/n$ for $n\ge1$, and deduce $(1-4x)g'(x)=2g(x)$.
\item Conclude that $g(x)=(1-4x)^{-1/2}$.
\item Show that $\binom{-1/2}{j}(-4)^j=\binom{2j}{j}$ for $j\ge0$.
\item From $c(x)=(1-\sqrt{1-4x})/(2x)$ and part~(b), recover $C_n=\binom{2n}{n}/(n+1)$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{ex:poissongw}
Let the offspring law be Poisson$(c)$ with $c>1$, and let $\rho=1-\eta$ be the survival probability. Show that $\rho$ is the unique root in $(0,1]$ of $1-\rho=e^{-c\rho}$, and that $\rho\sim 2(c-1)$ as $c\downarrow1$.
\end{exercise}

\section{Solutions}

\begin{solution}{ex:thinning}
\idea A binomial thinning of a Poisson generating function is a shift of the exponent.
\writeup Given $N=n$, the number kept is binomial$(n,p)$, with generating function $(1-p+ps)^n$. Remove the conditioning:
\[
 \sum_{n\ge0}e^{-\lambda}\frac{\lambda^n}{n!}(1-p+ps)^n=\exp\bigl(\lambda(1-p+ps)-\lambda\bigr)=\exp(\lambda p(s-1)).
\]
That is the Poisson$(\lambda p)$ function.
\end{solution}

\begin{solution}{ex:golden}
\idea One substitution checks a root. The mean is the derivative, or a direct sum.
\writeup With $p=2/3$, $f(1/2)=1-2/3+(2/3)(1/4)=1/3+1/6=1/2$. The mean is $2p=4/3>1$, in agreement with survival probability $1-1/2=1/2$.
\end{solution}

\begin{solution}{ex:lotka}
\idea The mean equals $1$ and the distribution is not a single child for certain, so Theorem~\ref{thm:extinction} gives extinction probability $1$.
\writeup $f(s)=\tfrac14+\tfrac12 s+\tfrac14 s^2=\bigl((1+s)/2\bigr)^2$. Then $f'(s)=(1+s)/2$ and $f'(1)=1$. Since $p_1=1/2<1$, the theorem yields $\eta=1$. The equation $s=f(s)$ is $(1-s)^2=0$, a double root at $1$, so there is no smaller root in $[0,1)$.
\end{solution}

\begin{solution}{ex:randomsumvar}
\idea Condition on $N$. The conditional mean is linear in $N$, and the conditional variance is $N$ times one variance.
\writeup Given $N=n$, $S$ is a sum of $n$ independent copies, so $\mathbb E[S\mid N=n]=n\mu$ and $\operatorname{Var}(S\mid N=n)=n\sigma^2$, where $\mu=\mathbb E[X_1]$ and $\sigma^2=\operatorname{Var}X_1$. Therefore $\mathbb E[S]=\mathbb E[\mathbb E[S\mid N]]=\mu\mathbb E[N]$. The law of total variance, which is the computation
\[
 \mathbb E[S^2]=\mathbb E\bigl[\mathbb E[S^2\mid N]\bigr]=\mathbb E\bigl[\operatorname{Var}(S\mid N)+(\mathbb E[S\mid N])^2\bigr]
 =\sigma^2\mathbb E[N]+\mu^2\mathbb E[N^2],
\]
gives $\operatorname{Var}S=\sigma^2\mathbb E[N]+\mu^2\mathbb E[N^2]-\mu^2(\mathbb E[N])^2=\mathbb E[N]\sigma^2+(\mathbb E[X_1])^2\operatorname{Var}N$.
\end{solution}

\begin{solution}{ex:progeny}
\idea One individual, then a random number of independent copies of the same total.
\writeup If the founder has $k$ children, the total progeny is $1$ plus $k$ independent copies of $T$. On the event that every copy is finite,
\[
 \mathbb E[s^T\mathbf 1_{T<\infty}]=\sum_{k\ge0}p_k s\,\bigl(h(s)\bigr)^k=s f\bigl(h(s)\bigr).
\]
The indicator of finiteness of the whole line is the indicator that every subtree is finite.
\end{solution}

\begin{solution}{ex:firstpassage}
\idea One step up finishes the wait. One step down leaves two successive climbs, each distributed as $\tau$.
\writeup With probability $p$ the first step is $+1$ and $\tau=1$. With probability $q$ the first step is $-1$, the walk is at $-1$, and the time to reach $+1$ is a copy of $\tau$ to get back to $0$, plus an independent copy to reach $+1$. Therefore
\[
 F(s)=ps\cdot1+qs\,F(s)\,F(s).
\]
The quadratic $qs F^2-F+ps=0$ gives $F(s)=\bigl(1-\sqrt{1-4pqs^2}\bigr)/(2qs)$ for $q>0$, taking the root that vanishes at $s=0$. Let $s\uparrow1$. Then $P(\tau<\infty)=F(1-)$ equals the smaller of $1$ and the value $(1-|p-q|)/(2q)$. If $p\ge q$, then $|p-q|=p-q$ and $(1-(p-q))/(2q)=(1-p+q)/(2q)=(2q)/(2q)=1$. If $p<q$, then $|p-q|=q-p$ and $(1-(q-p))/(2q)=(1-q+p)/(2q)=(2p)/(2q)=p/q$. So $P(\tau<\infty)=\min(1,p/q)$.
\end{solution}

\begin{solution}{ex:binseries}
\idea The central binomial coefficients satisfy a first-order ratio. That ratio is a differential equation, and the equation has one bounded solution.
\writeup (a) For $n\ge1$,
\[
 \frac{a_n}{a_{n-1}}=\frac{(2n)!}{(n!)^2}\cdot\frac{((n-1)!)^2}{(2n-2)!}=\frac{(2n)(2n-1)}{n^2}=\frac{2(2n-1)}{n}.
\]
Hence $n a_n=2(2n-1)a_{n-1}=(4n-2)a_{n-1}$. On $|x|<1/4$ the ratio test gives a positive radius of convergence $1/4$, because $a_n/a_{n-1}\to4$. Termwise differentiation is legitimate on $|x|\le r<1/4$ by the comparison just made. The coefficient of $x^{n-1}$ in $(1-4x)g'(x)$ is $n a_n-4(n-1)a_{n-1}=(4n-2)a_{n-1}-4(n-1)a_{n-1}=2a_{n-1}$, which is the coefficient of $x^{n-1}$ in $2g(x)$.

(b) Let $u(x)=(1-4x)^{1/2}g(x)$ for $|x|<1/4$. Differentiating the product and using $g'=2g/(1-4x)$ gives
\[
 u'=(-2)(1-4x)^{-1/2}g+(1-4x)^{1/2}\cdot\frac{2g}{1-4x}
 =(-2)(1-4x)^{-1/2}g+2(1-4x)^{-1/2}g=0.
\]
So $u$ is constant. At $x=0$ it equals $1$. Therefore $g(x)=(1-4x)^{-1/2}$.

(c) 
\[
 \binom{-1/2}{j}=\frac{\bigl(-\tfrac12\bigr)\bigl(-\tfrac32\bigr)\cdots\bigl(\tfrac12-j\bigr)}{j!}
 =\frac{(-1)^j\,1\cdot3\cdot5\cdots(2j-1)}{2^j\,j!}.
\]
Multiply by $(-4)^j=(-1)^j 2^{2j}$:
\[
 \binom{-1/2}{j}(-4)^j=\frac{2^j\,1\cdot3\cdots(2j-1)}{j!}.
\]
And $\binom{2j}{j}=(2j)!/(j!)^2$ equals the same quantity, because $(2j)!=2^j j!\,1\cdot3\cdots(2j-1)$.

(d) From the chapter, $xc(x)=(1-\sqrt{1-4x})/2$. Differentiate:
\[
 \frac{\mathrm d}{\mathrm dx}\bigl(xc(x)\bigr)=\frac{1}{\sqrt{1-4x}}=g(x).
\]
The series $xc(x)=\sum_{n\ge0}C_n x^{n+1}$ differentiates to $\sum_{n\ge0}(n+1)C_n x^n$. Comparing coefficients with $g$ gives $(n+1)C_n=\binom{2n}{n}$.
\end{solution}

\begin{solution}{ex:poissongw}
\idea The Poisson generating function is exponential, so the fixed-point equation is transcendental and still elementary.
\writeup Here $f(s)=\exp(c(s-1))$. The equation $s=f(s)$ is $s=e^{c(s-1)}$. Set $\rho=1-s$. Then $1-\rho=\exp(-c\rho)$. At $\rho=0$ both sides equal $1$. The derivative of the right side at $0$ is $-c e^{0}=-c$, and the derivative of the left side is $-1$. For $c>1$ the graphs meet again in $(0,1)$: consider $\phi(\rho)=e^{-c\rho}-(1-\rho)$. Then $\phi(0)=0$, $\phi'(0)=1-c<0$, so $\phi$ is negative just to the right of $0$, and $\phi(1)=e^{-c}>0$, so a root exists in $(0,1)$. Uniqueness: $\phi''(\rho)=c^2 e^{-c\rho}>0$, so $\phi'$ has at most one zero and $\phi$ has at most two zeros. One is at $0$, so there is exactly one in $(0,1)$. Theorem~\ref{thm:extinction} says $\eta$ is the smallest root of $s=f(s)$ in $[0,1]$, hence $\eta=1-\rho$ for this $\rho$, and the survival probability is $\rho$.

For the slope, differentiate $1-\rho=e^{-c\rho}$ with respect to $c$, treating $\rho=\rho(c)$:
\[
 -\rho'=e^{-c\rho}(-\rho-c\rho')=-(1-\rho)(\rho+c\rho').
\]
At $c=1$, $\rho=0$ is the only root, and the equation becomes indeterminate. Set $\rho=(c-1)\alpha(c)$ with $\alpha$ to be found, or expand $e^{-c\rho}=1-c\rho+c^2\rho^2/2+O(\rho^3)$. Then $1-\rho=1-c\rho+c^2\rho^2/2+O(\rho^3)$, so $\rho(c-1)=c^2\rho^2/2+O(\rho^3)$. For $\rho\ne0$, $c-1=c^2\rho/2+O(\rho^2)$. As $c\downarrow1$, $\rho\to0$: any subsequential limit $r\in[0,1]$ would solve $1-r=e^{-r}$, whose only solution is $r=0$ (since $e^{-r}>1-r$ for $r>0$). The Taylor remainder is uniform for $c\in[1,2]$ and $0\le\rho\le1$, so the preceding identity then gives $\rho\sim2(c-1)$.
\end{solution}
