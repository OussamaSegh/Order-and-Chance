\chapter{Random Variables and Their Laws}

\begin{center}\itshape
Shuffle $52$ cards. The expected number of cycles is the harmonic number\\
$H_{52}\approx4.538$, and you do not need the law of that number to see it.
\end{center}

A random variable is a function from the sample space to the real numbers. Volume~1, Chapter~10, defined it on a finite set, proved that expectation is linear with no independence hypothesis, and computed the classical finite laws. The same chapter built a countable space as a convergent series of masses, and on that space it defined the geometric waiting time and the Poisson masses. Two jobs were left open. The Poisson masses were not obtained as a limit of binomial laws, and the number of cycles in a random permutation was not averaged. Both are sums, or limits of sums. Listing the law is the long way round.

\section{Laws on a countable space}

Let $\Omega=\{\omega_1,\omega_2,\ldots\}$ with masses $p_j\ge0$ summing to $1$, as in Volume~1, Chapter~10. A random variable $X$ is a function $\Omega\to\R$. The \emph{law} of $X$ is the function $x\mapsto P(X=x)$. The events $\{X=x\}$ partition $\Omega$, so the masses of the law sum to $1$.

If $\sum_j |X(\omega_j)|p_j<\infty$, the expectation is the absolutely convergent series $\mathbb E[X]=\sum_j X(\omega_j)p_j$. Absolute convergence is what permits regrouping. Volume~1 records the nonnegative case as a supremum of finite sub-sums, and the signed case under this hypothesis. Linearity for a finite sum of variables with finite expectations reduces to the finite case and a limit. The same reduction gives $\mathbb E[X]=\sum_x x\,P(X=x)$ whenever $\sum_x |x|P(X=x)<\infty$.

Variables that depend on finitely many digits of the uniform number in Chapter~1 have finite expectations, computed as finite sums. The same uniform point also supplies independent choices among any $r\ge2$ equally likely outcomes: write it in base $r$, choosing the expansion ending in zeros at an endpoint. Every prescribed string of $m$ base-$r$ digits is one half-open interval of length $r^{-m}$, so its digits are independent and uniform on $\{0,\ldots,r-1\}$. The coupon draws below use that model. This chapter stays with variables whose expectations converge absolutely.

\begin{definitionx}[variance]
If $\mathbb E[X^2]<\infty$, then $\operatorname{Var}X=\mathbb E[(X-\mathbb EX)^2]=\mathbb E[X^2]-(\mathbb EX)^2$. The identity is the expansion of the square.
\end{definitionx}

Volume~1 proves that a finite sum of pairwise independent random variables with finite variances has variance equal to the sum of the variances. Dependence can make the variance smaller, as in sampling without replacement, or leave it unchanged, as for the fixed points below. Expectation never asks which of these is happening.

\section{The Poisson limit}

Volume~1 writes the masses $e^{-\lambda}\lambda^k/k!$ and checks that they sum to $1$, with mean and variance $\lambda$. They arise when successes are rare.

\begin{theorem}[Poisson limit of binomial laws]\label{thm:poissonlimit}
Let $\lambda>0$, and let $p_n\in[0,1]$ satisfy $np_n\to\lambda$. For each fixed integer $k\ge0$,
\[
 \binom{n}{k}p_n^k(1-p_n)^{n-k}\to e^{-\lambda}\frac{\lambda^k}{k!}.
\]
\end{theorem}
\begin{proof}
Since $np_n\to\lambda>0$, one has $p_n\to0$. For every large $n$ this forces $0<p_n<1$: otherwise $np_n$ could not approach a positive limit. The difference quotient of $\ln$ at $1$, along the increment $-x$, is $(\ln(1-x)-\ln 1)/(-x)$. Volume~1, Chapter~3, says this tends to $\ln'(1)=1$. Multiplying by $-1$ gives $\ln(1-x)/x\to-1$ as $x\to0$:
\[
 \frac{\ln(1-x)}{x}
 =\frac{\ln(1-x)-\ln 1}{-x}\cdot(-1)\to-1
\]
as $x\to0$. For $k=0$ the probability is $(1-p_n)^n$, so
\[
 n\ln(1-p_n)=(np_n)\cdot\frac{\ln(1-p_n)}{p_n}\to\lambda\cdot(-1)=-\lambda,
\]
and $(1-p_n)^n\to e^{-\lambda}$.

For $k\ge1$ and $n\ge k$,
\[
 \binom{n}{k}p_n^k(1-p_n)^{n-k}
 =\frac1{k!}\Biggl(\prod_{j=0}^{k-1}(np_n-jp_n)\Biggr)\,(1-p_n)^{n-k}.
\]
Each term $np_n-jp_n$ tends to $\lambda$. The product tends to $\lambda^k$. And $(1-p_n)^{n-k}=(1-p_n)^n(1-p_n)^{-k}\to e^{-\lambda}$, because $(1-p_n)^{-k}\to1$.
\end{proof}

\begin{example}[one hundred rare trials]\label{ex:rare}
Take $n=100$ and $p=0.02$, so $np=2$. The probability of no success is $(0.98)^{100}$. The integral form of Taylor's formula, Volume~1, Chapter~8, at order $1$ and centre $0$, gives, for $0<x<1$,
\[
 \ln(1-x)=-x-\int_0^x\frac{x-t}{(1-t)^2}\,\mathrm dt.
\]
On $[0,x]$ the weight $x-t$ is nonnegative, and $1\le 1/(1-t)^2\le 1/(1-x)^2$. The integral of $x-t$ from $0$ to $x$ is $x^2/2$, so
\[
 \frac{x^2}2\le\int_0^x\frac{x-t}{(1-t)^2}\,\mathrm dt\le\frac{x^2}{2(1-x)^2}.
\]
The left inequality is strict. On $[\tfrac x2,\tfrac{3x}4]$ one has $x-t\ge x/4$ and $1/(1-t)^2-1\ge 1/(1-x/2)^2-1>0$. The difference of the integrands is continuous and at least the positive constant $(x/4)\bigl(1/(1-x/2)^2-1\bigr)$ on an interval of length $x/4$. Volume~1's comparison of integrals then makes the difference of the two integrals positive. Therefore
\[
 x+\frac{x^2}2<-\ln(1-x)\le x+\frac{x^2}{2(1-x)^2}.
\]
For $x=0.02$ the two ends are $0.0202$ and $0.02+0.0004/(2\cdot0.98^2)$. The second number is $0.0202082\ldots$, and $100$ times it is less than $2.02083$. Multiplying the inequality by $100$,
\[
 2.02<-\,100\ln(0.98)<2.02083,
\]
so
\[
 e^{-2.02083}<(0.98)^{100}<e^{-2.02}.
\]
The bounds give $0.132545\ldots<(0.98)^{100}<0.132655\ldots$. Direct evaluation gives $(0.98)^{100}=0.132619\ldots$, while the Poisson mass is $e^{-2}=0.135335\ldots$. The gap at $n=100$ is about $0.0027$. The bounds alone do not fix the fourth decimal place. Theorem~\ref{thm:poissonlimit} says the gap closes as $n$ grows with $np$ held at $2$. It does not say the gap is already invisible.
\end{example}

\begin{examplex}[an explicit gap for no successes]
The same integral bound gives a rate, still only for the probability of no success. Let $\lambda>0$ and let $n>\lambda$ be an integer. Set $x=\lambda/n$. The inequality of Example~\ref{ex:rare} yields
\[
 \lambda+\frac{\lambda^2}{2n}
 <-n\ln\Bigl(1-\frac\lambda n\Bigr)
 \le\lambda+\frac{\lambda^2}{2n(1-\lambda/n)^2}.
\]
Exponentiating and reversing the inequalities,
\[
 \exp\Bigl(-\lambda-\frac{\lambda^2}{2n(1-\lambda/n)^2}\Bigr)
 \le\Bigl(1-\frac\lambda n\Bigr)^n
 <e^{-\lambda}.
\]
Write $a=\lambda^2/(2n(1-\lambda/n)^2)>0$. The mean value theorem in Volume~1, Chapter~3, gives $e^{-a}=1-ae^{\xi}$ for some $\xi\in(-a,0)$. Since $e^{\xi}<1$, one has $e^{-a}>1-a$, so $1-e^{-a}<a$. Therefore
\[
 0<e^{-\lambda}-\Bigl(1-\frac\lambda n\Bigr)^n
 \le e^{-\lambda}\frac{\lambda^2}{2n(1-\lambda/n)^2}.
\]
For $\lambda=2$ and $n=100$ the upper bound is about $0.00282$, and the gap computed above is about $0.00272$. Chapter~10 replaces the comparison of the whole laws by Le Cam's inequality, an explicit bound in total variation. The bound here is only the $k=0$ term.
\end{examplex}

Fixed points are the next count of the same kind. Let $F_n$ be the number of fixed points of a uniform random permutation. Volume~1 shows $\mathbb E[F_n]=1$, and that $P(F_n=0)=D_n/n!$ differs from $e^{-1}$ by at most $1/(n+1)!$. The full law converges as well.

\begin{proposition}[fixed points tend to Poisson]\label{prop:fixedpoisson}
For each fixed integer $m\ge0$,
\[
 P(F_n=m)\to e^{-1}\frac1{m!}.
\]
\end{proposition}
\begin{proof}
A permutation of $n$ elements with exactly $m$ fixed points is counted by choosing the $m$ points and deranging the rest:
\[
 P(F_n=m)=\binom{n}{m}\frac{D_{n-m}}{n!}=\frac1{m!}\cdot\frac{D_{n-m}}{(n-m)!},
\]
for $n\ge m$, and $D_0=1$. Volume~1, Chapter~2, gives $D_r/r!=\sum_{j=0}^r(-1)^j/j!$, and Chapter~10 of the same volume compares the partial sum with $e^{-1}$. The difference tends to $0$ as $r=n-m\to\infty$.
\end{proof}

\section{Cycles}

Let $\pi$ be uniform on the permutations of $\{1,\ldots,n\}$. Let $X_k$ be the number of cycles of length $k$, and let $C_n=X_1+\cdots+X_n$.

\begin{figure}[ht]
\centering
\fig{ch02-cycles}
\caption{The expected number of cycles of length $k$ is $1/k$. The bars are the first eight terms, $k=1,\ldots,8$. The expected number of cycles of all lengths in a permutation of $n$ letters is the whole sum $H_n=\sum_{k=1}^n 1/k$, not the sum of these eight bars alone.}
\label{fig:cycles}
\end{figure}

\begin{proposition}[mean number of cycles]\label{prop:cycles}
For each $k\in\{1,\ldots,n\}$, $\mathbb E[X_k]=1/k$. Therefore $\mathbb E[C_n]=H_n$.
\end{proposition}
\begin{proof}
Choose $k$ elements in $\binom{n}{k}$ ways, arrange them in a cycle in $(k-1)!$ ways, and permute the rest in $(n-k)!$ ways. Each permutation is counted once for every cycle of length $k$ that it has, so
\[
 \mathbb E[X_k]=\binom{n}{k}\frac{(k-1)!(n-k)!}{n!}=\frac1k.
\]
Sum over $k$. For a deck of $52$ cards the integral test in Volume~1, Chapter~3, compares a decreasing positive function with the areas of the unit bars. Applied to $1/x$, it gives $\ln53\le H_{52}\le1+\ln52$: the sum $\sum_{k=1}^{52}1/k$ sits between $\int_1^{53}x^{-1}\,\mathrm dx$ and $1+\int_1^{52}x^{-1}\,\mathrm dx$. Those bounds are $\ln53\approx3.970$ and $1+\ln52\approx4.951$. The sum itself is $4.538\ldots$. A shuffled deck has, on average, fewer than five cycles.
\end{proof}

\begin{examplex}[six permutations, counted by hand]
The same count can be checked by writing every permutation of $\{1,2,3\}$. The identity has three cycles. Each of the three transpositions has two: the $2$-cycle and one fixed point. Each of the two $3$-cycles is a single cycle. The total is $3+3\cdot2+2\cdot1=11$, so the mean number of cycles is $11/6$. And $H_3=1+\frac12+\frac13=\frac{11}6$.

Counting by length gives the same check. The identity contributes three $1$-cycles and the three transpositions contribute one each, six in all, so the mean number of $1$-cycles is $1$. Only the transpositions contribute a $2$-cycle, three of them, and the mean is $\frac12$. The two $3$-cycles give mean $\frac13$.
\end{examplex}

Two cycles of length greater than $n/2$ cannot sit inside $\{1,\ldots,n\}$. Exercise~\ref{ex:prisoners} uses that disjointness.

\section{The coupon collector}

Each box contains one of $n$ coupons, uniformly and independently of the other boxes. Let $T$ be the number of boxes needed to see every coupon. Volume~1, Chapter~10, asks for $\mathbb E[T]=nH_n$ and notes that the law is a distribution on finite words.

\begin{theorem}[coupon collector]\label{thm:coupon}
For $n\ge1$, $\mathbb E[T]=nH_n$. Moreover $H_n/\ln n\to1$, so $\mathbb E[T]\sim n\ln n$.
\end{theorem}
\begin{proof}
Let $T_0=0$ and, for $0\le i\le n-1$, let $T_{i+1}$ be the number of further boxes needed to obtain a new coupon after $i$ distinct coupons are already in hand. Then $T=T_1+\cdots+T_n$.

Given the past, $T_i$ is a geometric waiting time with success probability $p_i=(n-i+1)/n$. That is the first success on independent trials, in the sense of Volume~1. The success probability depends only on how many coupons are already in hand, not on which ones. Every past with $i-1$ coupons produces the same conditional law, so the unconditional mean is $1/p_i$.

Stopped at time $T$, the experiment is a countable set of finite words. There are $n$ stages, and each stage is a word in an alphabet of $n$ letters. A countable union of finite sets is countable. Volume~1's tail sum gives $\mathbb E[T_i]=1/p_i=n/(n-i+1)$. Each of these $n$ expectations is finite, so linearity gives
\[
 \mathbb E[T]=\sum_{i=1}^n\frac{n}{n-i+1}=n\sum_{j=1}^n\frac1j=nH_n.
\]

Since $1/x$ is decreasing, the bar comparison of Volume~1, Chapter~3, gives
\[
 \ln(n+1)=\int_1^{n+1}\frac{\mathrm dx}x\le\sum_{k=1}^n\frac1k=H_n\le1+\int_1^n\frac{\mathrm dx}x=1+\ln n
\]
for every $n\ge1$. The left inequality at $n=1$ is $\ln2\le1$, and $e>1+1+1/2=5/2>2$ from the series for $e$ in Volume~1. For $n\ge2$, divide by $\ln n$. Then $\ln(n+1)/\ln n\to1$ and $(1+\ln n)/\ln n\to1$, so $H_n/\ln n\to1$. Therefore $\mathbb E[T]\sim n\ln n$.
\end{proof}

\begin{think}
With $n$ coupons, the last new coupon takes about $n$ boxes on its own. Which earlier stage, if any, costs more?
\end{think}

None. The waiting time $T_i$ has mean $n/(n-i+1)$, which is largest at the last stage, $i=n$, where the mean is $n$. The first stage has mean $1$. The expensive part of the collection is the end, not the beginning. That is why $\mathbb E[T]$ is of order $n\ln n$ rather than of order $n$.

The same stages have a variance. A geometric waiting time needs one identity first.

\begin{lemma}[second moment of a geometric wait]\label{lem:geovar}
Let $0<p\le1$ and let $G$ be the trial number of the first success on independent trials of probability $p$, so $P(G=k)=p(1-p)^{k-1}$ for integers $k\ge1$. Then $\mathbb E[G]=1/p$ and $\operatorname{Var}G=(1-p)/p^2$.
\end{lemma}
\begin{proof}
Volume~1 gives the mean by the tail sum $\mathbb E[G]=\sum_{j\ge0}P(G>j)=\sum_{j\ge0}(1-p)^j=1/p$. For the second factorial moment, rearrange the nonnegative double sum:
\[
 \sum_{j\ge1}j\,P(G>j)=\sum_{j\ge1}j\sum_{k>j}P(G=k)=\sum_{k\ge2}P(G=k)\sum_{j=1}^{k-1}j=\sum_{k\ge2}P(G=k)\frac{k(k-1)}{2}.
\]
The inner sum is $k(k-1)/2$, so $2\sum_{j\ge1}j\,P(G>j)=\mathbb E[G(G-1)]$. Let $r=1-p\in[0,1)$ and $S_N=\sum_{j=1}^N jr^j$. Then $S_N-rS_N=\sum_{j=1}^N r^j-Nr^{N+1}=r(1-r^N)/(1-r)-Nr^{N+1}$. To see $Nr^N\to0$, pick an integer $m>r/(1-r)$, so $q=((m+1)/m)r<1$. For $n\ge m$ the sequence $a_n=nr^n$ satisfies $a_{n+1}\le q a_n$, hence $a_n\le a_m q^{n-m}$. Volume~1 gives $q^{n}\to0$, so $a_n\to0$. Therefore $S_N\to r/(1-r)^2=(1-p)/p^2$. Thus $\mathbb E[G(G-1)]=2(1-p)/p^2$. Then
\[
 \operatorname{Var}G=\mathbb E[G(G-1)]+\mathbb E[G]-(\mathbb E[G])^2=\frac{2(1-p)}{p^2}+\frac1p-\frac1{p^2}=\frac{1-p}{p^2}.
\]
If $p=1$, both sides are $0$.
\end{proof}

\begin{proposition}[variance of the coupon time]\label{prop:couponvar}
The stage lengths $T_1,\ldots,T_n$ are independent, $T_i$ is geometric with success probability $p_i=(n-i+1)/n$, and
\[
 \operatorname{Var}T=n^2\sum_{k=1}^n\frac1{k^2}-nH_n.
\]
In particular $0\le\operatorname{Var}T<2n^2$.
\end{proposition}
\begin{proof}
Once $i-1$ distinct coupons are in hand, each new box is a fresh coupon with probability $p_i$, whatever those $i-1$ coupons were and however long the earlier stages took. The conditional law of $T_i$ given $(T_1,\ldots,T_{i-1})$ is therefore geometric with success probability $p_i$, and it does not depend on the earlier lengths. The joint law factors, so the stages are independent.

Lemma~\ref{lem:geovar} gives $\operatorname{Var}T_i=(1-p_i)/p_i^2$. Independence adds the variances. Set $k=n-i+1$. As $i$ runs from $1$ to $n$, the index $k$ runs from $n$ down to $1$, and $1-p_i=(n-k)/n$, so
\[
 \frac{1-p_i}{p_i^2}=\frac{n-k}{n}\cdot\frac{n^2}{k^2}=\frac{n^2}{k^2}-\frac nk.
\]
Summing over $k$ produces the displayed formula.

The sum $\sum_{k=1}^n k^{-2}$ is at most $1+\int_1^n x^{-2}\,\mathrm dx=2-1/n<2$, by the same bar comparison Volume~1 uses for $H_n$, applied to the decreasing function $x^{-2}$ on $[1,\infty)$. The subtracted term $nH_n$ is positive, so the variance is less than $2n^2$. For $n=2$ the formula is $4(1+1/4)-2(1+1/2)=2$. Directly, $T_1=1$ and $T_2$ is geometric with $p=1/2$, whose variance is $2$.
\end{proof}

Since $0\le\operatorname{Var}T<2n^2$ and $\mathbb E[T]=nH_n$,
\[
 \frac{\sqrt{\operatorname{Var}T}}{\mathbb E[T]}<\frac{\sqrt2}{H_n}.
\]
The right side tends to $0$ because $H_n\to\infty$. The standard deviation is negligible beside the mean.

Indicators do the same job for boxes. Throw $m\ge1$ balls independently and uniformly into $n$ boxes. The base-$n$ digits of the uniform number in Chapter~1 supply the throws.

\begin{proposition}[empty boxes]\label{prop:empty}
Let $Z$ be the number of empty boxes after $m\ge1$ throws into $n$ boxes. For $n\ge1$,
\[
 \mathbb E[Z]=n\Bigl(1-\frac1n\Bigr)^m.
\]
For $n\ge2$,
\[
 \mathbb E[Z(Z-1)]=n(n-1)\Bigl(1-\frac2n\Bigr)^m.
\]
If $m=n$, then $\mathbb E[Z]/n=(1-1/n)^n\to e^{-1}$ as $n\to\infty$.
\end{proposition}
\begin{proof}
Let $I_j$ indicate that box $j$ is empty. Each ball misses box $j$ with probability $1-1/n$, and the $m$ balls are independent, so $P(I_j=1)=(1-1/n)^m$. Linearity gives the first formula.

For $n\ge2$ and $j\ne\ell$, both boxes are missed with probability $1-2/n$. There are $n(n-1)$ ordered pairs $(j,\ell)$, and $Z(Z-1)=\sum_{j\ne\ell}I_jI_\ell$. Linearity gives the second formula. The case $m=n$ is the $k=0$ clause of Theorem~\ref{thm:poissonlimit} with $\lambda=1$.
\end{proof}

Two boxes and two balls give $\mathbb E[Z]=2\cdot(1/2)^2=1/2$. The four equally likely outcomes are $(1,1)$, $(1,2)$, $(2,1)$ and $(2,2)$. Two of them leave one box empty, and two leave none. The average is $1/2$. The second formula gives $\mathbb E[Z(Z-1)]=2\cdot1\cdot0^2=0$: with two balls in two boxes, $Z$ is $0$ or $1$, so $Z(Z-1)=0$.

The digits of Chapter~1, the cycle count, and the coupon time are all random variables. A sum of independent copies has a generating function that multiplies. Chapter~3 computes with that product, and reads the extinction of a family name off a fixed point of the product.

\section{Exercises}

\begin{exercise}[$\star$]\label{ex:records}
In a uniform random permutation of $\{1,\ldots,n\}$, call index $k$ a \emph{record} when $\pi(k)>\pi(j)$ for every $j<k$. Show that the expected number of records is $H_n$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:fixedvar}
Let $F_n$ be the number of fixed points of a uniform random permutation of $n\ge2$ elements. Show that $\operatorname{Var}F_n=1$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:quicksort}
Quicksort on a list of $n$ distinct keys chooses a uniform pivot, compares it with the other $n-1$ keys, and recurses on the two sides. Let $Q_n$ be the number of comparisons, with $Q_0=Q_1=0$. Show that $\mathbb E[Q_n]=2(n+1)H_n-4n$ for $n\ge1$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:factorial}
Let $X$ be Poisson with parameter $\lambda>0$. Show that $\mathbb E[X(X-1)\cdots(X-m+1)]=\lambda^m$ for each integer $m\ge1$, and recover $\operatorname{Var}X=\lambda$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:reservoir}
A stream of $n$ items $x_1,\ldots,x_n$ is read once. Keep the first item. When item $i$ arrives, for $i\ge2$, replace the stored item by $x_i$ with probability $1/i$, independently of the past. Show that after $n$ items each \emph{input position} has probability $1/n$ of supplying the stored item, even if some items have equal values.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:prisoners}
\begin{enumerate}[label=(\alph*)]
\item Show that the number of permutations of $\{1,\ldots,m\}$ that contain a cycle of a given length $k>m/2$ is $m!/k$. Deduce that a uniform permutation of $2n$ elements has some cycle of length greater than $n$ with probability $H_{2n}-H_n$, and that this probability tends to $\ln2$.
\item One hundred prisoners have numbers placed uniformly at random in one hundred boxes. Each prisoner may open fifty boxes. They are all freed if every prisoner finds their own number. They agree that prisoner $i$ opens box $i$, then the box labelled with the number found there, and so on. Show that they succeed with probability $1-(H_{100}-H_{50})$, which equals $0.3118\ldots$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{ex:birthdaywait}
Balls are thrown independently and uniformly into $N$ boxes until the first time two share a box. Let $W_N$ be the number of balls thrown, including the ball that causes the collision. Show that $\mathbb E[W_N]\sim\sqrt{\pi N/2}$, using $\int_0^\infty e^{-x^2/2}\,\mathrm dx=\sqrt{\pi/2}$ from Volume~1.
\end{exercise}

\begin{exercise}[$\star$]\label{ex:twopoisson}
Let $X$ and $Y$ be independent, Poisson$(\lambda)$ and Poisson$(\mu)$. Show, from the limit theorem or from the series, that $X+Y$ is Poisson$(\lambda+\mu)$.
\end{exercise}

\section{Hints}

\begin{hint}{ex:records}
Among the first $k$ values, the largest is equally likely to sit in any of the $k$ positions.
\end{hint}
\begin{hint}{ex:fixedvar}
Compute $\mathbb E[I_i I_j]$ for the indicators of fixed points, and separate $i=j$ from $i\ne j$.
\end{hint}
\begin{hint}{ex:quicksort}
Condition on the rank of the pivot. The resulting recurrence is solved by passing to $\mathbb E[Q_n]/(n+1)$.
\end{hint}
\begin{hint}{ex:factorial}
Shift the index in the series for $e^\lambda$.
\end{hint}
\begin{hint}{ex:reservoir}
Induct on $n$. At step $n$ the new item is kept with probability $1/n$, and otherwise the previous uniform choice stands.
\end{hint}
\begin{hint}{ex:prisoners}
For $k>m/2$ there is at most one cycle of length $k$, so the events are disjoint. The prisoners all succeed exactly when no cycle is longer than $50$.
\end{hint}
\begin{hint}{ex:birthdaywait}
Write $\mathbb E[W_N]=\sum_{m\ge0}P(W_N>m)$ and compare the product for $P(W_N>m)$ with a Gaussian integral after the change of scale $m=\sqrt N\,u$.
\end{hint}
\begin{hint}{ex:twopoisson}
Write the probability of $X+Y=k$ as the sum over $X=j$, for $0\le j\le k$. Independence lets you multiply the two Poisson masses; the binomial theorem finishes the sum.
\end{hint}

\section{Solutions}

\begin{solution}{ex:records}
\idea A record at $k$ is a statement about the relative order of $k$ values, and that order is uniform.
\writeup Let $I_k$ be the indicator that $k$ is a record. Look at $\pi(1),\ldots,\pi(k)$. These are $k$ distinct values, and their order is a uniform random permutation of those values. The largest one is equally likely to sit in any of the $k$ positions, so it sits at position $k$ with probability $1/k$. Thus $\mathbb E[I_k]=1/k$, and the expected number of records is $H_n$.
\end{solution}

\begin{solution}{ex:fixedvar}
\idea The diagonal terms contribute $1-1/n$. The off-diagonal covariances contribute $1/n$. They cancel in the way that leaves $1$.
\writeup Let $I_i$ indicate that $\pi(i)=i$. Then $P(I_i=1)=1/n$ and $\operatorname{Var}I_i=(1/n)(1-1/n)$. For $i\ne j$, $P(I_i=I_j=1)=1/(n(n-1))$, so
\[
 \operatorname{Cov}(I_i,I_j)=\frac1{n(n-1)}-\frac1{n^2}=\frac1{n^2(n-1)}.
\]
There are $n(n-1)$ ordered pairs $(i,j)$ with $i\ne j$. Their covariances sum to $1/n$. The variances sum to $1-1/n$. Therefore $\operatorname{Var}F_n=1$.
\end{solution}

\begin{solution}{ex:quicksort}
\idea The pivot's rank is uniform on $\{1,\ldots,n\}$, and the two sides are themselves uniform random lists.
\writeup The first pass costs $n-1$ comparisons. If the pivot has rank $K$, uniform on $\{1,\ldots,n\}$, the two sublists have lengths $K-1$ and $n-K$. With fresh uniform pivots on each sublist, their expected comparison counts are $\mathbb E[Q_{K-1}]$ and $\mathbb E[Q_{n-K}]$; their mutual independence is not needed for the expectation recurrence. So
\[
 \mathbb E[Q_n]=n-1+\frac1n\sum_{k=1}^n\bigl(\mathbb E[Q_{k-1}]+\mathbb E[Q_{n-k}]\bigr)
 =n-1+\frac2n\sum_{j=0}^{n-1}\mathbb E[Q_j].
\]
Set $a_n=\mathbb E[Q_n]$, with $a_0=a_1=0$. Then $na_n=n(n-1)+2\sum_{j=0}^{n-1}a_j$. The same identity at $n-1$, for $n\ge2$, is $(n-1)a_{n-1}=(n-1)(n-2)+2\sum_{j=0}^{n-2}a_j$. Subtract:
\[
 na_n-(n-1)a_{n-1}=2(n-1)+2a_{n-1},
\]
hence $na_n=(n+1)a_{n-1}+2(n-1)$, and
\[
 \frac{a_n}{n+1}=\frac{a_{n-1}}{n}+\frac{2(n-1)}{n(n+1)}.
\]
Decompose $(n-1)/(n(n+1))=A/n+B/(n+1)$. Then $A(n+1)+Bn=n-1$. The value $n=0$ gives $A=-1$, and $n=-1$ gives $B=2$. So $(n-1)/(n(n+1))=-1/n+2/(n+1)$, and
\[
 \frac{2(n-1)}{n(n+1)}=\frac{4}{n+1}-\frac{2}{n}.
\]
Therefore
\[
 \frac{a_n}{n+1}-\frac{a_{n-1}}{n}=\frac{4}{n+1}-\frac{2}{n}.
\]
Sum from $n=2$ to $N$. The left side telescopes to $a_N/(N+1)-a_1/2=a_N/(N+1)$. The right side is
\[
 4\sum_{n=2}^N\frac1{n+1}-2\sum_{n=2}^N\frac1n
 =4\sum_{m=3}^{N+1}\frac1m-2\sum_{n=2}^N\frac1n
 =4\Bigl(H_{N+1}-1-\frac12\Bigr)-2\bigl(H_N-1\bigr)
 =4H_{N+1}-6-2H_N+2
 =4H_{N+1}-2H_N-4.
\]
Since $H_{N+1}=H_N+1/(N+1)$,
\[
 \frac{a_N}{N+1}=4H_N+\frac4{N+1}-2H_N-4=2H_N-4+\frac4{N+1},
\]
and $a_N=2(N+1)H_N-4(N+1)+4=2(N+1)H_N-4N$. For $N=1$ the formula gives $2\cdot2\cdot1-4=0$, which matches $a_1$.
\end{solution}

\begin{solution}{ex:factorial}
\idea Each falling factorial peels one factor of $\lambda$ off the series.
\writeup 
\[
 \mathbb E[X(X-1)\cdots(X-m+1)]
 =\sum_{k\ge m}k(k-1)\cdots(k-m+1)\,e^{-\lambda}\frac{\lambda^k}{k!}
 =e^{-\lambda}\lambda^m\sum_{j\ge0}\frac{\lambda^j}{j!}=\lambda^m,
\]
where $j=k-m$. For $m=1$, $\mathbb EX=\lambda$. For $m=2$, $\mathbb E[X(X-1)]=\lambda^2$, so $\mathbb E[X^2]=\lambda^2+\lambda$ and $\operatorname{Var}X=\lambda$.
\end{solution}

\begin{solution}{ex:reservoir}
\idea The new item has one chance in $i$ to be the survivor at step $i$, and the induction hypothesis shares the remaining probability equally.
\writeup After one item, the store is $x_1$. Suppose that after $n-1$ items the stored item is uniform on the first $n-1$ items. At step $n$ the algorithm keeps $x_n$ with probability $1/n$. For each $j<n$, the probability that $x_j$ is stored at the end is the probability that it was stored after $n-1$ items, times the probability it is not replaced:
\[
 \frac1{n-1}\cdot\Bigl(1-\frac1n\Bigr)=\frac1{n-1}\cdot\frac{n-1}n=\frac1n.
\]
So each of the $n$ input positions has probability $1/n$ of supplying the stored item; equal item values do not alter the position probabilities.
\end{solution}

\begin{solution}{ex:prisoners}
\idea A long cycle, if it exists, is unique. The prisoners follow that cycle and run out of openings exactly when it is longer than fifty.
\writeup (a) If $k>m/2$, a permutation has at most one cycle of length $k$. The number of permutations that have one is $\binom{m}{k}(k-1)!(m-k)!=m!/k$. The probability is $1/k$. These events, for $k=n+1,\ldots,2n$ and $m=2n$, are pairwise disjoint. The probability of their union is
\[
 \sum_{k=n+1}^{2n}\frac1k=H_{2n}-H_n.
\]
The integral comparison gives $\ln\bigl((2n+1)/(n+1)\bigr)\le H_{2n}-H_n\le\ln2$, because
\[
 \int_{n+1}^{2n+1}\frac{\mathrm dx}x\le\sum_{k=n+1}^{2n}\frac1k\le\int_n^{2n}\frac{\mathrm dx}x.
\]
The lower end is $\ln\bigl((2n+1)/(n+1)\bigr)\to\ln2$, and the upper end is $\ln2$. By the squeeze theorem of Volume~1, Chapter~3, $H_{2n}-H_n\to\ln2$.

(b) Prisoner $i$ follows the unique cycle of the permutation that contains $i$. That prisoner finds their number among the fifty opened boxes if and only if the cycle has length at most $50$. Every prisoner succeeds if and only if every cycle has length at most $50$, which is the event that there is no cycle of length greater than $50$. By (a) with $m=100$ and $n=50$, that failure probability is $H_{100}-H_{50}$. The sum $\sum_{k=51}^{100}1/k$ equals $0.68817\ldots$, so the success probability is $0.31182\ldots$.
\end{solution}

\begin{solution}{ex:birthdaywait}
\idea The collision arrives near scale $\sqrt N$. On that scale the empty-product probability is a Gaussian integrand, and the expectation is an integral of the tail.
\writeup For $m\ge0$, write $p_m=P(W_N>m)$. Then $p_m=1$ for $m\le1$, $p_m=0$ for $m>N$, and
\[
 p_m=\prod_{j=1}^{m-1}\Bigl(1-\frac jN\Bigr),\qquad 2\le m\le N.
\]
Volume~1's tail sum gives $\mathbb E[W_N]=\sum_{m\ge0}p_m$.

The tangent bound $1-x\le e^{-x}$ from Volume~1 yields, for $2\le m\le N$,
\[
 p_m\le\exp\Bigl(-\sum_{j=1}^{m-1}\frac jN\Bigr)
 =\exp\Bigl(-\frac{(m-1)m}{2N}\Bigr).
\]
For $m\ge2$ one has $(m-1)m-(m-2)^2=3m-4\ge2$, so
\[
 p_m\le\exp\Bigl(-\frac{(m-2)^2}{2N}\Bigr).
\]
The values $p_0$ and $p_1$ are $1$. The exponential on the right is strictly smaller than $1$ at $m=1$, so that comparison is stated only for $m\ge2$.

For the matching lower estimate, Example~\ref{ex:rare} gives $-\ln(1-x)\le x+x^2/(2(1-x)^2)$. On $[0,1/2]$ one has $1/(2(1-x)^2)\le2$, hence $-\ln(1-x)\le x+2x^2$. Volume~1, Chapter~8, also records $-\ln(1-x)\ge x$ for $0\le x<1$. Thus, whenever $m-1\le N/2$,
\[
 \frac{(m-1)m}{2N}
 \le-\ln p_m
 \le\frac{(m-1)m}{2N}+\frac{2m^3}{3N^2},
\]
where $\sum_{j=1}^{m-1}j^2\le\int_0^m x^2\,\mathrm dx=m^3/3$.

Fix $A>0$. For $N\ge\max(4,4A^2)$ and $2\le m\le A\sqrt N$, one has $(m-1)/N\le A/\sqrt N\le1/2$, and the cubic error is at most $2A^3/(3\sqrt N)$. So
\[
 \exp\Bigl(-\frac{m^2}{2N}-\frac{C_A}{\sqrt N}\Bigr)
 \le p_m
 \le\exp\Bigl(-\frac{(m-2)^2}{2N}\Bigr),
\]
with $C_A=2A^3/3$. Shift the upper bound by two steps: with $k=m-2$,
\[
 \frac1{\sqrt N}\sum_{m=2}^{\lfloor A\sqrt N\rfloor}\exp\Bigl(-\frac{(m-2)^2}{2N}\Bigr)
 =\frac1{\sqrt N}\sum_{k=0}^{\lfloor A\sqrt N\rfloor-2}\exp\Bigl(-\frac{k^2}{2N}\Bigr).
\]
Once $\lfloor A\sqrt N\rfloor\ge2$, this differs from the sum running up to $\lfloor A\sqrt N\rfloor$ by at most $2/\sqrt N$. The function $e^{-u^2/2}$ is continuous on the compact interval $[0,A]$, so both sums, and the matching lower sum of $\exp(-m^2/(2N))$, are Riemann sums for $\int_0^A e^{-u^2/2}\,\mathrm du$. The factor $e^{-C_A/\sqrt N}$ tends to $1$. The two terms $p_0$ and $p_1$ contribute $2/\sqrt N$. Therefore
\[
 \frac1{\sqrt N}\sum_{m\le A\sqrt N}p_m\to\int_0^A e^{-u^2/2}\,\mathrm du.
\]

For the tail, $p_m=0$ once $m>N$. For $2\le m\le N$, the bound above gives $p_m\le\exp(-(m-2)^2/(2N))$. The map $x\mapsto\exp(-x^2/(2N))$ decreases on $[0,\infty)$. The bar comparison in the integral test of Volume~1, Chapter~3, says that a decreasing nonnegative function satisfies $f(k)\le\int_{k-1}^k f$ for each integer $k\ge1$. Hence, for an integer $K\ge1$,
\[
 \sum_{k\ge K}\exp\Bigl(-\frac{k^2}{2N}\Bigr)
 \le\int_{K-1}^\infty\exp\Bigl(-\frac{x^2}{2N}\Bigr)\,\mathrm dx.
\]
The substitution $u=x/\sqrt N$ turns the right side into $\sqrt N\int_{(K-1)/\sqrt N}^\infty e^{-u^2/2}\,\mathrm du$. Choose $K$ to be the least integer strictly larger than $A\sqrt N-2$, and take $A>3$ and $N$ large enough that this $K$ is at least $1$. Then $(K-1)/\sqrt N\ge A-3$, so the tail $\sum_{m>A\sqrt N}p_m$, divided by $\sqrt N$, is at most $\int_{A-3}^\infty e^{-u^2/2}\,\mathrm du$. That integral tends to $0$ as $A\to\infty$. First let $N\to\infty$ with $A$ fixed, then let $A\to\infty$. The quotient $\mathbb E[W_N]/\sqrt N$ converges to $\int_0^\infty e^{-u^2/2}\,\mathrm du=\sqrt{\pi/2}$. Hence $\mathbb E[W_N]\sim\sqrt{\pi N/2}$.
\end{solution}

\begin{solution}{ex:twopoisson}
\idea The series for the sum is the binomial theorem applied to the product of the two exponential series.
\writeup From the series,
\[
 P(X+Y=n)=\sum_{k=0}^n e^{-\lambda}\frac{\lambda^k}{k!}\,e^{-\mu}\frac{\mu^{n-k}}{(n-k)!}
 =e^{-(\lambda+\mu)}\frac{(\lambda+\mu)^n}{n!},
\]
because $\sum_{k=0}^n\binom{n}{k}\lambda^k\mu^{n-k}=(\lambda+\mu)^n$.
\end{solution}
