% Root-lifting ladder; the solid spine is a coherent sequence.
\begin{tikzpicture}[font=\small,x=1cm,y=1cm]
  \foreach \j/\r/\m in {0/2/5,1/7/25,2/57/125,3/182/625} {
    \node[draw,rounded corners=2pt,minimum width=1.2cm,fill=gray!8] (a\j) at (0,-1.45*\j) {$\r$};
    \node[anchor=west] at (3,-1.45*\j) {modulo $\m$};
  }
  \foreach \j/\next/\digit in {0/1/1,1/2/2,2/3/1} {
    \draw[->,very thick,MidnightBlue] (a\j)--node[left] {digit $\digit$} (a\next);
    \foreach \k in {1,2,3,4} {
      \draw[dashed,gray] (a\j.south)--(.4*\k+.25,-1.45*\j-.85);
      \node[gray] at (.4*\k+.25,-1.45*\j-.95) {$\times$};
    }
  }
\end{tikzpicture}
