% Convergents of sqrt(2). Label guides preserve crowded true positions.
\begin{tikzpicture}[x=8.2cm,font=\small]
  \def\L{1.28}
  \def\R{1.62}
  \pgfmathsetmacro{\X}{(\R-\L)}
  \draw[ocInk] (0,0)--(1,0);
  \foreach \v/\lab in {1.3/{1.3},1.4/{1.4},1.5/{1.5},1.6/{1.6}} {
    \pgfmathsetmacro{\t}{(\v-\L)/\X}
    \draw (\t,0.05)--(\t,-0.05) node[below] {\lab};
  }
  \pgfmathsetmacro{\s}{(1.41421356237-\L)/\X}
  \draw[ocOlive,thick] (\s,-0.25)--(\s,1.52);
  \node[ocOlive,above] at (\s,1.52) {$\sqrt2$};
  \pgfmathsetmacro{\a}{(7/5-\L)/\X}
  \pgfmathsetmacro{\b}{(41/29-\L)/\X}
  \pgfmathsetmacro{\c}{(3/2-\L)/\X}
  \pgfmathsetmacro{\d}{(17/12-\L)/\X}
  \pgfmathsetmacro{\e}{(99/70-\L)/\X}
  % Below-limit labels sit well below the decimal tick labels.
  \draw[ocBlue!65] (\a,0)--(0.22,-0.72);
  \draw[ocBlue!65] (\b,0)--(0.48,-0.72);
  \node[ocBlue,below] at (0.22,-0.72) {$\dfrac75$};
  \node[ocBlue,below] at (0.48,-0.72) {$\dfrac{41}{29}$};
  % Above-limit labels have separated horizontal anchors and ample height.
  \draw[ocCoralDeep!65] (\d,0)--(0.22,0.83);
  \draw[ocCoralDeep!65] (\e,0)--(0.52,0.83);
  \draw[ocCoralDeep!65] (\c,0)--(0.72,0.83);
  \node[ocCoralDeep,above] at (0.22,0.83) {$\dfrac{17}{12}$};
  \node[ocCoralDeep,above] at (0.52,0.83) {$\dfrac{99}{70}$};
  \node[ocCoralDeep,above] at (0.72,0.83) {$\dfrac32$};
  \fill[ocBlue] (\a,0) circle (1.6pt);
  \fill[ocBlue] (\b,0) circle (1.6pt);
  \draw[ocCoralDeep,thick] (\c,0) circle (1.7pt);
  \draw[ocCoralDeep,thick] (\d,0) circle (1.7pt);
  \draw[ocCoralDeep,thick] (\e,0) circle (1.7pt);
  \node[anchor=west,font=\small] at (0,-1.75)
    {\textcolor{ocBlue}{filled: below $\sqrt2$}\quad
     \textcolor{ocCoralDeep}{open: above $\sqrt2$}};
\end{tikzpicture}
