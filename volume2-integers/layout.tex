% pdf-only
% Leave the ISBN line absent until a real paperback ISBN has been assigned.
% Set \bookisbnprint in main.tex to restore it automatically, and the build adds a barcode.
\usepackage{etoolbox}
\makeatletter
\patchcmd{\maketitle}
  {ISBN (paperback): \bookisbnprint}
  {\ifx\bookisbnprint\oc@empty\else ISBN (paperback): \bookisbnprint\fi}
  {}{\PackageError{volume1}{Could not patch the ISBN line}{Check common/frontmatter.tex.}}

% The series catalogue can occupy two pages. Keep the title on a recto,
% and suppress the running head on the catalogue's continuation page.
\ifprintformat
\patchcmd{\maketitle}
  {\clearpage\thispagestyle{empty}\vspace*{0.12\textheight}}
  {\thispagestyle{empty}\cleardoublepage\thispagestyle{empty}\vspace*{0.12\textheight}}
  {}{\PackageError{volume1}{Could not keep the title page on a recto}{Check common/frontmatter.tex.}}
\fi
\makeatother

% Keep diagrams and their explanatory captions together on the final book pages.
\renewcommand{\fig}[1]{\adjustbox{max width=\linewidth,max height=0.52\textheight}{\input{figures/#1}}}
