% Swapping the two edges of a parallelogram reverses the sign of the determinant.
\begin{tikzpicture}[>=Stealth,scale=1]
  \fill[MidnightBlue!12] (0,0) -- (2.4,0.35) -- (3.05,1.9) -- (0.65,1.55) -- cycle;
  \draw[MidnightBlue] (0,0) -- (2.4,0.35) -- (3.05,1.9) -- (0.65,1.55) -- cycle;
  \draw[BrickRed,very thick,->] (0,0) -- (2.4,0.35) node[below right,font=\scriptsize,inner sep=1pt] {$v$};
  \draw[ForestGreen,very thick,->] (0,0) -- (0.65,1.55) node[left,font=\scriptsize,inner sep=1pt] {$w$};
  \draw[BrickRed,semithick,->] (0.78,0.14) .. controls (0.62,0.38) and (0.40,0.52) .. (0.22,0.62);
  \node[font=\scriptsize] at (1.5,-0.95) {$\det(v\ w)>0$};
  \begin{scope}[xshift=6.2cm]
    \fill[MidnightBlue!12] (0,0) -- (0.65,1.55) -- (3.05,1.9) -- (2.4,0.35) -- cycle;
    \draw[MidnightBlue] (0,0) -- (0.65,1.55) -- (3.05,1.9) -- (2.4,0.35) -- cycle;
    \draw[BrickRed,very thick,->] (0,0) -- (0.65,1.55) node[left,font=\scriptsize,inner sep=1pt] {$w$};
    \draw[ForestGreen,very thick,->] (0,0) -- (2.4,0.35) node[below right,font=\scriptsize,inner sep=1pt] {$v$};
    \draw[BrickRed,semithick,->] (0.22,0.62) .. controls (0.40,0.52) and (0.62,0.38) .. (0.78,0.14);
    \node[font=\scriptsize] at (1.5,-0.95) {$\det(w\ v)<0$};
  \end{scope}
\end{tikzpicture}
