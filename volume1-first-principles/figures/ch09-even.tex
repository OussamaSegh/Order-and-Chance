% A fair die, conditioned on an even face. Of 2, 4, 6, the faces above 3 are 4 and 6.
\begin{tikzpicture}[x=1.2cm,y=1.2cm]
  \foreach \k in {1,...,6} {
    \ifnum\k=2
      \fill[MidnightBlue!18] (\k,0) rectangle ++(0.9,0.9);
    \fi
    \ifnum\k=4
      \fill[BrickRed!32] (\k,0) rectangle ++(0.9,0.9);
    \fi
    \ifnum\k=6
      \fill[BrickRed!32] (\k,0) rectangle ++(0.9,0.9);
    \fi
    \draw[MidnightBlue] (\k,0) rectangle ++(0.9,0.9);
    \node[font=\small] at (\k+0.45,0.45) {$\k$};
  }
  \node[font=\scriptsize,anchor=north] at (2.45,-0.14) {kept};
  \node[font=\scriptsize,anchor=north] at (4.45,-0.14) {kept};
  \node[font=\scriptsize,anchor=north,BrickRed!75!black] at (4.45,-0.46) {$>3$};
  \node[font=\scriptsize,anchor=north] at (6.45,-0.14) {kept};
  \node[font=\scriptsize,anchor=north,BrickRed!75!black] at (6.45,-0.46) {$>3$};
\end{tikzpicture}
