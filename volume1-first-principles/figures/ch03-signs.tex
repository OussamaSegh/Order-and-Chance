% Descartes counts sign changes. One change forces one positive root; two changes allow 0 or 2 negative roots.
\begin{tikzpicture}[font=\scriptsize,>=Stealth]
  \node[anchor=east] at (-0.15,1.35) {$X^3-X-1$};
  \foreach \x/\s/\name in {0.7/+/{X^3},2.3/-/{X},3.9/-/{1}} {
    \node[circle,draw=BrickRed,thick,inner sep=1.5pt,font=\normalsize] at (\x,1.35) {$\s$};
    \node[anchor=north] at (\x,1.05) {$\name$};
  }
  \draw[MidnightBlue,thick,->] (1.15,1.35) -- (1.85,1.35);
  \node[anchor=west,align=left] at (4.7,1.35) {one change:\\exactly one positive root};

  \node[anchor=east] at (-0.15,0) {$P(-X)$};
  \foreach \x/\s/\name in {0.7/-/{X^3},2.3/+/{X},3.9/-/{1}} {
    \node[circle,draw=BrickRed,thick,inner sep=1.5pt,font=\normalsize] at (\x,0) {$\s$};
    \node[anchor=north] at (\x,-0.3) {$\name$};
  }
  \draw[MidnightBlue,thick,->] (1.15,0) -- (1.85,0);
  \draw[MidnightBlue,thick,->] (2.75,0) -- (3.45,0);
  \node[anchor=west,align=left] at (4.7,0) {two changes:\\$0$ or $2$ negative roots};
\end{tikzpicture}
