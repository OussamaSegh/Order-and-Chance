% Allowed transitions for a circular binary word with no two adjacent ones.
\begin{tikzpicture}[>=Stealth,node distance=2.4cm]
  \node[circle,draw=MidnightBlue,thick,minimum size=8mm] (z) {$0$};
  \node[circle,draw=BrickRed,thick,minimum size=8mm,right=of z] (o) {$1$};
  \draw[MidnightBlue,thick,->] (z) edge[loop above] node[font=\scriptsize] {$1$} (z);
  \draw[MidnightBlue,thick,->] (z) edge[bend left] node[font=\scriptsize,above] {$1$} (o);
  \draw[BrickRed,thick,->] (o) edge[bend left] node[font=\scriptsize,below] {$1$} (z);
  \node[font=\scriptsize,align=center] at (1.2,-1.35) {no arrow $1\to 1$:\\that step would place two ones together};
\end{tikzpicture}
