% Fifth roots of unity. Conjugate pairs sum to the real numbers c and d.
\begin{tikzpicture}[>=Stealth,scale=1.02]
  \def\R{1.55}
  \draw[gray!45,->] (-2.85,0) -- (2.15,0);
  \draw[gray!45,->] (0,-1.9) -- (0,1.95);
  \draw[black!35] (0,0) circle (\R);
  \foreach \k in {0,1,2,3,4} {
    \fill[MidnightBlue] ({\R*cos(72*\k)},{\R*sin(72*\k)}) circle (1.7pt);
    \draw[MidnightBlue!70,thin] (0,0) -- ({\R*cos(72*\k)},{\R*sin(72*\k)});
  }
  \draw[BrickRed,thick] ({\R*cos(72)},{\R*sin(72)}) -- ({\R*cos(288)},{\R*sin(288)});
  \draw[ForestGreen!75!black,thick] ({\R*cos(144)},{\R*sin(144)}) -- ({\R*cos(216)},{\R*sin(216)});
  \draw[BrickRed,very thick,->] (0,0) -- ({2*\R*cos(72)},0);
  \draw[ForestGreen!75!black,very thick,->] (0,0) -- ({2*\R*cos(144)},0);
  \node[font=\scriptsize,right] at ({\R+0.06},0.08) {$1$};
  \node[font=\scriptsize,above right] at ({\R*cos(72)},{\R*sin(72)}) {$\omega$};
  \node[font=\scriptsize,above left] at ({\R*cos(144)},{\R*sin(144)}) {$\omega^{2}$};
  \node[font=\scriptsize,below left] at ({\R*cos(216)},{\R*sin(216)}) {$\omega^{3}$};
  \node[font=\scriptsize,below right] at ({\R*cos(288)},{\R*sin(288)}) {$\omega^{4}$};
  \node[font=\scriptsize,BrickRed,below] at ({2*\R*cos(72)},-0.02) {$c$};
  \node[font=\scriptsize,ForestGreen!50!black,below] at ({2*\R*cos(144)},-0.02) {$d$};
\end{tikzpicture}
