% Pairs (i,j) with 1\le j\le i\le4. Left: outer index i. Right: outer index j. Same shaded cells.
\begin{tikzpicture}[scale=0.72]
  \newcommand{\occell}[1]{%
    \foreach \i in {1,...,4} {%
      \foreach \j in {1,...,4} {%
        \pgfmathtruncatemacro{\ocx}{\i-1}%
        \pgfmathtruncatemacro{\ocy}{4-\j}%
        \ifnum\j>\i
          \fill[gray!14] (\ocx,\ocy) rectangle (\i,{5-\j});
        \else
          \fill[#1] (\ocx,\ocy) rectangle (\i,{5-\j});
        \fi
        \node[font=\scriptsize] at ({\i-0.5},{4.5-\j}) {$(\i,\j)$};
      }%
    }%
    \draw[MidnightBlue,thin] (0,0) grid (4,4);
    \draw[MidnightBlue,thick] (0,0) rectangle (4,4);
    \foreach \i in {1,...,4} {%
      \node[font=\scriptsize,above] at ({\i-0.5},4.02) {$\i$};
    }%
    \foreach \j in {1,...,4} {%
      \node[font=\scriptsize,left] at (-0.04,{4.5-\j}) {$\j$};
    }%
    \node[font=\scriptsize,above] at (2,4.55) {$i$};
    \node[font=\scriptsize,left] at (-0.55,2) {$j$};
  }
  \begin{scope}
    \occell{Goldenrod!48}
    % Column i has height i, starting at the top (j=1).
    \draw[BrickRed,very thick,->] (0.5,3.82) -- (0.5,3.18);
    \draw[BrickRed,very thick,->] (1.5,3.82) -- (1.5,2.18);
    \draw[BrickRed,very thick,->] (2.5,3.82) -- (2.5,1.18);
    \draw[BrickRed,very thick,->] (3.5,3.82) -- (3.5,0.18);
    \node[font=\small,align=center] at (2,-0.85) {$\displaystyle\sum_{i=1}^{4}\sum_{j=1}^{i}$\\[2pt] down each column};
  \end{scope}
  \begin{scope}[xshift=7.4cm]
    \occell{ForestGreen!32}
    % Row j starts at column i=j and runs to i=4.
    \draw[BrickRed,very thick,->] (0.18,3.5) -- (3.82,3.5);
    \draw[BrickRed,very thick,->] (1.18,2.5) -- (3.82,2.5);
    \draw[BrickRed,very thick,->] (2.18,1.5) -- (3.82,1.5);
    \draw[BrickRed,very thick,->] (3.18,0.5) -- (3.82,0.5);
    \node[font=\small,align=center] at (2,-0.85) {$\displaystyle\sum_{j=1}^{4}\sum_{i=j}^{4}$\\[2pt] along each row};
  \end{scope}
\end{tikzpicture}
