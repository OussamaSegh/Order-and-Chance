% Removing compulsory gaps turns a restricted subset into an ordinary one.
\begin{tikzpicture}[x=0.85cm,y=0.9cm,>=Stealth]
  \foreach \i in {1,...,7} {
    \draw[gray!50] (\i,1.6) circle (0.28);
    \node[font=\scriptsize,gray] at (\i,2.15) {$\i$};
  }
  \foreach \i in {1,3,5} {
    \draw[BrickRed,thick,fill=BrickRed!15] (\i,1.6) circle (0.28);
  }
  \node[font=\small,anchor=west] at (7.6,1.6) {$a=(1,3,5)$};
  \draw[MidnightBlue,thick,->] (4,1.05) -- (4,0.35);
  \node[font=\scriptsize,MidnightBlue,anchor=west] at (4.15,0.7) {$b_i=a_i-(i-1)$};
  \foreach \i in {1,...,5} {
    \draw[gray!50] (\i,0) circle (0.28);
    \node[font=\scriptsize,gray] at (\i,-0.55) {$\i$};
  }
  \foreach \i in {1,2,3} {
    \draw[MidnightBlue,thick,fill=MidnightBlue!12] (\i,0) circle (0.28);
  }
  \node[font=\small,anchor=west] at (5.6,0) {$b=(1,2,3)$};
\end{tikzpicture}
