\begin{tikzpicture}[>=Stealth,scale=1.05]
  \draw[gray!45,->] (-0.4,0) -- (3.5,0);
  \draw[gray!45,->] (0,-0.35) -- (0,3.55);
  \coordinate (U) at ({2.15*cos(25)},{2.15*sin(25)});
  \coordinate (V) at ({2.15*cos(85)},{2.15*sin(85)});
  \coordinate (S) at ($(U)+(V)$);
  \draw[MidnightBlue,very thick,->] (0,0) -- (U);
  \draw[ForestGreen!75!black,very thick,->] (0,0) -- (V);
  \draw[MidnightBlue!45,dashed] (U) -- (S);
  \draw[ForestGreen!55!black,dashed] (V) -- (S);
  \draw[BrickRed,very thick,->] (0,0) -- (S);
  \draw[gray!70,thin] (0.55,0) arc (0:25:0.55);
  \draw[gray!70,thin] (0.78,0) arc (0:85:0.78);
  \node[font=\scriptsize,MidnightBlue,below right] at (U) {$e^{ia}$};
  \node[font=\scriptsize,ForestGreen!50!black,above left] at (V) {$e^{ib}$};
  \node[font=\scriptsize,BrickRed,above right] at (S) {$e^{ia}+e^{ib}$};
  \node[font=\scriptsize,gray] at (0.95,0.18) {$a$};
  \node[font=\scriptsize,gray] at (0.55,0.72) {$b$};
\end{tikzpicture}
