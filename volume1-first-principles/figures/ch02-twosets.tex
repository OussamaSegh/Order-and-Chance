% Two-set inclusion-exclusion for multiples of 2 and 3 up to 10.
% Only 2: 2,4,8,10 (four). Both: 6 (one). Only 3: 3,9 (two). Outside: 1,5,7 (three).
\begin{tikzpicture}[scale=1.05]
  \draw[rounded corners=3pt,gray!60] (-3.5,-2.15) rectangle (3.5,2.15);
  \node[font=\scriptsize,gray,anchor=north west] at (-3.4,2.05) {$\{1,\dots,10\}$};
  \draw[MidnightBlue,thick] (-0.85,0.15) circle (1.45);
  \draw[BrickRed,thick] (0.85,0.15) circle (1.45);
  \node[font=\small\bfseries,MidnightBlue] at (-2.15,1.35) {$2$};
  \node[font=\small\bfseries,BrickRed] at (2.15,1.35) {$3$};
  \node[font=\scriptsize,align=center,MidnightBlue] at (-1.45,0.15) {$2$\\[1pt]$4$\\[1pt]$8$\\[1pt]$10$};
  \node[font=\scriptsize,BrickRed] at (0,0.15) {$6$};
  \node[font=\scriptsize,align=center,BrickRed] at (1.45,0.15) {$3$\\[1pt]$9$};
  \node[font=\scriptsize,anchor=north] at (0,-1.55) {in neither circle: $1,5,7$};
\end{tikzpicture}
