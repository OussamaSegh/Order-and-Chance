% Sixth roots of unity, drawn by order. The dashed triangle is the subgroup of cube roots of unity.
\begin{tikzpicture}[>=Stealth,scale=1.05]
  \def\R{1.7}
  \draw[gray!40] (0,0) circle (\R);
  \draw[gray!45,->] (-2.35,0) -- (2.45,0);
  \draw[gray!45,->] (0,-2.25) -- (0,2.35);
  \draw[BrickRed,dashed] ({\R*cos(120)},{\R*sin(120)}) -- ({\R*cos(240)},{\R*sin(240)}) -- (\R,0) -- cycle;
  \fill[MidnightBlue] (\R,0) circle (2pt);
  \fill[BrickRed] ({\R*cos(60)},{\R*sin(60)}) circle (2.2pt);
  \fill[ForestGreen!75!black] ({\R*cos(120)},{\R*sin(120)}) circle (2pt);
  \fill[Plum] (-\R,0) circle (2pt);
  \fill[ForestGreen!75!black] ({\R*cos(240)},{\R*sin(240)}) circle (2pt);
  \fill[BrickRed] ({\R*cos(300)},{\R*sin(300)}) circle (2.2pt);
  \node[font=\scriptsize,right] at ({\R+0.08},0.1) {$1$};
  \node[font=\scriptsize,BrickRed,above right] at ({\R*cos(60)},{\R*sin(60)}) {$\omega$};
  \node[font=\scriptsize,ForestGreen!50!black,above left] at ({\R*cos(120)},{\R*sin(120)}) {$\omega^{2}$};
  \node[font=\scriptsize,Plum,left] at ({-\R-0.06},0.1) {$-1$};
  \node[font=\scriptsize,ForestGreen!50!black,below left] at ({\R*cos(240)},{\R*sin(240)}) {$\omega^{4}$};
  \node[font=\scriptsize,BrickRed,below right] at ({\R*cos(300)},{\R*sin(300)}) {$\omega^{5}$};
\end{tikzpicture}
