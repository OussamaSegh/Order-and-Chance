  \def\oc@this{#1}%
  \par\noindent\hangindent=2.2em\makebox[2.2em][l]{\small #1}%
  \ifx\oc@this\TOMENUM{\small\textbf{#2}}\else{\small #2}\fi

\renewcommand{\maketitle}{%
  \cleardoublepage
  % Half-title
  \thispagestyle{empty}%
  \vspace*{0.22\textheight}%
  \begin{center}
    {\sffamily\Large\seriesname}\par\bigskip
    \ifx\TOMETITLE\@empty\else{\LARGE Tome \TOMENUM\par\medskip\TOMETITLE\par}\fi
  \end{center}
  \clearpage
  % Series page
  \thispagestyle{empty}%
  {\raggedright\let\oc@this\relax
   \noindent{\sffamily\large\seriesname}\par
   \noindent{\small\itshape\seriessubtitle}\par\medskip
   \input{../common/series}\par}%
  \clearpage
  % Title page
  \thispagestyle{empty}%
  \vspace*{0.12\textheight}%
  \begin{center}
    {\sffamily\@title\par}\vskip 0.8em
    {\large\itshape\volumesubtitle{\TOMENUM}\par}\vskip 0.8em
    {\ifprintformat\color{black!55}\else\color{ocCoralDeep}\fi\rule{0.38\textwidth}{0.4pt}}\vskip 2em
    {\large\bookauthors\par}\vfill
    {\small\bookedition\par}
  \end{center}
  \clearpage
  % Copyright page: the text sits at the foot of the page, as is the custom.
  \thispagestyle{empty}%
  \vspace*{\fill}%
  {\footnotesize\raggedright\setlength{\parskip}{0.6em}\setlength{\parindent}{0pt}%
   \textit{\seriesname}\ifx\TOMETITLE\@empty\else. Tome~\TOMENUM: \textit{\TOMETITLE}\fi\par
   \bookcopyright\par
   \booklicence\par
   \bookedition. \bookpublisher. This printing typeset on \today.\par
   ISBN (paperback): \bookisbnprint\ifx\bookisbnebook\oc@empty\else\\ISBN (eBook): \bookisbnebook\fi\par
   Typeset with \LaTeX\ in Palatino.\par}%
  \clearpage
  \global\let\maketitle\relax}
\makeatother
