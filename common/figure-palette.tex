% Figure palette (design 5e; see docs/design/interior.md). Mathematical figures keep their
% sources and colour names; this file redefines the names they use so that every figure is
% drawn in the series palette: charcoal, rust and coral, with ochre, brown and neutral greys
% for further categories. Loaded after xcolor[dvipsnames] by the books and by tools/figure.tex
% (EPUB/reader figures). Prose and captions name the new colours (charcoal, rust, coral,
% ochre, brown, dark grey, light grey).

% rust (= ocCoralDeep): the second data colour, the one also used for text-sized marks
\definecolor{BrickRed}{HTML}{A34232}
\definecolor{Maroon}{HTML}{A34232}
\definecolor{red}{HTML}{A34232}
% coral (= ocCoral)
\definecolor{BurntOrange}{HTML}{FF8A78}

\definecolor{Orange}{HTML}{FF8A78}

\definecolor{yellow}{HTML}{FF8A78}

\definecolor{ForestGreen}{HTML}{B08D57}
