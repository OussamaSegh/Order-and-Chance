% Chapter openers (design 5e). Every numbered chapter of the main matter opens on a full
% charcoal verso: masthead, volume line, the coral numeral, "CHAPTER n", the ivory title,
% a coral rule, the chapter's epigraph and a motif. The text starts on the facing recto
% under a short heading (label, title, rule). A blank page is inserted when needed so the
% opener is always a verso. See docs/design/interior.md.
%
% Epigraph: the chapter's opening \begin{center}\itshape ... \end{center} (directly after
% \chapter{...} and its optional \label) moves onto the opener; the source is unchanged.
%
% Motif hook (the only per-chapter design input):
%   \chaptermotif{<TikZ code>}   before \chapter{...}; applies to that chapter only.
% The code is drawn in a 4.76 in x 3.46 in viewport, origin at its lower-left corner (the
% proportions of the cover illustration area), then scaled to the free space below the
% title and epigraph. Opacity is premixed over charcoal, so motifs stay opaque vectors.
% \chaptermotif{} gives an opener without a motif. Without the hook, every chapter shows
% its volume's cover illustration (cover-motif.tex, generated by tools/series_cover.py).

% Opacity becomes a colour mixed with the charcoal page (TikZ applies the colour option
% before the opacity option, so "." and \tikz@strokecolor already name the ink).
\newcommand{\oc@op@premix}{%
  \def\pgfsetstrokeopacity##1{%
    \ifx\tikz@strokecolor\pgfutil@empty\colorlet{oc@pms}{.!\oc@op@pct{##1}!ocBlack}%
    \else\colorlet{oc@pms}{\tikz@strokecolor!\oc@op@pct{##1}!ocBlack}\fi
    \pgfsetstrokecolor{oc@pms}}%
  \def\pgfsetfillopacity##1{%
    \ifx\tikz@fillcolor\pgfutil@empty\colorlet{oc@pmf}{.!\oc@op@pct{##1}!ocBlack}%
    \else\colorlet{oc@pmf}{\tikz@fillcolor!\oc@op@pct{##1}!ocBlack}\fi
    \pgfsetfillcolor{oc@pmf}}}

  \useasboundingbox (0,0) rectangle (4.76in,3.46in);#1\end{tikzpicture}}}

% Part pages (design 5e, option B): a recto with a charcoal band over the top 3.4 in
% (to the bleed at the top and sides) holding the masthead, the volume line and the light
% coral Roman part numeral; below it, on the paper, PART n (Fira, numeral in deep coral),
% the charcoal Cormorant title and a deep-coral rule-diamond-rule. No motif or epigraph.
% The opener of the part's first chapter is printed on its back, facing that chapter's
% first text page. Called by \part (common/interior-style.tex) on the recto it has
% already cleared to. \ocPartPaper is the page below the band: white, as the paper is
% (set it to ocCream for an ivory page).
\colorlet{ocPartPaper}{white}
\newsavebox{\oc@pt@text}

    {\ocDisplay\linespread{1}\fontsize{38}{42}\selectfont\color{ocBlack}\raggedright\hyphenpenalty10000\exhyphenpenalty10000 \ocBreakDashes{#2}\par}%

  \xdef\oc@op@numeral{#1}%

% Page coordinates: origin at the top-left corner of the paper (trim corner + bleed).
\newcommand{\oc@op@art}{%

  \setlength{\oc@opY}{\dimexpr3.75in+\ht\oc@op@text+\dp\oc@op@text+0.45in+0.5\oc@opH\relax}%

% Reads \chapter[short]{title}, an optional \label and an opening epigraph, typesets the
% opener (numbered main-matter chapters only), then calls the original \chapter.
\def\oc@centername{center}
\def\oc@skip#1{\def\oc@skip@k{#1}\oc@skip@loop}
\def\oc@skip@loop{\@ifnextchar\par{\oc@skip@par}{\oc@skip@k}}
\long\def\oc@skip@par#1{\oc@skip@loop}
\def\oc@ch@go{\@ifnextchar[{\oc@ch@opt}{\oc@ch@noopt}}
\long\def\oc@ch@noopt#1{\oc@ch@a{#1}{#1}}
\long\def\oc@ch@opt[#1]#2{\oc@ch@a{#1}{#2}}
\long\def\oc@ch@a#1#2{\def\oc@ch@short{#1}\def\oc@ch@title{#2}%
  \let\oc@ch@label\@empty\let\oc@ch@epi\@empty\oc@skip\oc@ch@lab}
\def\oc@ch@lab{\@ifnextchar\label{\oc@ch@getlab}{\oc@ch@afterlab}}
\def\oc@ch@getlab\label#1{\def\oc@ch@label{\label{#1}}\oc@skip\oc@ch@afterlab}
\def\oc@ch@afterlab{\@ifnextchar\begin{\oc@ch@begin}{\oc@ch@finish{}}}
\def\oc@ch@begin\begin#1{\def\oc@tmp{#1}%
  \ifx\oc@tmp\oc@centername\expandafter\@firstoftwo\else\expandafter\@secondoftwo\fi
  {\@ifnextchar\itshape{\oc@ch@epigraph}{\oc@ch@finish{\begin{center}}}}%
  {\oc@ch@finish{\begin{#1}}}}
\long\def\oc@ch@epigraph\itshape#1\end#2{\def\oc@tmp{#2}%
  \ifx\oc@tmp\oc@centername\expandafter\@firstoftwo\else\expandafter\@secondoftwo\fi
  {\def\oc@ch@epi{#1}\oc@ch@finish{}}%
  {\oc@ch@finish{\begin{center}\itshape#1\end{#2}}}}
\long\def\oc@ch@finish#1{%
  \if@mainmatter
    \begingroup\advance\c@chapter\@ne\xdef\oc@ch@num{\thechapter}\endgroup
    \begingroup\edef\oc@tmp{\endgroup\noexpand\oc@op@page{\oc@ch@num}%
      {\unexpanded\expandafter{\oc@ch@title}}{\noexpand\oc@ch@epi}}\oc@tmp
    \expandafter\oc@chapter@orig\expandafter[\expandafter{\oc@ch@short}]{\oc@ch@title}\oc@ch@label#1%
  \else
    \expandafter\oc@chapter@orig\expandafter[\expandafter{\oc@ch@short}]{\oc@ch@title}\oc@ch@label
    \ifx\oc@ch@epi\@empty\else\begin{center}\itshape\oc@ch@epi\end{center}\fi#1%
  \fi}
\AtBeginDocument{\let\oc@chapter@orig\chapter
  \renewcommand{\chapter}{\@ifstar{\oc@chapter@orig*}{\oc@ch@go}}}
