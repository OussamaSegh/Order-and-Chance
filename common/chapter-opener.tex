% Openers (design 5e). Part pages and chapter openers share this file.
%
% Part page: a full charcoal recto. Masthead, volume line, the coral part numeral,
% PART n, the ivory title and the part rule. No epigraph, and no illustration: the
% cover motif and \chaptermotif are not drawn here.
%
% Chapter opener: no band. On the white page, a small coral CHAPTER n (or
% APPENDIX A) label, then the black title in the body serif, left aligned, then
% a short coral rule. The opening italic and the prose follow on that same page.
% The first page uses the plain style (folio at the foot, no running head).
% Running heads on later pages are unchanged.
% Numbered chapters start on the next page, recto or verso: the class option is
% openany, and a blank is not inserted to force a side. A part's first chapter
% therefore still sits on the back of the part page.
%
% Epigraph: the chapter's opening \begin{center}\itshape ... \end{center} (directly after
% \chapter{...} and its optional \label) is the centred italic under the band; the source
% is unchanged.
%
% \chaptermotif is accepted and ignored. The print openers do not draw it, and they
% do not input cover-motif.tex.

% Pack the paragraph into lines no wider than the measure, then move words
% onto a short last line until it is at least about half a line.
\cs_new_protected:Npn \__oc_epi_join:N #1
  {
    \tl_clear:N \l__oc_epi_line_tl
    \seq_map_inline:Nn #1
      {
        \tl_if_blank:nF { ##1 }
          {
            \tl_if_empty:NTF \l__oc_epi_line_tl
              { \tl_set:Nn \l__oc_epi_line_tl { ##1 } }
              { \tl_put_right:Nn \l__oc_epi_line_tl { ~ ##1 } }
          }
      }
  }
\cs_new_protected:Npn \__oc_epi_pack:
  {
    \seq_set_split:NnV \l__oc_epi_words_seq { ~ } \l__oc_epi_tl
    \seq_clear:N \l__oc_epi_lines_seq
    \tl_clear:N \l__oc_epi_line_tl
    \seq_map_inline:Nn \l__oc_epi_words_seq
      {
        \tl_if_blank:nF { ##1 }
          {
            \tl_if_empty:NTF \l__oc_epi_line_tl
              { \tl_set:Nn \l__oc_epi_line_tl { ##1 } }
              {
                \settowidth {\dimen@} { \l__oc_epi_line_tl \ ##1 }
                \dim_compare:nNnTF { \dimen@ } > { \oc@epiW }
                  {
                    \seq_put_right:NV \l__oc_epi_lines_seq \l__oc_epi_line_tl
                    \tl_set:Nn \l__oc_epi_line_tl { ##1 }
                  }
                  { \tl_put_right:Nn \l__oc_epi_line_tl { ~ ##1 } }
              }
          }
      }
    \tl_if_empty:NF \l__oc_epi_line_tl
      { \seq_put_right:NV \l__oc_epi_lines_seq \l__oc_epi_line_tl }
    \int_step_inline:nn { 6 }
      {
        \int_step_inline:nnn { 1 } { \seq_count:N \l__oc_epi_lines_seq - 1 }
          {
            \tl_set:Nx \l__oc_epi_line_tl { \seq_item:Nn \l__oc_epi_lines_seq { ####1 } }
            \tl_set:Nx \l__oc_epi_last_tl { \seq_item:Nn \l__oc_epi_lines_seq { ####1 + 1 } }
            \settowidth {\dimen@} { \l__oc_epi_line_tl }
            \dim_set:Nn \l_tmpa_dim { \dimen@ }
            \settowidth {\dimen@} { \l__oc_epi_last_tl }
            \dim_compare:nNnT { \dimen@ } < { 0.82 \l_tmpa_dim }
              {
                \seq_set_split:NnV \l__oc_epi_prev_seq { ~ } \l__oc_epi_line_tl
                \int_compare:nNnT { \seq_count:N \l__oc_epi_prev_seq } > { 1 }
                  {
                    \tl_set:Nx \l__oc_epi_word_tl { \seq_item:Nn \l__oc_epi_prev_seq { -1 } }
                    \settowidth {\dimen@} { \l__oc_epi_word_tl \ \l__oc_epi_last_tl }
                    \dim_compare:nNnT { \dimen@ } < { \oc@epiW }
                      {
                        \seq_set_eq:NN \l__oc_epi_words_seq \l__oc_epi_prev_seq
                        \seq_pop_right:NN \l__oc_epi_words_seq \l_tmpa_tl
                        \__oc_epi_join:N \l__oc_epi_words_seq
                        \settowidth {\dimen@} { \l__oc_epi_line_tl }
                        \dim_set:Nn \l_tmpb_dim { \dimen@ }
                        \settowidth {\dimen@} { \l__oc_epi_word_tl \ \l__oc_epi_last_tl }
                        \dim_compare:nNnT { \l_tmpb_dim } > { \dimen@ }
                          {
                            \exp_args:NnnV \seq_set_item:Nnn \l__oc_epi_lines_seq
                              { ####1 } \l__oc_epi_line_tl
                            \tl_put_left:Nn \l__oc_epi_last_tl { ~ }
                            \tl_put_left:NV \l__oc_epi_last_tl \l__oc_epi_word_tl
                            \exp_args:NnnV \seq_set_item:Nnn \l__oc_epi_lines_seq
                              { ####1 + 1 } \l__oc_epi_last_tl
                          }
                      }
                  }
              }
          }
      }
    \seq_map_inline:Nn \l__oc_epi_lines_seq
      { \par \noindent ##1 \par }
  }
\cs_new_protected:Npn \oc@epi@split #1
  {
    \seq_set_split:Nnn \l__oc_epi_seq { \\ } { #1 }
    \bool_set_true:N \l__oc_epi_fit_bool

      {
        \seq_map_inline:Nn \l__oc_epi_seq

        \tl_replace_all:Nnn \l__oc_epi_tl { \\ } { ~ }

      }
  }

% ---------------------------------------------------------------- \chapter wrapper
% Reads \chapter[short]{title}, then any mix of \label and \chaptermark, then an
% opening \begin{center}\itshape. \chaptermark is kept and replayed after the
% real \chapter, so the short running head survives and the epigraph still
% uses the shared centred italic. Two chapters put \chaptermark before the epigraph
% (Volume 8, the unitary-matrices chapter, and Volume 12, log-correlated fields).
\def\oc@centername{center}
\def\oc@skip#1{\def\oc@skip@k{#1}\oc@skip@loop}
\def\oc@skip@loop{\@ifnextchar\par{\oc@skip@par}{\oc@skip@k}}
\long\def\oc@skip@par#1{\oc@skip@loop}
\def\oc@ch@go{\@ifnextchar[{\oc@ch@opt}{\oc@ch@noopt}}
\long\def\oc@ch@noopt#1{\oc@ch@a{#1}{#1}}
\long\def\oc@ch@opt[#1]#2{\oc@ch@a{#1}{#2}}
\long\def\oc@ch@a#1#2{\def\oc@ch@short{#1}\def\oc@ch@title{#2}%
  \let\oc@ch@label\@empty\let\oc@ch@mark\@empty\let\oc@ch@epi\@empty\oc@skip\oc@ch@lab}
\def\oc@ch@lab{%
  \@ifnextchar\label{\oc@ch@getlab}{%
    \@ifnextchar\chaptermark{\oc@ch@getmark}{\oc@ch@afterlab}}}
\long\def\oc@ch@getlab\label#1{%
  \begingroup\toks@=\expandafter{\oc@ch@label\label{#1}}%
  \xdef\oc@ch@label{\the\toks@}\endgroup
  \oc@skip\oc@ch@lab}
\long\def\oc@ch@getmark\chaptermark#1{%
  \begingroup\toks@=\expandafter{\oc@ch@mark\chaptermark{#1}}%
  \xdef\oc@ch@mark{\the\toks@}\endgroup
  \oc@skip\oc@ch@lab}
\def\oc@ch@afterlab{\@ifnextchar\begin{\oc@ch@begin}{\oc@ch@finish{}}}
\def\oc@ch@begin\begin#1{\def\oc@tmp{#1}%
  \ifx\oc@tmp\oc@centername\expandafter\@firstoftwo\else\expandafter\@secondoftwo\fi
  {\@ifnextchar\itshape{\oc@ch@epigraph}{\oc@ch@finish{\begin{center}}}}%
  {\oc@ch@finish{\begin{#1}}}}
\long\def\oc@ch@epigraph\itshape#1\end#2{\def\oc@tmp{#2}%
  \ifx\oc@tmp\oc@centername\expandafter\@firstoftwo\else\expandafter\@secondoftwo\fi
  {\def\oc@ch@epi{#1}\oc@ch@finish{}}%
  {\oc@ch@finish{\begin{center}\itshape#1\end{#2}}}}
\long\def\oc@ch@finish#1{%
  \if@mainmatter
    % Next page, either side. \chapter would otherwise honour \@openright and
    % insert a blank verso. Restore the flag before the body is read, so the
    % heading's \everypar (no indent on the first paragraph) survives.
    \@openrightfalse
    \expandafter\oc@chapter@orig\expandafter[\expandafter{\oc@ch@short}]{\oc@ch@title}%
    \@openrighttrue
    \oc@ch@label\oc@ch@mark#1%
  \else
    \expandafter\oc@chapter@orig\expandafter[\expandafter{\oc@ch@short}]{\oc@ch@title}\oc@ch@label\oc@ch@mark
    \ifx\oc@ch@epi\@empty\else\begin{center}\itshape\oc@ch@epi\end{center}\fi#1%
  \fi}
\AtBeginDocument{\let\oc@chapter@orig\chapter
  \renewcommand{\chapter}{\@ifstar{\oc@chapter@orig*}{\oc@ch@go}}}
\makeatother
