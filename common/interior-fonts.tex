% Shared, bundled text and mathematical faces for XeLaTeX interiors and figures.
\usepackage{fontspec}
\usepackage[math-style=TeX,bold-style=TeX]{unicode-math}
\IfFileExists{../common/fonts/STIXTwoText-Regular.otf}
  {\def\ocFontPath{../common/fonts/}}
  {\def\ocFontPath{common/fonts/}}
\setmainfont{STIXTwoText-Regular.otf}[
  Path=\ocFontPath,BoldFont=STIXTwoText-Bold.otf,
  ItalicFont=STIXTwoText-Italic.otf,BoldItalicFont=STIXTwoText-BoldItalic.otf,
  Ligatures=TeX]
\setsansfont{InterText-Regular.ttf}[
  Path=\ocFontPath,BoldFont=InterText-SemiBold.ttf,
  ItalicFont=InterText-Italic.ttf,BoldItalicFont=InterText-SemiBoldItalic.ttf,
  Ligatures=TeX]
\setmonofont{DejaVuSansMono.ttf}[
  Path=\ocFontPath,BoldFont=DejaVuSansMono-Bold.ttf,
  ItalicFont=DejaVuSansMono-Oblique.ttf,BoldItalicFont=DejaVuSansMono-BoldOblique.ttf,
  Scale=MatchLowercase]
\setmathfont{STIXTwoMath-Regular.otf}[Path=\ocFontPath]
% Use the font's Unicode bold alphabet for legacy vector notation, rather than
% requesting a nonexistent separate bold font through amsmath's boldsymbol.
\renewcommand{\boldsymbol}[1]{\symbf{#1}}
\def\blacksquare{\mdlgblksquare}

% Exact display faces already used by the approved covers.
\newfontfamily\ocCoverDisplay{Inter-Light.ttf}[
 Path=\ocFontPath,BoldFont=Inter-SemiBold.ttf,ItalicFont=Inter-LightItalic.ttf,
 BoldItalicFont=InterDisplay-SemiBoldItalic.ttf]
