% Packages and macros used by figures; shared by the books and tools/figure.tex.
\usepackage{tikz}
\usetikzlibrary{arrows.meta,calc,positioning,decorations.pathreplacing}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
% Diagram lettering. The 11 pt book sets body text in EB Garamond at 11.45 pt
% (Scale=1.05). Captions are Fira Sans at 10 pt: Medium for the figure label,
% Book for the prose (common/interior-style.tex). Diagram words use that same
% 10 pt Fira Sans Book. Notation stays in Garamond-Math, which at 10 pt and
% Scale=1.05 is 10.5 pt, the size of mathematics in a caption. The size is an
% absolute \fontsize so the 10 pt standalone run that builds EPUB plates
% matches the book. Geometry is not scaled.
%
% \tiny (6 pt), \scriptsize (8 pt), \footnotesize (9 pt) and \small (10 pt in
% this class, 9 pt in a 10 pt class) are the sizes figures actually request.
% Inside a picture they all become this one step, so an explicit font override
% cannot leave a label below it. A node that then asks for \bfseries keeps the
% bold sans (Fira Medium). Superscripts use the math script font, not these
% text commands. Part pages and chapter openers set their own faces and sizes
% in the node text or the font key, and those win over the picture default.
\newcommand{\ocFigText}{\sffamily\mdseries\upshape\fontsize{10}{12}\selectfont}
\tikzset{
  every picture/.append style={
    font=\ocFigText,
    execute at begin picture={%
      \let\tiny\ocFigText
      \let\scriptsize\ocFigText
      \let\footnotesize\ocFigText
      \let\small\ocFigText
    }%
  }%
}
