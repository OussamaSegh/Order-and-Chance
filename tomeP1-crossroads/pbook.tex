% Problem books call their exercises problems (the EPUB does the same through tools/epub.lua).
\renewenvironment{exercise}[1][]{%
  \refstepcounter{exo}%
  \begin{tcolorbox}[enhanced,breakable,colback=ocExercise!6,colframe=ocExercise,
    boxrule=0.6pt,arc=2pt,left=6pt,right=6pt,
    title={\textbf{Problem \theexo}\ifx\relax#1\relax\else\ \textit{(#1)}\fi}]}%
  {\end{tcolorbox}}
\renewenvironment{solution}[1]{%
  \par\medskip\noindent\textbf{\color{ocReminder}Solution of Problem~\ref{#1}.}\quad}%
  {\hfill$\blacksquare$\par\medskip}
\renewenvironment{hint}[1]{%
  \par\smallskip\noindent\textbf{\color{ocThink}Hint for Problem~\ref{#1}.}\quad}%
  {\par\smallskip}
