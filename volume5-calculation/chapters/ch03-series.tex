\chapter{Classic Series and Infinite Products}\label{ch:series}

\begin{center}\itshape
$\dfrac{\pi^2}6=1+\dfrac14+\dfrac19+\dfrac1{16}+\cdots$\\[0.6em]
$\dfrac2\pi=\dfrac{\sqrt2}2\cdot\dfrac{\sqrt{2+\sqrt2}}2\cdot\dfrac{\sqrt{2+\sqrt{2+\sqrt2}}}2\cdots$
\end{center}

In 1735 Euler announced that $1+\frac14+\frac19+\cdots=\frac{\pi^2}6$. The Basel problem had stopped the Bernoullis. His argument treated $\frac{\sin x}x$ as a polynomial with roots $\pm\pi,\pm2\pi,\dots$, hence as the product $\prod_n\bigl(1-\frac{x^2}{n^2\pi^2}\bigr)$. The coefficients of $x^2$ then say $-\frac16=-\sum_n\frac1{n^2\pi^2}$. He knew the step from a polynomial to an infinite product was not yet a proof, and he spent years checking the value by other routes.

The missing step is a formula for $\pi\cot\pi x$, a sum over its poles. Herglotz's argument proves it from continuity and a maximum. Integrate that formula and the sine is a product over its roots. Expand the formula in powers of $x$, and the Bernoulli numbers of Chapter~\ref{ch:finite} give every value $\zeta(2k)$. The same circle of ideas shows that $\pi^2$ is irrational.

Volume~1, Exercise~6.7, already gave $\zeta(2)=\frac{\pi^2}6$ from a polynomial of finite degree. This chapter adds two proofs. Chapter~\ref{ch:fourier} adds two by Fourier series, and Chapter~\ref{ch:complex} adds one by residues. Six proofs of one identity: each one is a different reason the same number appears.

\begin{think}
Sine vanishes at every integer multiple of $\pi$. A polynomial with those roots, and with value $1$ at $0$, would be $\prod(1-x/a_i)$. Does that product have to be $\sin x/x$? Multiply by $e^x$ and the roots do not change. Does the function?
\end{think}

\section{Infinite products}

\begin{definition}
Let $(u_n)_{n\ge1}$ be real numbers. The \emph{infinite product} $\prod_{n\ge1}u_n$ \emph{converges} if the partial products
$P_N=u_1\cdots u_N$ have a limit $P$, and its value is then $P$.
\end{definition}

One zero factor sends the whole product to $0$. So does $\prod\frac12$, whose partial products are $2^{-N}$. The products we use have factors near $1$ and a positive limit.

\begin{proposition}\label{prop:products}
Let $a_n>-1$ be real numbers with $\sum|a_n|<\infty$. Then $\prod(1+a_n)$ converges to a limit in $\left]0,+\infty\right[$, and
$\prod(1+a_n)=\exp\sum\ln(1+a_n)$. If the $a_n=a_n(x)$ are continuous functions of $x$ in an interval and
$\sum_n\sup_x|a_n(x)|<\infty$, the product is a continuous function of $x$.
\end{proposition}
\begin{proof}
For $|a|\le\frac12$,
\[
\begin{aligned}|\ln(1+a)|&=\Bigl|\sum_{k\ge1}\frac{(-1)^{k-1}a^k}k\Bigr|\eqbr\le\sum_k|a|^k=\frac{|a|}{1-|a|}\le2|a|.\end{aligned}
\]
The sequence $a_n$ tends to $0$, so $|a_n|\le\frac12$ for $n\ge n_0$. Absolute convergence of $\sum\ln(1+a_n)$ follows, and $P_N=\exp\sum_{n\le N}\ln(1+a_n)$ tends to
$\exp\sum_n\ln(1+a_n)>0$ because the exponential is continuous. In the second case the same bound, with the Weierstrass test, gives uniform convergence of
$\sum\ln(1+a_n(x))$ in $x$ (for $n\ge n_0$, where $\sup|a_n|\le\frac12$). Lemma~\ref{lem:uniform} makes that sum continuous, and the exponential of a continuous function is continuous.
\end{proof}

\begin{example}
$\prod_{n\ge2}\bigl(1-\frac1{n^2}\bigr)=\frac12$, since $\prod_{n=2}^N\frac{(n-1)(n+1)}{n\cdot n}=\frac{N+1}{2N}$ telescopes. At $N=4$ the partial product is $\frac58$. Check: $\frac34\cdot\frac89\cdot\frac{15}{16}=\frac58$.
\end{example}

\section{The partial fractions of the cotangent}

Near each integer $n$, one has $\pi\cot\pi x\approx\frac1{x-n}$: a simple pole of residue $1$. Euler's guess was that the function is the sum of these poles.

\begin{theorem}[Euler]\label{thm:cot}
For every real $x\notin\Z$,
\[
\begin{aligned}\pi\cot\pi x&=\frac1x+\sum_{n\ge1}\Bigl(\frac1{x+n}+\frac1{x-n}\Bigr)\eqbr=\frac1x+\sum_{n\ge1}\frac{2x}{x^2-n^2}.\end{aligned}
\]
\end{theorem}
\begin{proof}[Proof (Herglotz's trick)]
Let $f(x)=\pi\cot\pi x$ and $g(x)=\frac1x+\sum_{n\ge1}\frac{2x}{x^2-n^2}$, both defined on $\R\setminus\Z$. On a closed bounded interval that misses $\Z$, one has $\bigl|\frac{2x}{x^2-n^2}\bigr|\le\frac{2A}{n^2-A^2}$ for $n>A=\max|x|$. Lemma~\ref{lem:uniform} gives uniform convergence, so $g$ is continuous. Write $g_N(x)=\sum_{n=-N}^N\frac1{x+n}$, and pass to the limit $g=\lim g_N$. Four properties of $f$ agree with the same four properties of $g$.

\emph{Both are odd.} Cotangent is odd, and replacing $x$ by $-x$ changes the sign of every term of $g$.

\emph{Both have period $1$.} For $f$, that is the period of $\cot$. For $g$, shift the window: $g_N(x+1)=\sum_{m=-N+1}^{N+1}\frac1{x+m}=g_N(x)+\frac1{x+N+1}-\frac1{x-N}$. The two extra terms tend to $0$.

\emph{Both satisfy $\varphi\bigl(\frac x2\bigr)+\varphi\bigl(\frac{x+1}2\bigr)=2\varphi(x)$.} For $f$, set $y=\frac{\pi x}2$. The cofunction identity $\cot\bigl(y+\frac\pi2\bigr)=-\tan y$ gives
\[
\begin{aligned}\cot y-\tan y&=\frac{\cos^2y-\sin^2y}{\sin y\cos y}\eqbr=\frac{2\cos2y}{\sin2y}=2\cot2y.\end{aligned}
\]
For $g$, let $G_N(x)=g_N\bigl(\frac x2\bigr)+g_N\bigl(\frac{x+1}2\bigr)$. Then
\[
\begin{aligned}G_N(x)&=\sum_{n=-N}^N\Bigl(\frac2{x+2n}+\frac2{x+2n+1}\Bigr)\eqbr=2g_{2N}(x)+\frac2{x+2N+1},\end{aligned}
\]
Send $N\to\infty$ and the same relation appears for $g$.

\emph{Both are $\frac1x+o(1)$ at $0$.} For $g$, the tail is $g(x)-\frac1x=\sum_{n\ge1}\frac{2x}{x^2-n^2}$, and uniform convergence near $0$ sends it to $0$. For $f$, Taylor's formula gives $\sin\pi x=\pi x-\frac{(\pi x)^3}6+O(x^5)$ and $\cos\pi x=1-\frac{(\pi x)^2}2+O(x^4)$, so
\[
\begin{aligned}\pi\cot\pi x-\frac1x&=\frac{\pi x\cos\pi x-\sin\pi x}{x\sin\pi x}\eqbr=\frac{-\frac{(\pi x)^3}3+O(x^5)}{\pi x^2+O(x^4)}\to0.\end{aligned}
\]

Set $h=f-g$ off the integers. Then $h$ is continuous, odd, and of period $1$, it obeys the same averaging relation, and $h(x)\to0$ as $x\to0$. Periodicity spreads that limit to every integer. Define $h(n)=0$ on $\Z$. The function is now continuous on $\R$, and the averaging relation survives by continuity. A continuous periodic function attains its maximum $M$, say at $x_0$. The average of two values that are at most $M$ equals $M$ only if both equal $M$, so $h(x_0/2)=M$. Iterate: $h(x_0/2^k)=M$. Let $k\to\infty$. Then $M=h(0)=0$. The same argument on $-h$ shows that the minimum is $0$. Hence $h=0$.
\end{proof}

What just happened is a small maximum principle. The averaging relation $h(x/2)+h((x+1)/2)=2h(x)$ will not let a continuous periodic function have a non-zero peak. Chapter~\ref{ch:transform} proves the same expansion again, from Poisson summation. Chapter~\ref{ch:complex} turns the poles of $\pi\cot\pi z$ into a method for summing series.

\section{The product for the sine}

\begin{theorem}[Euler]\label{thm:sineproduct}
For every real $x$,
\[
\sin\pi x=\pi x\prod_{n\ge1}\Bigl(1-\frac{x^2}{n^2}\Bigr).
\]
\end{theorem}
\begin{proof}
Proposition~\ref{prop:products} says the product $P(x)=\pi x\prod_{n\ge1}\bigl(1-\frac{x^2}{n^2}\bigr)$ converges for every $x$. A factor may vanish, and that happens at an integer, where $P(x)=0$. On $\left]-1,1\right[$ every factor is positive, so $P$ is continuous there.

Fix $0<x<1$. On $[0,x]$ the summands $\frac{2t}{t^2-n^2}$ are continuous, and for $n\ge2$ they are bounded by $\frac2{n^2-1}$. The series in Theorem~\ref{thm:cot}, with the first term removed, therefore converges uniformly on $[0,x]$. The term $n=1$ is continuous on that interval as well. Integrate term by term from $0$ to $x$ (Lemma~\ref{lem:uniform}(c)):
\[
\begin{aligned}\int_0^x\Bigl(\pi\cot\pi t-\frac1t\Bigr)\dd t&=\sum_{n\ge1}\int_0^x\frac{2t}{t^2-n^2}\dd t\\&=\sum_{n\ge1}\ln\Bigl(1-\frac{x^2}{n^2}\Bigr).\end{aligned}
\]
The integrand on the left is the derivative of $\ln\frac{\sin\pi t}{\pi t}$, and that logarithm tends to $0$ at $0$. The left side is therefore $\ln\frac{\sin\pi x}{\pi x}$. Exponentiate: $\sin\pi x=P(x)$ for $0<x<1$. Both sides are odd and both vanish at $0$, so the identity holds on $\left]-1,1\right[$.

To carry the identity past $1$, write the partial products as
\[
\begin{aligned}P_N(x)&=\pi x\prod_{n=1}^N\frac{(n-x)(n+x)}{n^2}\eqbr=\frac{(-1)^N\pi}{(N!)^2}\prod_{k=-N}^N(x+k).\end{aligned}
\]
Then, for $x\ne N$,
\[
\begin{aligned}P_N(x+1)&=\frac{(-1)^N\pi}{(N!)^2}\prod_{k=-N+1}^{N+1}(x+k)\eqbr=P_N(x)\,\frac{x+N+1}{x-N},\end{aligned}
\]
Let $N\to\infty$. Then $P(x+1)=-P(x)$. The sine satisfies $\sin\pi(x+1)=-\sin\pi x$ as well, so the identity on $\left]-1,1\right[$ moves to $\left]0,2\right[$, then to every interval $\left]k-1,k+1\right[$, and so to all of $\R$.
\end{proof}

\begin{corollary}[Wallis, 1656; Vi\`ete, 1593]\label{cor:wallis}
\[
\begin{aligned}\frac\pi2&=\prod_{n\ge1}\frac{(2n)^2}{(2n-1)(2n+1)}\eqbr=\frac21\cdot\frac23\cdot\frac43\cdot\frac45\cdot\frac65\cdot\frac67\cdots,\end{aligned}
\]
\[
\begin{aligned}\frac2\pi&=\prod_{k\ge1}\cos\frac\pi{2^{k+1}}\eqbr=\frac{\sqrt2}2\cdot\frac{\sqrt{2+\sqrt2}}2\cdot\frac{\sqrt{2+\sqrt{2+\sqrt2}}}2\cdots\end{aligned}
\]
\end{corollary}
\begin{proof}
At $x=\frac12$ the sine product reads $1=\frac\pi2\prod_n\bigl(1-\frac1{4n^2}\bigr)=\frac\pi2\prod_n\frac{(2n-1)(2n+1)}{(2n)^2}$. The factors are positive and the product converges, so the product of the reciprocals is the reciprocal of the product. For Vi\`ete, iterate $\sin y=2\sin\frac y2\cos\frac y2$. For $y\ne0$,
\[
\begin{gathered}\sin y=2^m\sin\frac y{2^m}\prod_{k=1}^m\cos\frac y{2^k},\\\prod_{k=1}^m\cos\frac y{2^k}=\frac{\sin y}{y}\cdot\frac{y/2^m}{\sin(y/2^m)}\to\frac{\sin y}y.\end{gathered}
\]
At $y=\frac\pi2$ the limit is $\frac2\pi$. The first cosine is $\cos\frac\pi4=\frac{\sqrt2}2$, and the half-angle formula $\cos\frac\theta2=\sqrt{\frac{1+\cos\theta}2}$ builds the nested radicals.
\end{proof}

Wallis printed the first product in the \emph{Arithmetica Infinitorum} in 1656. It is the sine formula at the single point $x=\frac12$, and Figure~\ref{fig:rw03-partial} is that specialisation, one factor at a time.

Vi\`ete's formula, from 1593, is the first infinite product in the history of mathematics. It has a geometric meaning
(Figure~\ref{fig:viete}): the area of the regular $2^{k+1}$-gon inscribed in the unit circle is
$2^{k}\sin\frac{2\pi}{2^{k+1}}$, and passing from $2^k$ to $2^{k+1}$ sides multiplies the area by $\frac1{\cos(\pi/2^k)}$. Starting
from the square, of area $2$, and approaching the disc, of area $\pi$, the product of the factors $\cos\frac\pi{2^k}$ ($k\ge2$)
is $\frac2\pi$.

\section{Euler's formula for \texorpdfstring{$\zeta(2k)$}{zeta(2k)}}

Write $\zeta(s)=\sum_{n\ge1}n^{-s}$ for $s>1$. The case $s=2$ is the Basel sum from the opening page.

\portrait{euler}{Leonhard Euler (1707--1783). Portrait by Jakob Emanuel Handmann, 1753. Public domain.}

\begin{theorem}[Euler, 1735--1740]\label{thm:zeta2k}
For every $k\ge1$,
\[
\zeta(2k)=\frac{(-1)^{k+1}B_{2k}(2\pi)^{2k}}{2\,(2k)!}.
\]
In particular $\zeta(2)=\frac{\pi^2}6$, $\zeta(4)=\frac{\pi^4}{90}$, $\zeta(6)=\frac{\pi^6}{945}$, $\zeta(8)=\frac{\pi^8}{9450}$.
\end{theorem}
\begin{proof}
\emph{The analytic side.} Take $0<|x|<1$. Theorem~\ref{thm:cot} gives
\[
\begin{aligned}\pi x\cot\pi x&=1-2\sum_{n\ge1}\frac{x^2/n^2}{1-x^2/n^2}\eqbr=1-2\sum_{n\ge1}\sum_{k\ge1}\frac{x^{2k}}{n^{2k}}.\end{aligned}
\]
The double series converges absolutely, because $\sum_n\frac{x^2/n^2}{1-x^2/n^2}<\infty$. Lemma~\ref{lem:packets} lets the sums change order: $\pi x\cot\pi x=Z(x)$, where $Z(x)=1-2\sum_{k\ge1}\zeta(2k)x^{2k}$ is a power series for $|x|<1$. Multiply by $\frac{\sin\pi x}{\pi x}=S(x)=\sum_j\frac{(-1)^j(\pi x)^{2j}}{(2j+1)!}$, a series that converges everywhere. The Cauchy product of absolutely convergent series gives $\cos\pi x=S(x)Z(x)$ as power series on $\left]-1,1\right[$. Power series coefficients are unique, so the same identity holds as formal power series.

\emph{The formal side.} The proof of Proposition~\ref{prop:bernoulli}(b) gave the formal identity $(e^t-1)f(t)=\frac t2(e^t+1)$, where $f(t)=\sum_kB_{2k}\frac{t^{2k}}{(2k)!}$. Put $t=2iy$, as a formal series with complex coefficients, and multiply by $e^{-iy}$. Use $e^{a}e^{b}=e^{a+b}$:
\[
(e^{iy}-e^{-iy})f(2iy)=iy\,(e^{iy}+e^{-iy}).
\]
Match coefficients. As formal series, $e^{iy}-e^{-iy}=2i\sin y$ and $e^{iy}+e^{-iy}=2\cos y$, so $\sin y\cdot f(2iy)=y\cos y$. Set $y=\pi x$ and divide by $\pi x$. The result is $S(x)W(x)=\cos\pi x$, where $W(x)=f(2\pi ix)=\sum_k(-1)^kB_{2k}\frac{(2\pi x)^{2k}}{(2k)!}$.

Thus $S\,Z=S\,W$ in $\C[[x]]$. The series $S$ has constant term $1$, so it is invertible and $Z=W$. The coefficients of $x^{2k}$ are $-2\zeta(2k)=(-1)^kB_{2k}\frac{(2\pi)^{2k}}{(2k)!}$. The listed values are the cases $B_2=\frac16$, $B_4=-\frac1{30}$, $B_6=\frac1{42}$, $B_8=-\frac1{30}$.
\end{proof}

Figure~\ref{fig:rw03-basel} is the case $k=1$ drawn as a partial sum. The line is the closed form $\pi^2/6$. After twelve terms the sum is still about $0.080$ short of that line, approaching from below.

\end{figure}

Two consequences follow. Since $\zeta(2k)>0$, the signs of the $B_{2k}$ alternate: $(-1)^{k+1}B_{2k}>0$. And
$\zeta(2k)\to1$ as $k\to\infty$, because
$0<\zeta(2k)-1\le2^{-2k}+\int_2^\infty x^{-2k}\dd x=2^{-2k}\bigl(1+\frac2{2k-1}\bigr)\le3\cdot4^{-k}$. So
\[
|B_{2k}|=\frac{2(2k)!}{(2\pi)^{2k}}\zeta(2k)\sim\frac{2(2k)!}{(2\pi)^{2k}},
\]
which grows faster than any exponential: this is the growth announced in Chapter~\ref{ch:finite}. As for $\zeta(3)$, nobody knows a closed
form. Ap\'ery proved in 1978 that it is irrational. In 2026 Aabir Fauzan posted a preprint on Zenodo arguing that $\zeta(5)$ is irrational (doi:10.5281/zenodo.22826419). Alex Raymond Hyman formalised that argument in the Archive of Formal Proofs (\emph{Zeta\_5\_Irrational}, 26 October 2026). The formalisation's only oracle is \emph{holds}, from seventy-two numeric inequalities discharged by interval approximation. That argument is not a theorem of this volume, and it is not one to quote as proved here.

\begin{remark}
The coefficients $-2\zeta(2k)$ do not tend to $0$, so $\pi x\cot\pi x$ is a power series on $\left]-1,1\right[$ and stops being one at the endpoints. In the variable $t=2\pi ix$ that series is $t/(e^t-1)$, and its radius is exactly $2\pi$: the distance from $0$ to the poles $\pm2\pi i$. Chapter~\ref{ch:complex} is where that distance becomes a theorem about every power series.
\end{remark}

\section{Irrationality of \texorpdfstring{$\pi$}{pi}}

Lambert proved in 1761 that $\pi$ is irrational, by a continued fraction for $\tan x$. Niven's argument of 1947, in the line of Hermite, gives more: $\pi^2$ is irrational, and therefore $\pi$ is irrational.

\begin{theorem}\label{thm:piirrational}
$\pi^2$ is irrational. In particular $\pi$ is irrational.
\end{theorem}
\begin{proof}
\emph{The test polynomial.} For $n\ge1$ let $f(x)=\frac{x^n(1-x)^n}{n!}$. Then:
(i) $0<f(x)<\frac1{n!}$ for $0<x<1$;
(ii) every derivative $f^{(k)}(0)$ is an integer: writing $f=\frac1{n!}\sum_{i=n}^{2n}c_ix^i$ with integers $c_i$,
$f^{(k)}(0)=\frac{k!}{n!}c_k$, which is $0$ for $k<n$ and an integer for $k\ge n$;
(iii) every $f^{(k)}(1)$ is an integer, because $f(1-x)=f(x)$ gives $f^{(k)}(1)=(-1)^kf^{(k)}(0)$.

\emph{An integer.} Suppose $\pi^2=\frac ab$ with positive integers $a$, $b$. Let
\[
F(x)=b^n\sum_{k=0}^n(-1)^k\pi^{2n-2k}f^{(2k)}(x).
\]
Since $b^n\pi^{2n-2k}=a^{n-k}b^k$ is an integer, (ii) and (iii) show that $F(0)$ and $F(1)$ are integers. Since $f^{(2n+2)}=0$
($f$ has degree $2n$), the sum telescopes in $F''+\pi^2F=b^n\pi^{2n+2}f=\pi^2a^nf$. Therefore the function $\Phi(x)=F'(x)\sin\pi x-\pi F(x)\cos\pi x$ satisfies
\[
\begin{aligned}\Phi'(x)&=\bigl(F''(x)+\pi^2F(x)\bigr)\sin\pi x\eqbr=\pi^2a^nf(x)\sin\pi x,\end{aligned}
\]
and integrating over $[0,1]$,
\[
\begin{aligned}\pi a^n\int_0^1f(x)\sin\pi x\dd x&=\Bigl[\frac{\Phi(x)}\pi\Bigr]_0^1\eqbr=F(1)+F(0)\in\Z,\end{aligned}
\]
since $\Phi(1)=\pi F(1)$ and $\Phi(0)=-\pi F(0)$. \emph{Too small.} By (i), the left side lies strictly between $0$ and $\frac{\pi a^n}{n!}$. Since $\frac{\pi a^n}{n!}\to0$, choosing $n$ large makes it an
integer strictly between $0$ and $1$, a contradiction.
\end{proof}

The shape is worth keeping. Algebra makes the integral a non-zero integer. Analysis makes it smaller than $1$. Hermite used the same shape in 1873 to prove that $e$ is transcendental, and Lindemann used it in 1882 to prove that $\pi$ is transcendental, which settled the squaring of the circle.

\section{What we proved, and what comes next}

The cotangent is the sum of its poles (Theorem~\ref{thm:cot}), by continuity and a maximum. Integrating that sum gave the sine as a product over its roots (Theorem~\ref{thm:sineproduct}), and with it Wallis's product and Vi\`ete's product. Comparing coefficients with the Bernoulli numbers gave every $\zeta(2k)$ as a rational multiple of a power of $\pi$ (Theorem~\ref{thm:zeta2k}), and the size of $B_{2k}$. An integral that the algebra forces to be an integer, and that the estimate forces below $1$, proved that $\pi^2$ is irrational.

The cotangent sees even powers only, so $\zeta(3)$ and $\zeta(5)$ stay out of reach. The think box has a straight answer. The function $e^x\frac{\sin x}x$ has the same roots as $\frac{\sin x}x$, so the roots do not determine the function by themselves. The proof used a second fact, the relation that ties $\varphi(x)$ to $\varphi(\frac x2)$ and $\varphi(\frac{x+1}2)$.

Chapter~\ref{ch:integrals} turns from sums and products to integrals. Exercise~\ref{ex:ritelli} already lets the two trade places: one double integral, computed in both orders, sums the Basel series.

\section{Exercises}

\begin{exercise}[$\star$]\label{ex:leibniz}
Prove $\frac1{1+x^2}=\sum_{k=0}^{n-1}(-1)^kx^{2k}+\frac{(-1)^nx^{2n}}{1+x^2}$, and from it Leibniz's series $\frac\pi4=1-\frac13+\frac15-\frac17+\cdots$. After $n$ terms the error is at most $\frac1{2n+1}$.
\end{exercise}

\begin{exercise}[$\star$]\label{ex:cotsums}
From Theorem~\ref{thm:cot}, prove that for real $a\notin\Z$,
\[
\sum_{n\ge1}\frac1{n^2-a^2}=\frac1{2a^2}-\frac{\pi\cot\pi a}{2a},
\]
Evaluate the sum at $a=\frac12$ and check that value by telescoping. What is the limit as $a\to0$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:zetavalues}
Use Theorem~\ref{thm:zeta2k} to compute $\zeta(6)$ and $\zeta(8)$. Then prove $\sum_{n\ge0}\frac1{(2n+1)^4}=\frac{\pi^4}{96}$ and $\sum_{n\ge1}\frac{(-1)^{n-1}}{n^2}=\frac{\pi^2}{12}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:ritelli}
\emph{Basel by a double integral.} The integrand is continuous and non-negative on $\left]0,\infty\right[^2$. Tonelli's theorem equates the two iterated integrals of a non-negative function. We admit it here; Volume~7, Theorem~9.5, proves it. Compute $\int_0^\infty\!\int_0^\infty\frac{\mathrm dx\,\mathrm dy}{(1+y)(1+x^2y)}$ in both orders. Deduce $\int_0^1\frac{-\ln x}{1-x^2}\dd x=\frac{\pi^2}8$, and then $\zeta(2)=\frac{\pi^2}6$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:eirrational}
(a) Prove that $e$ is irrational. If $e=\frac pq$, study $q!\,e-\sum_{k\le q}\frac{q!}{k!}$.
(b) Prove that $e$ is not a root of a quadratic $ae^2+be+c=0$ with integers $a,b,c$ not all zero. Rewrite the relation as $ae+b+\frac ce=0$ and multiply by $n!$ for a suitable $n$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{ex:sqrtperiod}
Let $d\ge2$ be an integer that is not a square, and set $a_0=\lfloor\sqrt d\rfloor$. Volume~2, Chapter~9, proved that the continued fraction of $\sqrt d$ is eventually periodic. Call a quadratic irrational $x$ \emph{reduced} when $x>1$ and the conjugate $x'$ satisfies $-1<x'<0$.
\begin{enumerate}[label=(\alph*)]
\item Prove that if $x$ is reduced, then every complete quotient $x_n$ is reduced. Here $x_0=x$ and $x_{n+1}=\frac1{x_n-\lfloor x_n\rfloor}$.
\item Prove that a reduced $x_n$ satisfies $\lfloor x_n\rfloor=\bigl\lfloor-\frac1{x_{n+1}'}\bigr\rfloor$, so $x_{n+1}$ determines $x_n$. Conclude that the continued fraction of a reduced quadratic irrational is purely periodic.
\item Apply the result to $\sqrt d+a_0$ and conclude $\sqrt d=[a_0;\overline{a_1,\dots,a_{\ell-1},2a_0}]$. The period ends with $2a_0$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{pb:coprime52}
Prove
\[
\sum_{\substack{a,b\ge1\\\gcd(a,b)=1}}\frac1{a^2b^2}=\frac52.
\]
For $s>1$, compute $\sum_{\gcd(a,b)=1}(ab)^{-s}$, and read the sum as a Dirichlet series $\sum_nc_nn^{-s}$ with an explicit coefficient $c_n$.
\end{exercise}

\begin{exercise}[$\star$]\label{ex:rw03a}
Prove that $\prod_{n=2}^{N}\bigl(1-\frac1{n^2}\bigr)=\frac{N+1}{2N}$ for every integer $N\ge2$, and find the limit as $N\to\infty$. Check $N=3$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:rw03b}
From $\zeta(2)=\pi^2/6$, evaluate $\sum_{k\ge1}(2k)^{-2}$ and $\sum_{k\ge0}(2k+1)^{-2}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:rw03c}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{ex:rw03d}

\begin{hint}{ex:leibniz}
Integrate on $[0,1]$. The remainder is bounded by $\int_0^1x^{2n}\dd x$.

\begin{hint}{ex:cotsums}

\begin{hint}{ex:zetavalues}
The even terms are $\sum\frac1{(2n)^4}=\frac{\zeta(4)}{16}$. For the alternating series, remove twice the even terms.
\end{hint}
\begin{hint}{ex:ritelli}
One order gives $\int_0^\infty\frac{\mathrm dx}{1+x^2y}=\frac\pi{2\sqrt y}$. The other order wants partial fractions of $\frac1{(1+y)(1+x^2y)}$ in $y$. Fold $\left]1,\infty\right[$ onto $\left]0,1\right[$ by $x\mapsto\frac1x$, then expand $\frac1{1-x^2}$.
\end{hint}
\begin{hint}{ex:eirrational}
In (a) you get an integer sitting in $\left]0,1\right[$. In (b), both $n!\,e$ and $\frac{n!}e$ sit near integers. The errors have controlled signs. Take $n$ large and compare the sizes of the remainders.
\end{hint}
\begin{hint}{ex:sqrtperiod}
(a) Conjugate: $x_{n+1}'=\frac1{x_n'-a_n}$. (b) From $x_n'=a_n+\frac1{x_{n+1}'}$ one gets $-\frac1{x_{n+1}'}=a_n-x_n'\in\left]a_n,a_n+1\right[$. If $x_m=x_{m+\ell}$, step backwards. (c) The number $\sqrt d+a_0$ has integer part $2a_0$ and the same fractional part as $\sqrt d$.
\end{hint}

\end{hint}

\section{Solutions}

\begin{solution}{ex:leibniz}
\idea Sum the geometric series, keep the remainder, and integrate.
\writeup The finite sum is $\sum_{k<n}(-x^2)^k=\frac{1-(-x^2)^n}{1+x^2}$. Integrate over $[0,1]$:
\[
\begin{aligned}\frac\pi4&=\int_0^1\frac{\mathrm dx}{1+x^2}\\&=\sum_{k=0}^{n-1}\frac{(-1)^k}{2k+1}+(-1)^n\int_0^1\frac{x^{2n}}{1+x^2}\dd x,\end{aligned}
\]
The remainder is at most $\int_0^1x^{2n}\dd x=\frac1{2n+1}$ in absolute value, and it tends to $0$. That is the series. The bound is slow: a million terms give about six digits.
\end{solution}

\begin{solution}{ex:cotsums}
\idea Theorem~\ref{thm:cot} is this sum, up to a sign and a constant factor.
\writeup For $a\notin\Z$, the cotangent expansion is $\sum_{n\ge1}\frac{2a}{a^2-n^2}=\pi\cot\pi a-\frac1a$, so
$\sum_{n\ge1}\frac1{n^2-a^2}=-\frac1{2a}\bigl(\pi\cot\pi a-\frac1a\bigr)=\frac1{2a^2}-\frac{\pi\cot\pi a}{2a}$. At $a=\frac12$, $\cot\frac\pi2=0$, so the sum is
$2$. The same value comes from the telescope $\sum_{n\ge1}\frac1{n^2-1/4}=\sum_{n\ge1}\bigl(\frac2{2n-1}-\frac2{2n+1}\bigr)=2$. As $a\to0$,
$\frac1{2a^2}-\frac{\pi\cot\pi a}{2a}=\frac{1-\pi a\cot\pi a}{2a^2}=\frac{\pi^2a^2/3+O(a^4)}{2a^2}\to\frac{\pi^2}6$, and the left side tends to
$\zeta(2)$, by uniform convergence near $0$. That is Basel again, read off the cotangent.
\end{solution}

\begin{solution}{ex:zetavalues}
\idea Insert the Bernoulli numbers, then split odd and even denominators.
\writeup The value $B_6=\frac1{42}$ gives $\zeta(6)=\frac{(2\pi)^6}{42\cdot2\cdot720}=\frac{64\pi^6}{60480}=\frac{\pi^6}{945}$. The value $B_8=-\frac1{30}$ gives
$\zeta(8)=\frac{(2\pi)^8}{30\cdot2\cdot40320}=\frac{256\pi^8}{2419200}=\frac{\pi^8}{9450}$. The even terms of $\zeta(4)$ sum to $\frac{\zeta(4)}{16}$, so
$\sum\frac1{(2n+1)^4}=\frac{15}{16}\cdot\frac{\pi^4}{90}=\frac{\pi^4}{96}$. The alternating sum is $\sum\frac{(-1)^{n-1}}{n^2}=\zeta(2)-2\cdot\frac{\zeta(2)}4=\frac{\zeta(2)}2=\frac{\pi^2}{12}$. Absolute convergence justifies the rearrangement.
\end{solution}

\begin{solution}{ex:ritelli}
\idea Evaluate one double integral in both orders. The $x$-integral is two arctangents and produces $\pi\cdot\frac\pi2$. The $y$-integral produces a logarithm, and that logarithm sums the odd squares.
\writeup Let $I$ be the integral of the positive continuous function $\frac1{(1+y)(1+x^2y)}$. Integrate in $x$ first. With
$x=\frac u{\sqrt y}$, $\int_0^\infty\frac{\mathrm dx}{1+x^2y}=\frac1{\sqrt y}\int_0^\infty\frac{\mathrm du}{1+u^2}=\frac\pi{2\sqrt y}$. Then with $y=t^2$,
$I=\frac\pi2\int_0^\infty\frac{\mathrm dy}{\sqrt y(1+y)}=\frac\pi2\int_0^\infty\frac{2\,\mathrm dt}{1+t^2}=\frac{\pi^2}2$. Now integrate in $y$ first, at $x\ne1$:
\[
\frac1{(1+y)(1+x^2y)}=\frac1{1-x^2}\Bigl(\frac1{1+y}-\frac{x^2}{1+x^2y}\Bigr),
\]
so $\int_0^\infty\frac{\mathrm dy}{(1+y)(1+x^2y)}=\frac1{1-x^2}\Bigl[\ln\frac{1+y}{1+x^2y}\Bigr]_0^\infty=\frac{\ln(1/x^2)}{1-x^2}=\frac{-2\ln x}{1-x^2}$, a
positive function, and continuity gives the value $1$ at $x=1$. Tonelli's theorem, admitted in the exercise, equates the iterated integrals, so $\int_0^\infty\frac{-2\ln x}{1-x^2}\dd x=\frac{\pi^2}2$. The substitution $x=\frac1u$ carries $\int_1^\infty\frac{-\ln x}{1-x^2}\dd x$ to $\int_0^1\frac{\ln u}{1-u^{-2}}\cdot\frac{\mathrm du}{u^2}=\int_0^1\frac{-\ln u}{1-u^2}\dd u$, hence
$\int_0^1\frac{-\ln x}{1-x^2}\dd x=\frac{\pi^2}8$. On $\left]0,1\right[$ expand $\frac{-\ln x}{1-x^2}=\sum_{n\ge0}(-\ln x)x^{2n}$, a series of non-negative terms. Integration by parts gives $\int_0^1(-\ln x)x^{2n}\dd x=\frac1{(2n+1)^2}$. A finite sum may pass inside the integral, so
the integral of $s_N(x)=\sum_{n=0}^N(-\ln x)x^{2n}$ is $\sum_{n=0}^N\frac1{(2n+1)^2}$. Fix $\varepsilon>0$ and choose $\delta\in(0,\frac12)$ so that $\frac{-\ln x}{1-x^2}$ integrates to less than $\varepsilon$ on each of $(0,\delta)$ and $(1-\delta,1)$. On $[\delta,1-\delta]$ one has $-\ln x\le-\ln\delta$ and $x\le1-\delta$, hence $1-x^2\ge1-(1-\delta)^2=2\delta-\delta^2\ge\delta$. The remainder $(-\ln x)\,x^{2N+2}/(1-x^2)$ is then at most $(-\ln\delta)\,(1-\delta)^{2N+2}/\delta$, and its integral tends to $0$. On each end interval the remainder is at most the full integrand, so those integrals stay below $\varepsilon$. Therefore $\int_0^1 s_N\to\frac{\pi^2}8$, which is $\sum_{n\ge0}\frac1{(2n+1)^2}=\frac{\pi^2}8$. The even terms of $\zeta(2)$ sum to $\frac{\zeta(2)}4$, so $\zeta(2)=\frac43\cdot\frac{\pi^2}8=\frac{\pi^2}6$.
\end{solution}

\begin{solution}{ex:eirrational}
\idea Multiply the series for $e$ by $n!$. What remains is an integer plus a tail of size about $\frac1n$. Part (b) needs the tails of $e$ and of $\frac1e$ together, and the alternating tail of $\frac1e$ has a known sign.
\writeup (a) If $e=\frac pq$, then $N=q!\,e-\sum_{k=0}^q\frac{q!}{k!}$ is an integer. The tail is $N=\sum_{k>q}\frac{q!}{k!}=\frac1{q+1}+\frac1{(q+1)(q+2)}+\cdots<\sum_{j\ge1}\frac1{(q+1)^j}=\frac1q\le1$, and $N>0$. No such integer exists.

(b) Suppose $ae^2+be+c=0$, so $ae+b+\frac ce=0$, with $a$ and $c$ not both zero (if $a=c=0$ then $b=0$). Replace the triple by its opposite if needed, and assume $a\ge0$. If $a=0$ then $c\ne0$ and $\frac ce=-b$, which contradicts the irrationality of $e$ from (a). Thus $a>0$. For
$n\ge1$ write $n!\,e=A_n+r_n$ and $n!\,e^{-1}=C_n+s_n$ with integers $A_n=\sum_{k\le n}\frac{n!}{k!}$,
$C_n=\sum_{k\le n}\frac{(-1)^kn!}{k!}$ and $0<r_n<\frac1n$ by (a); for the alternating tail
$s_n=(-1)^{n+1}\bigl(\frac1{n+1}-\frac1{(n+1)(n+2)}+\cdots\bigr)$. So $s_n$ has the sign of $(-1)^{n+1}$ and $|s_n|<\frac1{n+1}$. Multiply the relation by $n!$. Then $aA_n+bn!+cC_n=-(ar_n+cs_n)$ is an integer. If $c\ge0$, take $n$ odd, so $s_n>0$. If $c<0$, take $n$ even, so $s_n<0$ and $cs_n>0$. In both cases $ar_n+cs_n>0$, and for large $n$ it is less than $\frac{a+|c|}n<1$. The integer $-(ar_n+cs_n)$ would then lie strictly between $-1$ and $0$. No integer does.
\end{solution}

\begin{solution}{ex:sqrtperiod}
\idea Reduced numbers stay reduced under the continued-fraction step, and conjugates invert that step. An eventually periodic invertible map is periodic from the first term.
\writeup (a) Let $x_n$ be reduced and set $a_n=\lfloor x_n\rfloor\ge1$. Then $x_{n+1}=\frac1{x_n-a_n}>1$. Conjugation is a field automorphism of $\Q(\sqrt d)$ that fixes $\Q$, so $x_{n+1}'=\frac1{x_n'-a_n}$. Since $x_n'-a_n<-1$, one has $-1<x_{n+1}'<0$.

(b) Start from $x_n=a_n+\frac1{x_{n+1}}$ and conjugate: $x_n'=a_n+\frac1{x_{n+1}'}$, so $-\frac1{x_{n+1}'}=a_n-x_n'$. The inequality $-1<x_n'<0$ puts that quantity in $\left]a_n,a_n+1\right[$. Hence $a_n=\bigl\lfloor-\frac1{x_{n+1}'}\bigr\rfloor$, and $x_n=a_n+\frac1{x_{n+1}}$ is determined by $x_{n+1}$. Lagrange's theorem (Volume~2, Chapter~9) says $(x_n)$ is eventually periodic: $x_m=x_{m+\ell}$ for some $m\ge0$ and $\ell\ge1$. If $m\ge1$, the inverse step gives $x_{m-1}=x_{m+\ell-1}$. Repeat until $x_0=x_\ell$. The expansion is purely periodic.

(c) Set $x=\sqrt d+a_0$. Then $x>1$ and $x'=a_0-\sqrt d\in\left]-1,0\right[$, so $x$ is reduced and its expansion is purely periodic: $x=[\overline{b_0;b_1,\dots,b_{\ell-1}}]$. The integer part is $b_0=2a_0$, and $x-2a_0=\sqrt d-a_0$. From the first complete quotient on, $x$ and $\sqrt d$ agree, so $b_k=a_k$ for $k\ge1$. Therefore $\sqrt d=x-a_0=[a_0;a_1,\dots,a_{\ell-1},2a_0,a_1,\dots]$, that is $\sqrt d=[a_0;\overline{a_1,\dots,a_{\ell-1},2a_0}]$. For example $\sqrt{14}=[3;\overline{1,2,1,6}]$.
\end{solution}

\begin{solution}{pb:coprime52}
\idea Factor out the gcd. A sum over all pairs is a sum over $g$ and a sum over coprime pairs, and $g$ contributes $\zeta(2s)$. The coprime sum is $\frac{\zeta(s)^2}{\zeta(2s)}$. At $s=2$ the known values of $\zeta(2)$ and $\zeta(4)$ finish the computation.

\writeup Let $s>1$. Every series here has positive terms, so the order of summation is free. Each pair $(a,b)$ of positive integers is uniquely $(ga',gb')$ with $g=\gcd(a,b)$ and $\gcd(a',b')=1$. Therefore
\[
\zeta(s)^2=\sum_{a,b\ge1}\frac1{(ab)^s}=\sum_{g\ge1}\frac1{g^{2s}}\sum_{\gcd(a',b')=1}\frac1{(a'b')^s}=\zeta(2s)\,S(s),
\]
so $S(s)=\frac{\zeta(s)^2}{\zeta(2s)}$. With $\zeta(2)=\frac{\pi^2}6$ and $\zeta(4)=\frac{\pi^4}{90}$ (Theorem~\ref{thm:zeta2k}),
$S(2)=\frac{\pi^4}{36}\cdot\frac{90}{\pi^4}=\frac52$.

Group by the product $n=a'b'$. Then $S(s)=\sum_nc_nn^{-s}$, where $c_n$ counts ordered coprime pairs $(a',b')$ with $a'b'=n$. Coprimality sends each prime power $p^e\parallel n$ entirely to $a'$ or entirely to $b'$, so $c_n=2^{\omega(n)}$, with $\omega(n)$ the number of distinct prime factors of $n$. Hence $\sum_n2^{\omega(n)}n^{-s}=\frac{\zeta(s)^2}{\zeta(2s)}$.
\tools Volume~2, Chapters~1 and~4 (gcd, multiplicative functions); Theorem~\ref{thm:zeta2k} ($\zeta(2)$, $\zeta(4)$).
\end{solution}

\begin{solution}{ex:rw03a}
\idea The general factor is a ratio of consecutive integers, so the product is a telescope.
\writeup $1-\frac1{n^2}=\frac{(n-1)(n+1)}{n^2}$, hence
\[
\prod_{n=2}^N\Bigl(1-\frac1{n^2}\Bigr)=\prod_{n=2}^N\frac{(n-1)(n+1)}{n\cdot n}=\frac{1\cdot(N+1)}{2\cdot N}=\frac{N+1}{2N}.
\]
As $N\to\infty$ the quotient tends to $\frac12$. For $N=3$ both sides equal $\frac34\cdot\frac89=\frac23$, and $\frac{4}{6}=\frac23$.
\end{solution}

\begin{solution}{ex:rw03b}
\idea Separate even and odd denominators. The even terms are a copy of $\zeta(2)$, scaled by $\frac14$.
\writeup $\sum_{k\ge1}(2k)^{-2}=\frac14\sum_{k\ge1}k^{-2}=\frac14\cdot\frac{\pi^2}6=\frac{\pi^2}{24}$. The odd terms are the rest:
\[
\sum_{k\ge0}(2k+1)^{-2}=\zeta(2)-\frac{\pi^2}{24}=\frac{\pi^2}6-\frac{\pi^2}{24}=\frac{\pi^2}8.
\]
The two pieces add back to $\frac{\pi^2}6$. Numerically, $\frac{\pi^2}{24}\approx0.411$ and the even partial sum $\frac14+\frac1{16}=\frac5{16}=0.3125$ is still short of it; $\frac{\pi^2}8\approx1.234$ and $1+\frac19+\frac1{25}=\frac{259}{225}\approx1.151$ is still short of that.
\end{solution}

\begin{solution}{ex:rw03c}
\idea The tail after $\frac54$ sits under the integral of $x^{-2}$ from $2$ onward, and that integral is $\frac12$.
\writeup For $n\ge3$ and $x\in[n-1,n]$ one has $x\le n$, so $x^{-2}\ge n^{-2}$ and $\frac1{n^2}\le\int_{n-1}^n x^{-2}\dd x$. Summing from $n=3$ gives
\[
\sum_{n\ge3}\frac1{n^2}\le\int_2^\infty x^{-2}\dd x=\frac12.
\]
Therefore $0<\frac{\pi^2}6-\bigl(1+\frac14\bigr)=\frac{\pi^2}6-\frac54\le\frac12$. The numerical gap is $\frac{\pi^2}6-\frac54\approx1.64493-1.25=0.395$, which is less than $\frac12$.
\end{solution}

\begin{solution}{ex:rw03d}
\idea The first two factors are $\cos\frac\pi4$ and $\cos\frac\pi8$, from the half-angle formula in the proof of the corollary.
\writeup $\frac{\sqrt2}2\approx0.707$ and $\frac{\sqrt{2+\sqrt2}}2\approx0.924$. Their product is about $0.653$. Since $\frac2\pi\approx0.637$, the two-factor product is about $0.017$ above the limit. Figure~\ref{fig:rw03-wallis} continues the same list: four factors give about $0.638$.
\end{solution}
