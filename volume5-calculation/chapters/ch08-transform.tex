\chapter{The Fourier Transform and Poisson Summation}\label{ch:transform}

\begin{center}\itshape
I have had my results for a long time: but I do not yet know how I am to arrive at them.\\
\upshape --- a saying attributed to Gauss
\end{center}

Add $e^{-\pi n^2/100}$ over every integer $n$. The first terms fall slowly, and a few dozen of them still contribute. The sum begins $10.000000\ldots$, and the first digit that is not zero sits $135$ places after the decimal point. A pocket calculator prints $10$. The sum is larger than $10$.

Why $10$, the square root of $100$? One identity answers that, and it is the centre of this chapter.

On the circle a frequency is an integer. On the line every real number is a frequency, and a Fourier series becomes a Fourier integral. The arguments run parallel to Chapter~\ref{ch:fourier}. The new function is the Gaussian $e^{-\pi x^2}$: it equals its own transform, and it smooths the integrals below. The bridge between the two settings is the \emph{Poisson summation formula} $\sum_nf(n)=\sum_n\hat f(n)$. For this transform, the integers are their own dual.

That formula does three jobs here. It turns a slow sum into a fast one, the theta relation that Chapter~\ref{ch:zeta} uses for $\zeta$. It gives a second proof of Euler's partial fractions for $\pi\cot\pi x$. It fixes the sign of the quadratic Gauss sum, the sign left open in Volume~1 and in Volume~2, Exercise~6.7, which Gauss settled only in 1805. Two facts about information close the chapter. Samples at the right rate determine a band-limited signal (Shannon). A function and its transform cannot both be narrow (Heisenberg).

\begin{think}
On the circle the coefficients $c_n(f)$ are indexed by $\Z$. On the line $\hat f(\xi)$ is a number for every real $\xi$. What should $\sum_{n\in\Z}f(n)$ have in common with $\hat f$? Try $f(x)=e^{-\pi tx^2}$ with $t$ very small. Consecutive integers then see almost the same value, so the sum is close to $\int f$. That integral is $\hat f(0)$.
\end{think}

\section{The Fourier transform}

In this chapter, \emph{integrable} means continuous on $\R$ with $\int_{\R}|f|<\infty$. Write $\|f\|_1=\int|f|$.

\begin{definition}
The \emph{Fourier transform} of an integrable $f$ is
\[
\hat f(\xi)=\int_{\R}f(x)e^{-2\pi ix\xi}\dd x\qquad(\xi\in\R).
\]
\end{definition}

The $2\pi$ in the exponent keeps the formulas symmetric. A frequency is then a number of cycles per unit length.

\begin{proposition}\label{prop:ftbasic}
Let $f$ be integrable. Then $\hat f$ is continuous and $|\hat f|\le\|f\|_1$. For real $a$ and $\lambda>0$:
\begin{enumerate}[label=(\alph*)]
\item $f(x-a)$ has transform $e^{-2\pi ia\xi}\hat f(\xi)$, and $e^{2\pi iax}f(x)$ has transform $\hat f(\xi-a)$;
\item $f(x/\lambda)$ has transform $\lambda\hat f(\lambda\xi)$;
\item if $f$ is $C^1$ and $f'$ is integrable, then $\widehat{f'}(\xi)=2\pi i\xi\hat f(\xi)$;
\item if $xf(x)$ is integrable, then $\hat f$ is $C^1$ and $\hat f\,'$ is the transform of $-2\pi ixf(x)$.
\end{enumerate}
\end{proposition}
\begin{proof}
\emph{Bound.} Continuity of $\hat f$, and the estimate $|\hat f|\le\|f\|_1$, are Theorem~\ref{thm:domination} with the integrable majorant $|f(x)|$, which does not depend on $\xi$.

\emph{Shifts.} (a) and (b) are the substitutions $x\mapsto x+a$ and $x\mapsto\lambda x$.

\emph{Derivative.} For (c), integrability of $f'$ forces $f(x)=f(0)+\int_0^xf'$ to have limits at $\pm\infty$. Those limits are $0$, because $f$ itself is integrable. Integrate by parts on $[-A,B]$ and let $A,B\to\infty$: $\int f'(x)e^{-2\pi ix\xi}\dd x=2\pi i\xi\int f(x)e^{-2\pi ix\xi}\dd x$.

\emph{Multiplication by $x$.} For (d), the $\xi$-derivative of the integrand is $-2\pi ixf(x)e^{-2\pi ix\xi}$, and $|-2\pi ixf(x)e^{-2\pi ix\xi}|\le 2\pi|xf(x)|$. Theorem~\ref{thm:domination} lets the derivative pass under the integral.
\end{proof}

Differentiation on one side is multiplication by $2\pi i\xi$ on the other. Smoothness of $f$ becomes decay of $\hat f$: if $f,\dots,f^{(k)}$ are integrable, then $|\hat f(\xi)|\le\frac{\|f^{(k)}\|_1}{(2\pi|\xi|)^k}$.

\begin{theorem}\label{thm:gaussianft}
The Gaussian $g(x)=e^{-\pi x^2}$ satisfies $\hat g=g$. More generally, for $a>0$, $e^{-\pi x^2/a^2}$ has transform $ae^{-\pi a^2\xi^2}$.
\end{theorem}
\begin{proof}
\emph{A differential equation.} Since $g'(x)=-2\pi xg(x)$, Proposition~\ref{prop:ftbasic}(d) and~(c) give
\[
\begin{aligned}\hat g\,'(\xi)&=\widehat{-2\pi ixg}(\xi)=i\,\widehat{g'}(\xi)\eqbr=i\cdot2\pi i\xi\,\hat g(\xi)=-2\pi\xi\hat g(\xi).\end{aligned}
\]
So $\bigl(e^{\pi\xi^2}\hat g\bigr)'=0$. Hence $\hat g(\xi)=\hat g(0)e^{-\pi\xi^2}$. The constant is the integral $\hat g(0)=\int e^{-\pi x^2}\dd x=1$ (Example~\ref{ex:gaussfeynman}, with $x=u/\sqrt\pi$).

\emph{A change of scale.} Proposition~\ref{prop:ftbasic}(b) with $\lambda=a$ gives the general case.
\end{proof}

\section{Inversion and Plancherel}

Inversion needs two tools. One interchanges integrals over the whole plane, by reducing the question to Fubini on a square. The other says that a tall thin Gaussian, of integral $1$, reproduces the value of a continuous function.

\begin{lemma}[Fubini on the plane]\label{lem:fubiniplane}
Let $h$ be continuous on $\R^2$ with $|h(x,y)|\le a(x)b(y)$, where $a,b\ge0$ are integrable. Then $x\mapsto\int h(x,y)\dd y$ and $y\mapsto\int h(x,y)\dd x$ are
continuous and integrable, and $\int\bigl(\int h(x,y)\dd y\bigr)\dd x=\int\bigl(\int h(x,y)\dd x\bigr)\dd y$.
\end{lemma}
\begin{proof}
\emph{Each slice.} On a bounded interval of $x$, the function $a$ is bounded by some $M$, so $|h|\le Mb(y)$. Theorem~\ref{thm:domination} makes $x\mapsto\int h(x,y)\dd y$ continuous. Its absolute value is at most $a(x)\|b\|_1$, so the slice is integrable. The same argument with the variables exchanged handles the other order.

\emph{The square, then the tails.} On $[-R,R]^2$ the two iterated integrals agree by Theorem~\ref{thm:fubinirect}. Each full integral differs from its value on that square by at most $\|b\|_1\int_{|x|>R}a+\|a\|_1\int_{|y|>R}b$. Both tails tend to $0$ as $R\to\infty$.
\end{proof}

\begin{lemma}[Gaussian approximate identity]\label{lem:approxid}
For $\varepsilon>0$ let $G_\varepsilon(u)=\frac1\varepsilon e^{-\pi u^2/\varepsilon^2}$, so that $G_\varepsilon>0$ and $\int G_\varepsilon=1$. If $f$ is integrable and continuous at $x$, then
\[
\int_{\R}f(x-u)G_\varepsilon(u)\dd u\xrightarrow[\varepsilon\to0]{}f(x).
\]
A jump is not in the class: the choice of $\delta$ in the proof uses continuity at $x$.
\end{lemma}
\begin{proof}
\emph{Near the point.} Given $\eta>0$, continuity at $x$ supplies $\delta>0$ with $|f(x-u)-f(x)|\le\eta$ for $|u|\le\delta$. The difference between the integral and $f(x)$ is at most
\[
\begin{aligned}\eta&+\int_{|u|>\delta}|f(x-u)|G_\varepsilon(u)\dd u\eqbr+|f(x)|\int_{|u|>\delta}G_\varepsilon(u)\dd u.\end{aligned}
\]

\emph{The tails.} For $|u|>\delta$, $G_\varepsilon(u)\le\frac1\varepsilon e^{-\pi\delta^2/\varepsilon^2}$, so the middle term is at most $\frac{\|f\|_1}\varepsilon e^{-\pi\delta^2/\varepsilon^2}$, which tends to $0$. The last integral is $\int_{|v|>\delta/\varepsilon}e^{-\pi v^2}\dd v$, and that tends to $0$ as well.

\emph{Finish.} For small $\varepsilon$ the difference is at most $2\eta$.
\end{proof}

\begin{theorem}[Fourier inversion]\label{thm:ftinversion}
Let $f$ be continuous and integrable, with $\hat f$ integrable. Then for every $x$,
\[
f(x)=\int_{\R}\hat f(\xi)e^{2\pi ix\xi}\dd\xi.
\]
In particular $f$ is bounded by $\|\hat f\|_1$, and a continuous integrable function with $\hat f=0$ is identically zero. A jump is not in the class.
\end{theorem}

Every later use of the theorem has to check the class. The class is: $f$ continuous on $\R$, $\int|f|<\infty$, and $\int|\hat f|<\infty$. The transform $\hat f$ is always continuous. A convenient sufficient condition is that $f$ be of class $C^2$ with $f$, $f'$ and $f''$ integrable. Then $|\hat f(\xi)|\le\min\bigl(\|f\|_1,\frac{\|f''\|_1}{4\pi^2\xi^2}\bigr)$, and the right-hand side is integrable. Every function to which this volume applies inversion or Poisson summation is of this kind, or is checked by hand: Gaussians times exponentials and polynomials (here, in Chapter~\ref{ch:zeta} and in Chapter~\ref{ch:hr}), the triangle function and its transform, and $e^{-2\pi a|x|}$. A function with a jump lies outside the class. Its transform decays like $\frac1{|\xi|}$, so the transform is not integrable.
\begin{proof}
\emph{A damped integral.} Fix $x$, and for $\varepsilon>0$ let
\[
I_\varepsilon=\int\hat f(\xi)e^{-\pi\varepsilon^2\xi^2}e^{2\pi ix\xi}\dd\xi.
\]
As $\varepsilon\to0$, Theorem~\ref{thm:domination} with the bound $|\hat f(\xi)|$ gives $I_\varepsilon\to\int\hat f(\xi)e^{2\pi ix\xi}\dd\xi$.

\emph{Swap the integrals.} Write $\hat f$ as an integral. The function $h(\xi,y)=f(y)e^{-2\pi iy\xi}e^{-\pi\varepsilon^2\xi^2}e^{2\pi ix\xi}$ satisfies $|h|\le|f(y)|e^{-\pi\varepsilon^2\xi^2}$, so Lemma~\ref{lem:fubiniplane} gives
\[
\begin{aligned}I_\varepsilon&=\int_{\R}f(y)\Bigl(\int_{\R}e^{-\pi\varepsilon^2\xi^2}e^{2\pi i(x-y)\xi}\dd\xi\Bigr)\dd y\eqbr=\int_{\R}f(y)G_\varepsilon(x-y)\dd y,\end{aligned}
\]
The inner integral is the transform of $\xi\mapsto e^{-\pi\varepsilon^2\xi^2}$, up to the sign in the exponential. That Gaussian is even, so the sign does not matter. Theorem~\ref{thm:gaussianft} with $a=\frac1\varepsilon$ says the integral equals $\frac1\varepsilon e^{-\pi(x-y)^2/\varepsilon^2}$, which is $G_\varepsilon(x-y)$.

\emph{Finish.} Lemma~\ref{lem:approxid} gives $I_\varepsilon\to f(x)$.
\end{proof}

\begin{theorem}[Plancherel]\label{thm:plancherel}
Let $f$ be continuous and integrable, with $\hat f$ integrable. Then $|f|^2$ and $|\hat f|^2$ are integrable, and $\int|f|^2=\int|\hat f|^2$.
\end{theorem}
\begin{proof}
\emph{Squares.} Theorem~\ref{thm:ftinversion} and Proposition~\ref{prop:ftbasic} bound $f$ and $\hat f$. Both are integrable, so their squares are integrable.

\emph{The swap.} Apply Lemma~\ref{lem:fubiniplane} to $h(\xi,x)=\overline{\hat f(\xi)}f(x)e^{-2\pi ix\xi}$, which satisfies $|h|\le|\hat f(\xi)||f(x)|$:
\[
\begin{aligned}\int|\hat f|^2&=\int\overline{\hat f(\xi)}\Bigl(\int f(x)e^{-2\pi ix\xi}\dd x\Bigr)\dd\xi\\&=\int f(x)\overline{\Bigl(\int\hat f(\xi)e^{2\pi ix\xi}\dd\xi\Bigr)}\dd x\eqbr=\int|f|^2,\end{aligned}
\]
The inner integral in the second line is $f(x)$, by inversion, and the conjugate of that integral is $\overline{f(x)}$.
\end{proof}

As on the circle, Plancherel says the energy of a signal equals the energy of its spectrum. Volume~8 extends the transform to every square-integrable function, and the extension is unitary.

\section{The heat equation on the line}

\begin{proposition}\label{prop:heatline}
Let $f$ be integrable and bounded, and for $t>0$ let $H_t(x)=\frac1{\sqrt{4\pi t}}e^{-x^2/(4t)}$ and $u(t,x)=\int_{\R}f(y)H_t(x-y)\dd y$. Then $\partial_tu=\partial_x^2u$ for
$t>0$, $|u|\le\sup|f|$, and $u(t,x)\to f(x)$ as $t\to0^+$.
\end{proposition}
\begin{proof}
\emph{The kernel.} A direct computation gives $\partial_xH_t=-\frac x{2t}H_t$ and $\partial_tH_t=\partial_x^2H_t=\bigl(\frac{x^2}{4t^2}-\frac1{2t}\bigr)H_t$.

\emph{Under the integral.} For $t$ in a compact interval of $\left]0,\infty\right[$ these derivatives are bounded functions of $(t,x)$. Theorem~\ref{thm:domination}, with majorant a constant times $|f(y)|$, lets the derivatives pass under the integral. So $u$ satisfies the heat equation.

\emph{The limit.} Since $H_t>0$ and $\int H_t=1$, $|u|\le\sup|f|$. Finally $H_t=G_\varepsilon$ with $\varepsilon=\sqrt{4\pi t}$, because $\pi/\varepsilon^2=1/(4t)$. Lemma~\ref{lem:approxid} gives the limit.
\end{proof}

Theorem~\ref{thm:gaussianft} gives $\widehat{H_t}(\xi)=e^{-4\pi^2t\xi^2}$. On the Fourier side, heat multiplies the frequency $\xi$ by $e^{-4\pi^2t\xi^2}$. Large frequencies die first, as they do on the circle (Exercise~\ref{ex:heat}). The same kernel $H_t$ is the density of the normal law of variance $2t$. Heat spreads like a Brownian motion (Volume~12).

\section{The Poisson summation formula}

\portrait{poisson}{Sim\'eon Denis Poisson (1781--1840). Lithograph by Fran\c cois-S\'eraphin Delpech, after Nicolas Eustache Maurin. Public domain.}

Poisson stated the Gaussian case in 1823, in the continuation of his memoir on definite integrals (\emph{Journal de l'\'Ecole Polytechnique}, vol.~12, pp.~404--509; the identity is on p.~420). The argument below is the one this chapter uses. Periodize $f$, read the Fourier coefficients of that periodic function, and recognise them as values of $\hat f$.

\begin{theorem}[Poisson summation]\label{thm:poisson}
Let $f$ be continuous with $|f(x)|\le\frac C{(1+|x|)^2}$ and $|\hat f(\xi)|\le\frac C{(1+|\xi|)^2}$ for some constant $C$. Then for every real $x$,
\[
\sum_{n\in\Z}f(x+n)=\sum_{m\in\Z}\hat f(m)e^{2\pi imx},
\]
both series converging absolutely and uniformly. In particular $\sum_nf(n)=\sum_m\hat f(m)$.
\end{theorem}
\begin{proof}
\emph{The periodization.} On $[0,1]$, $|f(x+n)|\le\frac C{\max(1,|n|-1)^2}$, so $F(x)=\sum_nf(x+n)$ converges uniformly on $[0,1]$, and then on $\R$ by periodicity. The sum $F$ is continuous and $1$-periodic.

\emph{The coefficients.} They are $c_m(F)=\int_0^1F(x)e^{-2\pi imx}\dd x$. Integrate term by term:
\[
\begin{aligned}c_m(F)&=\sum_n\int_0^1f(x+n)e^{-2\pi imx}\dd x\\&=\sum_n\int_n^{n+1}f(y)e^{-2\pi imy}\dd y=\hat f(m).\end{aligned}
\]

\emph{Finish.} The series $P(x)=\sum_m\hat f(m)e^{2\pi imx}$ converges uniformly, because $\sum|\hat f(m)|<\infty$, and it has the same coefficients. So $F-P$ is continuous and $1$-periodic, with every Fourier coefficient equal to $0$. Corollary~\ref{cor:fejer}(b), applied to $x\mapsto(F-P)(\frac x{2\pi})$, gives $F=P$.
\end{proof}

\begin{corollary}[theta transformation]\label{cor:theta}
For $t>0$ let $\vartheta(t)=\sum_{n\in\Z}e^{-\pi n^2t}$. The series keeps the letter $\vartheta$ for this sum, and leaves $\theta$ free for angles. Then
\[
\vartheta\Bigl(\frac1t\Bigr)=\sqrt t\,\vartheta(t).
\]
\end{corollary}
\begin{proof}
Apply Theorem~\ref{thm:poisson} at $x=0$ to $f(x)=e^{-\pi tx^2}$. Theorem~\ref{thm:gaussianft} with $a=t^{-1/2}$ says the transform is $t^{-1/2}e^{-\pi\xi^2/t}$. Both functions decay faster than any power. So $\vartheta(t)=t^{-1/2}\vartheta(1/t)$.
\end{proof}

Figure~\ref{fig:rw08-comb} is the same identity at $t=\frac12$. The wide curve is $e^{-\pi x^2/2}$. Its transform is the narrow spike $\sqrt2\,e^{-2\pi\xi^2}$. The black stems are the samples at the integers. Both combs add to $1.4195$.

This is the sum from the opening page. For small $t$ the series for $\vartheta(t)$ is slow: about $\frac1{\sqrt t}$ terms matter. The series for $\vartheta(\frac1t)$ is fast. Corollary~\ref{cor:theta} says $\vartheta(0.01)=10\,\vartheta(100)$, and $\vartheta(100)=1+2e^{-100\pi}+\cdots$ differs from $1$ by about $10^{-136}$. Riemann used the identity in 1859 to extend $\zeta$ to the whole plane. Chapter~\ref{ch:zeta} follows him. Jacobi's $\vartheta$ also counts sums of squares (Volume~6) and describes heat on a circle.

The same formula gives a second proof of Euler's partial fractions for the cotangent (Theorem~\ref{thm:cot}).

\begin{proposition}\label{prop:sinsquare}
For every real $x\notin\Z$, $\displaystyle\sum_{n\in\Z}\frac1{(x+n)^2}=\frac{\pi^2}{\sin^2\pi x}$.
\end{proposition}
\begin{proof}
\emph{The triangle.} Let $T(t)=\max(0,1-|t|)$. It is continuous and vanishes outside $[-1,1]$, and
\[
\begin{aligned}\widehat T(\xi)&=2\int_0^1(1-t)\cos2\pi t\xi\dd t\eqbr=\frac{1-\cos2\pi\xi}{2\pi^2\xi^2}=\Bigl(\frac{\sin\pi\xi}{\pi\xi}\Bigr)^2\end{aligned}
\]
for $\xi\ne0$, while $\widehat T(0)=1$. The integration by parts that produces the middle expression is the same one as for a single cosine: $\int_0^1(1-t)\cos2\pi t\xi\dd t=\frac{1-\cos2\pi\xi}{4\pi^2\xi^2}$.

\emph{Poisson.} So $\widehat T$ is integrable, with $|\widehat T(\xi)|\le\frac C{(1+|\xi|)^2}$. Inversion says the transform of $\widehat T$ is $T(-t)=T(t)$. Theorem~\ref{thm:poisson} applied to $\widehat T$ gives
\[
\begin{aligned}\sum_{n\in\Z}\widehat T(x+n)&=\sum_{m\in\Z}T(m)e^{2\pi imx}\eqbr=T(0)=1.\end{aligned}
\]

\emph{Finish.} Since $\sin^2\pi(x+n)=\sin^2\pi x$ for every integer $n$, the left side equals $\frac{\sin^2\pi x}{\pi^2}$ times $\sum_n\frac1{(x+n)^2}$.
\end{proof}

\begin{corollary}[Theorem~\ref{thm:cot} again]
For $0<|x|<1$, $\pi\cot\pi x=\frac1x+\sum_{n\ge1}\frac{2x}{x^2-n^2}$.
\end{corollary}
\begin{proof}
\emph{A difference.} Let $h(x)=\pi\cot\pi x-\frac1x-\sum_{n\ge1}\frac{2x}{x^2-n^2}$ on $\left]-1,1\right[$, with the value $h(0)=0$. The function $h$ is odd. It is continuous, because $\pi\cot\pi x-\frac1x\to0$ at $0$.

\emph{The derivative.} Since $\frac{2x}{x^2-n^2}=\frac1{x-n}+\frac1{x+n}$, the derivative of the general term is $-\frac1{(x-n)^2}-\frac1{(x+n)^2}$. On $[-a,a]$ with $a<1$ that is bounded by $\frac2{(n-a)^2}$. Lemma~\ref{lem:uniform} lets us differentiate the series term by term. For $x\ne0$,
\[
h'(x)=-\frac{\pi^2}{\sin^2\pi x}+\frac1{x^2}+\sum_{n\ne0}\frac1{(x+n)^2}=0
\]
by Proposition~\ref{prop:sinsquare}.

\emph{Finish.} So $h$ is constant on $\left]0,1\right[$ and on $\left]-1,0\right[$. Continuity at $0$ forces the constant to be $h(0)=0$.
\end{proof}

Both sides of Theorem~\ref{thm:cot} are $1$-periodic (Chapter~\ref{ch:series}), so the identity holds for every $x\notin\Z$.

\section{The sign of the Gauss sum}

For $n\ge1$ let $G(n)=\sum_{k=0}^{n-1}e^{2\pi ik^2/n}$. For an odd prime $p$, this $G(p)$ is the Gauss sum $g$ of Proposition~\ref{prop:gaussfourier}, so $|G(p)|^2=p$. Volume~2, Exercise~6.7, gives $G(p)^2=\bigl(\frac{-1}p\bigr)p$. Hence $G(p)=\pm\sqrt p$ if $p\equiv1\pmod4$, and $G(p)=\pm i\sqrt p$ if $p\equiv3\pmod4$. The square fixes the value up to a sign.

\begin{think}
Why does no algebraic manipulation decide the sign? Replace $e^{2\pi i/p}$ by $e^{2\pi ia/p}$, with $a$ a non-residue modulo $p$. Every algebraic identity among the powers of $e^{2\pi i/p}$ survives, and $G(p)$ becomes $\sum_ke^{2\pi iak^2/p}$. What is this new sum?
\end{think}

Gauss wrote the sign down in 1801 and proved it only in 1805. The words he used for that evening, in a letter to Olbers of 3 September 1805, are the ones quoted in the Sources: the riddle solved like a flash of lightning. The proof below is Dirichlet's, from 1835. Its only analytic input is the value of one integral.

\begin{lemma}[a Fresnel integral]\label{lem:fresnel}
The improper integral $\int_0^\infty u^{-1/2}e^{iu}\dd u$ converges and equals $\sqrt\pi\,e^{i\pi/4}$. Consequently, for $n\ge1$,
\[
\int_{-\infty}^\infty e^{2\pi iy^2/n}\dd y=\frac{1+i}2\sqrt n.
\]
\end{lemma}
\begin{proof}
\emph{Convergence.} For $t\ge0$ and $0<A<B$, the function $u^{-1/2}e^{-tu}$ is positive and decreasing. The proof of Lemma~\ref{lem:secondmean}, with $e^{iu}$ in place of $\sin u$, gives $\bigl|\int_A^Bu^{-1/2}e^{-tu}e^{iu}\dd u\bigr|\le2A^{-1/2}$, uniformly in $t\ge0$. At $t=0$ this is the Cauchy criterion, so the integral converges. Call $J_t$ the same integral with the factor $e^{-tu}$. As in the proof of Theorem~\ref{thm:dirichletint}, for $A>0$,
\[
\begin{aligned}|J_t-J_0|&\le\int_0^Au^{-1/2}(1-e^{-tu})\dd u+4A^{-1/2}\eqbr\le\frac23tA^{3/2}+4A^{-1/2},\end{aligned}
\]
so $J_t\to J_0$ as $t\to0^+$. The bound $\int_0^A u^{-1/2}(1-e^{-tu})\dd u\le t\int_0^A u^{1/2}\dd u=\frac23 t A^{3/2}$ uses $1-e^{-tu}\le tu$.

\emph{A square root.} Now fix $t>0$ and, for real $b$, let $z=t-ib$ and $K(b)=\int_0^\infty u^{-1/2}e^{-zu}\dd u$. Theorem~\ref{thm:domination} gives $K'(b)=i\int_0^\infty u^{1/2}e^{-zu}\dd u$. An integration by parts gives $\int_0^\infty u^{1/2}e^{-zu}\dd u=\frac1{2z}K(b)$. Let $s(b)$ be the square root of $z$ with positive real part. It is differentiable, and $s^2=z$ gives $2ss'=-i$. Then
\[
(sK)'=s'K+sK'=-\frac i{2s}K+\frac{is}{2z}K=0,
\]
since $z=s^2$. So $s(b)K(b)=s(0)K(0)=\sqrt t\cdot\Gamma\bigl(\frac12\bigr)t^{-1/2}=\sqrt\pi$, and $J_t=K(1)=\frac{\sqrt\pi}{s(1)}$. As $t\to0$, $s(1)=\sqrt{t-i}$ tends to the square root of $-i$ with positive real part, which is $e^{-i\pi/4}$. Hence $J_0=\sqrt\pi\,e^{i\pi/4}$.

\emph{The substitution.} The full-line integral is twice the integral over $\left]0,\infty\right[$. Set $u=\frac{2\pi y^2}n$, so $y=\sqrt{nu/(2\pi)}$ and $\dd y=\sqrt{n/(2\pi)}\,\frac12 u^{-1/2}\dd u$. The factor in front of $J_0$ is $\sqrt{n/(2\pi)}$. Thus the integral equals $\sqrt{\frac n{2\pi}}\,J_0=\sqrt{\frac n2}\,e^{i\pi/4}$. Since $e^{i\pi/4}=\frac{1+i}{\sqrt2}$, this is $\frac{1+i}2\sqrt n$.
\end{proof}

\begin{theorem}[Gauss, 1805]\label{thm:gausssign}
For every $n\ge1$,
\[
G(n)=\frac{(1+i)(1+i^{-n})}2\sqrt n,
\]
that is,
\[
G(n)=\begin{cases}\sqrt n&\text{if }n\equiv1\pmod4,\\0&\text{if }n\equiv2\pmod4,\\i\sqrt n&\text{if }n\equiv3\pmod4,\\(1+i)\sqrt n&\text{if }n\equiv0\pmod4.\end{cases}
\]
In particular $G(p)=\sqrt p$ for primes $p\equiv1\pmod4$ and $G(p)=i\sqrt p$ for primes $p\equiv3\pmod4$.
\end{theorem}
\begin{proof}
\emph{A periodic sum.} Let $f(x)=e^{2\pi ix^2/n}$ and let $F$ be the $1$-periodic function with $F(x)=\sum_{k=0}^{n-1}f(x+k)$ for $0\le x<1$. It is piecewise $C^1$. Since $f(n)=f(0)=1$, the limit of $F$ from the left at $0$ is $\sum_{k=1}^nf(k)$, which equals $F(0)=G(n)$. So $F$ is continuous at $0$. Theorem~\ref{thm:dirichletfourier}, rescaled to period $1$, gives $G(n)=\lim_N\sum_{|j|\le N}c_j$, where
\[
c_j=\int_0^1F(x)e^{-2\pi ijx}\dd x=\int_0^ne^{2\pi i(x^2/n-jx)}\dd x.
\]

\emph{Completing the square.} One has $\frac{x^2}n-jx=\frac{(x-jn/2)^2}n-\frac{j^2n}4$. If $j$ is even, $e^{-2\pi ij^2n/4}=1$. If $j$ is odd, then $j^2\equiv1\pmod8$ and $e^{-2\pi ij^2n/4}=e^{-i\pi n/2}=i^{-n}$. Write $\Phi(a,b)=\int_a^be^{2\pi iy^2/n}\dd y$, and set $\varepsilon_j=1$ for $j$ even, $\varepsilon_j=i^{-n}$ for $j$ odd. Then
\[
c_j=\varepsilon_j\,\Phi\Bigl(-\frac{jn}2,n-\frac{jn}2\Bigr),
\]

\emph{The limit.} Take $N=2L+1$. The intervals $[-\frac{jn}2,n-\frac{jn}2]$ for even $j$ with $|j|\le N$ are $[-ln,n-ln]$, $|l|\le L$, and they tile $[-Ln,(L+1)n]$. For odd $j$ with $|j|\le N$ they tile $[-Ln-\frac n2,(L+1)n+\frac n2]$. So
\[
\begin{aligned}\sum_{|j|\le N}c_j&=\Phi\bigl(-Ln,(L+1)n\bigr)\eqbr+i^{-n}\,\Phi\Bigl(-Ln-\frac n2,(L+1)n+\frac n2\Bigr),\end{aligned}
\]
and as $L\to\infty$ both integrals tend to $\int_{\R}e^{2\pi iy^2/n}\dd y=\frac{1+i}2\sqrt n$ by Lemma~\ref{lem:fresnel}. Hence $G(n)=(1+i^{-n})\frac{1+i}2\sqrt n$. The four cases are the four values $i^{-n}=-i,-1,i,1$, in the order $n\equiv1,2,3,0\pmod4$.
\end{proof}

\begin{beauty}
A finite sum of roots of unity, an arithmetic object, comes out of an integral on the real line. The closed form is a piece of order. The sign is the other half: the think box shows that a non-residue $a$ turns $G(p)$ into $-G(p)$, and no identity among roots of unity can see which sign is which. The integral $\int e^{iu}u^{-1/2}\dd u$ uses the ordering of the reals, and that ordering fixes the sign. Exercise~\ref{ex:reciprocity} turns the same evaluation into one more proof of quadratic reciprocity.
\end{beauty}

\section{Sampling and uncertainty}

\begin{think}
Hearing stops near $20$ kHz, and a compact disc stores $44\,100$ samples each second. Why that count, rather than $20\,000$?
\end{think}

A signal is \emph{band-limited} to $W$ when $\hat f(\xi)=0$ for $|\xi|>W$. Write $\operatorname{sinc}(t)=\frac{\sin\pi t}{\pi t}$, with $\operatorname{sinc}(0)=1$. The function $\operatorname{sinc}$ vanishes at every non-zero integer.

\begin{theorem}[Shannon--Whittaker--Kotelnikov]\label{thm:shannon}
Let $f$ be integrable and band-limited to $W$. Then for every $x$,
\[
f(x)=\sum_{n\in\Z}f\Bigl(\frac n{2W}\Bigr)\operatorname{sinc}(2Wx-n),
\]
the symmetric partial sums converging. So $f$ is determined by its samples at the rate $2W$ per unit length.
\end{theorem}
\begin{proof}
\emph{Inversion on a finite interval.} The transform $\hat f$ is continuous and supported in $[-W,W]$, so it is integrable. Inversion gives $f(x)=\int_{-W}^W\hat f(\xi)e^{2\pi ix\xi}\dd\xi$.

\emph{Parseval.} Extend $\varphi=\hat f$ and $\psi(\xi)=e^{-2\pi ix\xi}$ from $[-W,W[$ to $2W$-periodic piecewise continuous functions. Apply the general form of Parseval's identity, Theorem~\ref{thm:parseval}, rescaled to period $2W$, with coefficients $c_n(\cdot)=\frac1{2W}\int_{-W}^W(\cdot)\,e^{-2\pi in\xi/(2W)}\dd\xi$. Inversion gives $c_n(\varphi)=\frac1{2W}f\bigl(-\frac n{2W}\bigr)$. A direct integration gives $c_n(\psi)=\operatorname{sinc}(2Wx+n)$, which is real. So
\[
\begin{aligned}\frac1{2W}f(x)&=\frac1{2W}\int_{-W}^W\varphi\bar\psi=\sum_nc_n(\varphi)\overline{c_n(\psi)}\\&=\frac1{2W}\sum_nf\Bigl(-\frac n{2W}\Bigr)\operatorname{sinc}(2Wx+n),\end{aligned}
\]
Replacing $n$ by $-n$ puts the samples in the order of the statement.
\end{proof}

The rate $2W$ is the \emph{Nyquist rate}. Sampling more slowly loses information. The frequencies $\xi$ and $\xi\pm2W$ produce the same samples, so they cannot be told apart. That is the \emph{aliasing} of Proposition~\ref{prop:aliasing} (Figure~\ref{fig:alias}). This answers the think box. A band of $20$ kHz needs at least $40\,000$ samples each second. The extra $10\%$ leaves room for the analogue filter that removes the frequencies above $20$ kHz before the samples are taken. No real filter cuts off at a single frequency.

\begin{theorem}[Heisenberg's inequality]\label{thm:heisenberg}
Let $f$ be of class $C^3$, with $f^{(k)}(x)=O\bigl((1+|x|)^{-3}\bigr)$ for $0\le k\le3$. Then
\[
\begin{aligned}\Bigl(\int_{\R}x^2|f(x)|^2\dd x\Bigr)&\Bigl(\int_{\R}\xi^2|\hat f(\xi)|^2\dd\xi\Bigr)\eqbr\ge\frac1{16\pi^2}\Bigl(\int_{\R}|f|^2\Bigr)^2,\end{aligned}
\]
with equality for $f(x)=e^{-\pi x^2}$.
\end{theorem}
\begin{proof}
\emph{An integration by parts.} The decay hypothesis makes every integral below converge, and $|\hat f(\xi)|\le\frac{\|f'''\|_1}{(2\pi|\xi|)^3}$, so $\xi\hat f(\xi)$ is integrable. The boundary terms $x|f(x)|^2$ tend to $0$, and
\[
\begin{aligned}\int|f|^2&=-\int x\bigl(|f|^2\bigr)'\dd x\eqbr=-2\operatorname{Re}\int x\bar f(x)f'(x)\dd x\\&\le2\Bigl(\int x^2|f|^2\Bigr)^{1/2}\Bigl(\int|f'|^2\Bigr)^{1/2}\end{aligned}
\]
by Cauchy--Schwarz.

\emph{Plancherel on the derivative.} Proposition~\ref{prop:ftbasic}(c) gives $\widehat{f'}=2\pi i\xi\hat f$, which is integrable, so Plancherel applies to $f'$: $\int|f'|^2=4\pi^2\int\xi^2|\hat f|^2$. Squaring the previous estimate gives the inequality.

\emph{Equality.} For $f=e^{-\pi x^2}=\hat f$, both factors on the left equal $\int x^2e^{-2\pi x^2}\dd x=\frac1{4\pi\sqrt2}$, and $\int|f|^2=\frac1{\sqrt2}$. The two sides match.
\end{proof}

The moments in the theorem are taken about the origin. They are variances when both means vanish. The centred form is the same inequality after a shift. Suppose $\int|f|^2=1$, and set $m=\int x|f(x)|^2\dd x$ and $\nu=\int\xi|\hat f(\xi)|^2\dd\xi$. Both integrals converge by the decay. Define $g(x)=e^{-2\pi i\nu x}f(x+m)$. Then $|g(x)|=|f(x+m)|$, so $\int x^2|g|^2=\int(y-m)^2|f(y)|^2\dd y$. The transform of $x\mapsto f(x+m)$ is $e^{2\pi im\xi}\hat f(\xi)$, and the factor $e^{-2\pi i\nu x}$ shifts the frequency by $\nu$. Hence $|\hat g(\xi)|=|\hat f(\xi+\nu)|$ and $\int\xi^2|\hat g|^2=\int(\eta-\nu)^2|\hat f(\eta)|^2\dd\eta$. The function $g$ meets the hypotheses, and $\int|g|^2=1$. Plancherel gives $\int|\hat f|^2=1$, so $|\hat f|^2$ is a density as well. The inequality for $g$ is the same bound on the variances of $|f|^2$ and of $|\hat f|^2$. In quantum mechanics, where position and momentum are a Fourier pair, this centred form is the uncertainty principle. In signal processing, a short pulse has a broad spectrum.

\section{What we proved, and what comes next}

On the line, the Fourier transform turns a derivative into multiplication (Proposition~\ref{prop:ftbasic}), fixes the Gaussian (Theorem~\ref{thm:gaussianft}), undoes itself (Theorem~\ref{thm:ftinversion}) and preserves energy (Theorem~\ref{thm:plancherel}). Poisson summation (Theorem~\ref{thm:poisson}) joins the line to the circle: the sum of $f$ over the integers equals the sum of $\hat f$ over the integers.

That identity answers the opening sum. For $f(x)=e^{-\pi x^2/100}$ it says $\vartheta(0.01)=10\,\vartheta(100)$ (Corollary~\ref{cor:theta}). The $10$ is the value $\hat f(0)=\int f$. The error is the next term, $20e^{-100\pi}\approx7\cdot10^{-136}$. The same identity gave the cotangent series a second proof, and Dirichlet's argument gave the sign of the Gauss sum (Theorem~\ref{thm:gausssign}). Two facts about information closed the chapter. Samples at rate $2W$ determine a signal of band $W$ (Theorem~\ref{thm:shannon}). A function and its transform cannot both be narrow (Theorem~\ref{thm:heisenberg}).

Three tools leave for later chapters. The relation for $\vartheta$ goes to Chapter~\ref{ch:zeta}, where it gives the functional equation of $\zeta$. Poisson summation with a shifted Gaussian goes to Chapter~\ref{ch:hr}, where it gives the transformation of Dedekind's $\eta$. The next chapter asks a different question about the same integrals: $\int_{-\infty}^{\infty}\frac{\cos x}{1+x^2}\dd x$ equals $\frac\pi e$, and the factor $e$ is not visible on the real line.

\end{enumerate}
\end{exercise}

\begin{solution}{ex:lorentz}
\idea The function is even, so the transform is a cosine integral. The integral of an exponential against a cosine is elementary.
\writeup Let $F_a$ be the transform of $e^{-2\pi a|x|}$. The hint is the real part of $\int_0^\infty e^{-(c-i\omega)x}\dd x=\frac1{c-i\omega}$:
\[
\begin{aligned}F_a(\xi)&=2\int_0^\infty e^{-2\pi ax}\cos(2\pi x\xi)\dd x\eqbr=\frac{2\cdot2\pi a}{(2\pi a)^2+(2\pi\xi)^2}=\frac a{\pi(a^2+\xi^2)}.\end{aligned}
\]
With $a=1$, both $f=e^{-2\pi|x|}$ and $\hat f=\frac1{\pi(1+\xi^2)}$ are integrable. Theorem~\ref{thm:ftinversion} gives $e^{-2\pi|x|}=\int\frac{e^{2\pi ix\xi}}{\pi(1+\xi^2)}\dd\xi$. Replace $x$ by $-x$ and swap the names of the variables. The transform of $\frac1{1+x^2}$ is $\pi e^{-2\pi|\xi|}$. Chapter~\ref{ch:complex} computes the same integral by residues.
\end{solution}

\begin{solution}{ex:hermite}
\idea Multiplication by $x$ becomes differentiation on the transform side, and the Gaussian is a fixed point of the transform.
\writeup By Proposition~\ref{prop:ftbasic}(d), the transform of $-2\pi ixe^{-\pi x^2}$ is $\frac{\mathrm d}{\mathrm d\xi}e^{-\pi\xi^2}=-2\pi\xi e^{-\pi\xi^2}$. Divide by $-2\pi i$. The transform of $xe^{-\pi x^2}$ is $\frac{\xi}ie^{-\pi\xi^2}=-i\xi e^{-\pi\xi^2}$. So this function is sent to $-i$ times itself. That is an eigenvalue remark, not a theorem of this chapter. Inversion gives $\mathcal F^2f(x)=f(-x)$, hence $\mathcal F^4=\mathrm{id}$, so any eigenvalue that does occur lies in $\{1,-i,-1,i\}$. The Hermite functions $H_k(x)e^{-\pi x^2}$ are said to take these four values according to $k$ modulo $4$. We do not prove that. The cases $k=0$ and $k=1$ are Theorem~\ref{thm:gaussianft} and this exercise.
\end{solution}

\begin{solution}{ex:coth}
\idea Poisson's formula trades the samples of $\frac1{n^2+a^2}$ for the samples of its transform. Those samples form a two-sided geometric series.
\writeup Let $f(x)=e^{-2\pi a|x|}$. It is continuous and decays exponentially, and $\hat f(\xi)=\frac a{\pi(a^2+\xi^2)}=O(\xi^{-2})$. Theorem~\ref{thm:poisson} at $x=0$ gives
\[
\begin{aligned}\sum_m\frac a{\pi(a^2+m^2)}&=\sum_ne^{-2\pi a|n|}=1+\frac{2e^{-2\pi a}}{1-e^{-2\pi a}}\eqbr=\frac{1+e^{-2\pi a}}{1-e^{-2\pi a}}=\coth\pi a.\end{aligned}
\]
Multiply by $\frac\pi a$. The term $m=0$ is $\frac1{a^2}$, so $\sum_{m\ge1}\frac1{m^2+a^2}=\frac\pi{2a}\coth\pi a-\frac1{2a^2}$. As $a\to0$ the left side tends to $\zeta(2)$: the convergence is uniform because each term is at most $\frac1{m^2}$. With $\coth y=\frac1y+\frac y3+O(y^3)$ the right side is $\frac\pi{2a}\bigl(\frac1{\pi a}+\frac{\pi a}3+O(a^3)\bigr)-\frac1{2a^2}=\frac{\pi^2}6+O(a^2)$.
\end{solution}

\begin{solution}{ex:periodize}
\idea Rescale the variable so that the period is $1$, and apply the formula already proved.
\writeup The function $g(x)=f(Lx)$ satisfies the hypotheses, and $\hat g(\xi)=\frac1L\hat f(\frac\xi L)$ by Proposition~\ref{prop:ftbasic}(b). Poisson for $g$ at the point $\frac xL$ gives $\sum_nf(x+nL)=\frac1L\sum_m\hat f(\frac mL)e^{2\pi imx/L}$. The left side is $L$-periodic. The right side is its Fourier series: the frequencies $\frac mL$ fill the line more densely as $L$ grows. For fixed $x$ and large $L$, the left side is $f(x)$ plus the terms $f(x+nL)$ with $n\ne0$. Each of those is $O(L^{-2})$, and the whole tail is $O(L^{-1})$. The right side is a Riemann sum for $\int\hat f(\xi)e^{2\pi ix\xi}\dd\xi$. The limit is the inversion formula. A Fourier integral is the Fourier series of a function whose period has been sent to infinity.
\end{solution}

\begin{solution}{ex:reciprocity}
\idea Compute $G(pq)$ in two ways. The Chinese remainder theorem splits it into a product of two twisted Gauss sums, each a Legendre symbol times $G(p)$ or $G(q)$. Theorem~\ref{thm:gausssign} evaluates all three sums.
\writeup (a) Let $\chi=\bigl(\frac\cdot p\bigr)$ and $\omega=e^{2\pi i/p}$. The number of $k$ with $k^2\equiv m$ is $1+\chi(m)$, so $G_a(p)=\sum_m(1+\chi(m))\omega^{am}=\sum_m\chi(m)\omega^{am}$. The sum of $\omega^{am}$ over all $m$ vanishes, which is why the extra $1$ drops out. Substitute $m=a^{-1}m'$. Then $\chi(a^{-1})g=\chi(a)G(p)$, as in the proof of Proposition~\ref{prop:gaussfourier}.

(b) As $a$ runs modulo $p$ and $b$ modulo $q$, the combination $k=aq+bp$ runs once through the residues modulo $pq$, by the Chinese remainder theorem. Since $k^2\equiv a^2q^2+b^2p^2$ modulo $pq$, $e^{2\pi ik^2/(pq)}=e^{2\pi iqa^2/p}e^{2\pi ipb^2/q}$. Summing separates the product: $G(pq)=G_q(p)G_p(q)$.

(c) Parts (a) and (b) give $G(pq)=\bigl(\frac qp\bigr)\bigl(\frac pq\bigr)G(p)G(q)$. For odd $m$, Theorem~\ref{thm:gausssign} gives $G(m)=\varepsilon_m\sqrt m$ with $\varepsilon_m=1$ if $m\equiv1\pmod4$ and $\varepsilon_m=i$ if $m\equiv3\pmod4$. So $\bigl(\frac pq\bigr)\bigl(\frac qp\bigr)=\frac{\varepsilon_{pq}}{\varepsilon_p\varepsilon_q}$. If $p$ or $q$ is $\equiv1\pmod4$, then $\varepsilon_{pq}=\varepsilon_p\varepsilon_q$: one factor is $1$, and $pq$ has the residue of the other, so the ratio is $1$. If both are $\equiv3\pmod4$, then $pq\equiv1\pmod4$ and the ratio is $\frac1{i\cdot i}=-1$. This is $(-1)^{\frac{p-1}2\frac{q-1}2}$.
\end{solution}

\begin{solution}{ex:rw08c}
\idea Both integrals are the Gaussian integral after a change of scale. Differentiating in the scale produces the factor $x^2$.
\writeup Set $u=x\sqrt2$ in $\int e^{-2\pi x^2}\dd x$. Then $\dd x=\dd u/\sqrt2$ and the integral equals $2^{-1/2}\int e^{-\pi u^2}\dd u=\frac1{\sqrt2}$. \begin{solution}{ex:rw08d}
\idea Reduce the squares modulo $n$ before summing the roots of unity. For $n=5$ the real part is a known cosine.
\writeup For $n=4$ the exponents $k^2/4$ modulo $1$ are $0,\frac14,0,\frac14$, so $G(4)=1+i+1+i=2+2i$. Theorem~\ref{thm:gausssign} gives $(1+i)\sqrt4=2+2i$. The real and imaginary parts are both $2$. For $n=5$ the exponents $k^2/5$ modulo $1$ are $0,\frac15,\frac45,\frac45,\frac15$, so $G(5)=1+2e^{2\pi i/5}+2e^{-2\pi i/5}=1+4\cos\frac{2\pi}5$. Volume~1, Chapter~5, gives $\cos\frac{2\pi}5=\frac{\sqrt5-1}4$, hence $G(5)=\sqrt5$. The same theorem gives $\sqrt5$ because $5\equiv1\pmod4$. Numerically $\sqrt5=2.236067977$.
\end{solution}
