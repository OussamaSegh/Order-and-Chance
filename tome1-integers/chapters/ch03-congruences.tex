\chapter{Congruences}

\begin{center}\itshape
What are the last two digits of $3^{3^{3^3}}$, a number with more than $3\times10^{12}$ digits?\\
Arithmetic modulo $n$ answers in four lines.
\end{center}

When it is $10$ o'clock and you wait $5$ hours, it is $3$ o'clock: a clock forgets everything but the remainder. In 1801 a twenty-four-year-old Gauss
opened his \emph{Disquisitiones Arithmeticae} by turning that forgetfulness into a calculus, with a new sign, $\equiv$. It was one of the great
notational inventions of mathematics, because it lets you compute with remainders exactly as with numbers, and throw away everything you do not need.
The last two digits of a number with three trillion digits become a computation you can do on the back of an envelope.

\section{The ring \texorpdfstring{$\Z/n\Z$}{Z/nZ}}

For $n\ge1$, $a\equiv b\pmod n$ means $n\mid a-b$. This is an equivalence relation whose classes are $\bar a=a+n\Z$, and there are exactly $n$ of them, $\bar0,\dots,\overline{n-1}$ (Euclidean division gives each $a$ a unique remainder).

\begin{proposition}\label{prop:ringZn}
If $a\equiv a'$ and $b\equiv b'\pmod n$, then $a+b\equiv a'+b'$ and $ab\equiv a'b'\pmod n$. Hence $\bar a+\bar b=\overline{a+b}$ and $\bar a\bar b=\overline{ab}$ are well defined and make $\Z/n\Z$ a commutative ring with $n$ elements.
\end{proposition}
\begin{proof}
$(a+b)-(a'+b')=(a-a')+(b-b')$ and $ab-a'b'=a(b-b')+b'(a-a')$ are multiples of $n$. The ring axioms hold in $\Z/n\Z$ because they hold in $\Z$ and $a\mapsto\bar a$ respects both operations.
\end{proof}

\begin{proposition}\label{prop:units}
$\bar a$ is invertible in $\Z/n\Z$ if and only if $\gcd(a,n)=1$. Hence $\Z/n\Z$ is a field if and only if $n$ is prime, and the group of units $(\Z/n\Z)^\times$ has $\varphi(n)$ elements.
\end{proposition}
\begin{proof}
$\bar a\bar u=\bar1$ means $au+nv=1$ for some $v$, which is B\'ezout's characterisation of $\gcd(a,n)=1$. The inverse is computed by the extended Euclidean algorithm. So the units are the classes of the integers in $\{1,\dots,n\}$ prime to $n$, which is the definition of $\varphi(n)$. Finally $\Z/n\Z$ is a field iff every $a\in\{1,\dots,n-1\}$ is prime to $n$ (and $n\ge2$), iff $n$ is prime.
\end{proof}

\begin{theorem}[Euler, Fermat]\label{thm:eulerthm}
If $\gcd(a,n)=1$ then $a^{\varphi(n)}\equiv1\pmod n$. In particular, for $p$ prime and $p\nmid a$, $a^{p-1}\equiv1\pmod p$.
\end{theorem}
\begin{proof}
Multiplication by $\bar a$ is a bijection of the finite group $(\Z/n\Z)^\times$ (its inverse is multiplication by $\bar a^{-1}$). So the product of all units $x$ equals the product of all $\bar ax$, which is $\bar a^{\varphi(n)}$ times the same product. Cancel that unit.
\end{proof}

\begin{think}
Compute $(p-1)!\bmod p$ for $p=5,7,11$. What do you notice? In $\Z/7\Z$, pair each of $1,\dots,6$ with its inverse: which numbers are left alone, and why can there be no others?
\end{think}

\begin{theorem}[Wilson]\label{thm:wilson}
For $p$ prime, $(p-1)!\equiv-1\pmod p$.
\end{theorem}
\begin{proof}
In the field $\Z/p\Z$, $x^2=1$ means $(x-1)(x+1)=0$, so $x=\pm1$: only $1$ and $-1$ are their own inverses. The other non-zero classes pair off with their inverses, each pair having product $1$. So $(p-1)!\equiv1\cdot(-1)=-1$.
\end{proof}

\section{The Chinese remainder theorem}

\begin{think}
Draw a grid with $3$ rows and $4$ columns, and write each $x=0,1,\dots,11$ in row $x\bmod3$ and column $x\bmod4$. Does every cell receive exactly one number? Now try $2$ rows and $4$ columns with $x=0,\dots,7$. What property of the pair of moduli makes the difference?
\end{think}

\begin{theorem}[Chinese remainder theorem]\label{thm:crt}
If $m,n\ge1$ are coprime, the map $\Z/mn\Z\to\Z/m\Z\times\Z/n\Z$, $x\mapsto(x\bmod m,\ x\bmod n)$ is a ring isomorphism. Explicitly, if $mu+nv=1$, the solution of $x\equiv a\pmod m$, $x\equiv b\pmod n$ is $x\equiv anv+bmu\pmod{mn}$.
\end{theorem}
\begin{proof}
The map is a well-defined ring morphism. It is injective: if $m\mid x$ and $n\mid x$ with $m,n$ coprime then $mn\mid x$ (Gauss). Both sides have $mn$ elements, so it is bijective. For the formula: $nv\equiv1\pmod m$ and $nv\equiv0\pmod n$, and symmetrically for $mu$.
\end{proof}

Figure~\ref{fig:crt} shows the theorem as a walk: adding $1$ moves one step down and one step right, wrapping around the edges like on a torus. When the moduli are coprime, the walk visits every cell before coming back.

\begin{figure}[ht]
\centering
\fig{ch03-crt}
\caption{Left: $x\mapsto(x\bmod3,x\bmod4)$ for $x=0,\dots,11$. Each step $x\to x+1$ moves diagonally (the arrows show the steps that do not wrap around), and every cell is hit exactly once. Right: with the non-coprime moduli $2$ and $4$, the walk closes up after $\operatorname{lcm}(2,4)=4$ steps, so half the cells (grey) are never reached and the others are hit twice.}
\label{fig:crt}
\end{figure}

Units correspond to pairs of units, so $\varphi(mn)=\varphi(m)\varphi(n)$ for coprime $m,n$. With $\varphi(p^k)=p^k-p^{k-1}$ (the non-multiples of $p$), this recovers $\varphi(n)=n\prod_{p\mid n}\left(1-\frac1p\right)$ from Tome~0.

\begin{proposition}[linear congruences]\label{prop:lincong}
$ax\equiv b\pmod n$ has a solution if and only if $d=\gcd(a,n)$ divides $b$, and then it has exactly $d$ solutions modulo $n$.
\end{proposition}
\begin{proof}
Solvability means $b\in a\Z+n\Z=d\Z$. If $d\mid b$, divide through: $\frac adx\equiv\frac bd\pmod{\frac nd}$ has a unique solution $x_0$ modulo $\frac nd$, since $\frac ad$ is a unit there; it gives $d$ classes modulo $n$.
\end{proof}

\section{Lifting modulo prime powers}

\begin{think}
Modulo $5$, $x=2$ solves $x^2\equiv-1$. Look for a solution modulo $25$ of the form $2+5t$. Expand the square: why does the term $25t^2$ not matter, and why is there exactly one good $t$ in $\{0,\dots,4\}$?
\end{think}

\begin{theorem}[Hensel's lemma]\label{thm:hensel}
Let $f\in\Z[X]$, $p$ prime, $k\ge1$, and $a\in\Z$ with $f(a)\equiv0\pmod{p^k}$ and $f'(a)\not\equiv0\pmod p$. Then there is a unique $t\in\{0,\dots,p-1\}$ such that $f(a+tp^k)\equiv0\pmod{p^{k+1}}$.
\end{theorem}
\begin{proof}
Taylor's formula for polynomials gives $f(a+h)=f(a)+hf'(a)+h^2g(a,h)$ with $g$ having integer coefficients. With $h=tp^k$ and $2k\ge k+1$:
$f(a+tp^k)\equiv f(a)+tp^kf'(a)\pmod{p^{k+1}}$. Writing $f(a)=cp^k$, the condition becomes $c+tf'(a)\equiv0\pmod p$, which has exactly one solution $t$ modulo $p$ since $f'(a)$ is invertible modulo $p$.
\end{proof}

Iterating, a simple root modulo $p$ lifts to a unique root modulo every $p^k$. This is Newton's method in the $p$-adic world (Chapter~12).

\section{Exercises}

\begin{exercise}[$\star$]\label{ex:last2}
Find the last two digits of $7^{2026}$.
\end{exercise}

\begin{exercise}[$\star$]\label{ex:sunzi}
\emph{Sunzi (3rd century).} Find the integers that leave remainders $2$, $3$, $2$ when divided by $3$, $5$, $7$.
\end{exercise}

\begin{exercise}[$\star$]\label{ex:digits}
Show that an integer is congruent modulo $9$ to the sum of its digits, and modulo $11$ to the alternating sum of its digits (starting from the units).
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:sqrt1}
\begin{enumerate}[label=(\alph*)]
\item Let $p$ be an odd prime and $k\ge1$. Show that the solutions of $x^2\equiv1\pmod{p^k}$ are $x\equiv\pm1$.
\item Let $n$ be odd with $r$ distinct prime factors. How many solutions does $x^2\equiv1\pmod n$ have?
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:wilsonconv}
Show that if $n>4$ is composite then $(n-1)!\equiv0\pmod n$. Deduce that $n>1$ is prime if and only if $(n-1)!\equiv-1\pmod n$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:hensel}
Find $x\in\{0,\dots,124\}$ with $x^2\equiv-1\pmod{125}$, by lifting a solution modulo $5$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:tower}
Show that the last two digits of $3^{3^{3^3}}$ are $87$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:korselt}
\emph{Korselt's criterion, one direction.} Let $n$ be squarefree and composite, and suppose $p-1\mid n-1$ for every prime $p\mid n$. Show that $a^n\equiv a\pmod n$ for every integer $a$. Check that $561=3\cdot11\cdot17$ qualifies: it is a Carmichael number (Tome~0, Chapter~1 notebook).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{ex:lucas}
\emph{Lucas' theorem.} Let $p$ be prime, and write $n=\sum_in_ip^i$ and $k=\sum_ik_ip^i$ in base $p$. Show that
\[
\binom nk\equiv\prod_i\binom{n_i}{k_i}\pmod p,
\]
with $\binom{n_i}{k_i}=0$ when $k_i>n_i$. \emph{Hint: in $(\Z/p\Z)[X]$, $(1+X)^p=1+X^p$.} Deduce that $\binom nk$ is odd if and only if each binary digit of $k$ is at most the corresponding digit of $n$, which explains the Sierpi\'nski triangle of Tome~0.
\end{exercise}

\begin{beauty}
The Chinese remainder theorem says that arithmetic modulo $n$ is arithmetic modulo each prime power of $n$, independently. It makes RSA decryption four times faster (notebook), and in Tome~2 it becomes a statement about products of rings. Hensel's lemma is the door to the $p$-adic numbers of Chapter~12, where $\sqrt{-1}$ exists as a genuine $5$-adic number, the limit of $2,7,57,\dots$ from Exercise~\ref{ex:hensel}.
\end{beauty}

\section{Solutions}

\begin{solution}{ex:last2}
\idea Last two digits means modulo $100$. Rather than computing $\varphi(100)=40$, look for a small power of $7$ that is $\equiv1$: $7^2=49$ and $49^2=2401$ already give $7^4\equiv1$. Then only $2026\bmod4$ matters.
\writeup $7^2=49$ and $7^4=2401\equiv1\pmod{100}$. As $2026=4\cdot506+2$, $7^{2026}=(7^4)^{506}\cdot7^2\equiv7^2=49\pmod{100}$.
\end{solution}

\begin{solution}{ex:sunzi}
\idea The moduli $3,5,7$ are pairwise coprime, so the Chinese remainder theorem promises one class modulo $105$. With small moduli, solving two congruences at a time by trial is faster than computing B\'ezout coefficients.
\writeup Modulo $15$: $x\equiv2\pmod3$ and $x\equiv3\pmod5$ give $x\equiv8\pmod{15}$ (the numbers $\equiv3\pmod5$ below $15$ are $3,8,13$, and only $8\equiv2\pmod3$). Then $x=8+15t\equiv2\pmod7$ means $1+t\equiv2$, so $t\equiv1$ and $x\equiv23\pmod{105}$. The answers are $23+105k$.
\end{solution}

\begin{solution}{ex:digits}
\idea A number is a polynomial in $10$ with the digits as coefficients. Modulo $9$, $10\equiv1$; modulo $11$, $10\equiv-1$. Substitute.
\writeup If $n=\sum d_i10^i$ then $10\equiv1\pmod9$ gives $n\equiv\sum d_i$, and $10\equiv-1\pmod{11}$ gives $n\equiv\sum(-1)^id_i$, since congruences are compatible with sums and products.
\end{solution}

\begin{solution}{ex:sqrt1}
\idea (a) Factor: $x^2-1=(x-1)(x+1)$. In a field that would end the story; modulo $p^k$ we need that $p$ cannot divide both factors, which holds because their difference is $2$ and $p$ is odd. (b) The Chinese remainder theorem turns one equation modulo $n$ into independent equations modulo each prime power, and the counts multiply.
\writeup (a) $p^k\mid(x-1)(x+1)$, and $p$ cannot divide both factors (it would divide their difference $2$). So one factor is prime to $p$, hence to $p^k$, and by Gauss' lemma $p^k$ divides the other: $x\equiv\pm1$. These two classes are distinct since $p^k\nmid2$.

(b) Write $n=p_1^{a_1}\cdots p_r^{a_r}$. By the Chinese remainder theorem (applied repeatedly), solving modulo $n$ amounts to solving modulo each $p_i^{a_i}$ independently, with $2$ solutions each by (a). So there are $2^r$ solutions.
\end{solution}

\begin{solution}{ex:wilsonconv}
\idea For composite $n=ab$, the factors $a$ and $b$ usually both appear in $(n-1)!$. The only obstacle is $a=b$, a square: then use $a$ and $2a$ instead, which requires $2a<a^2$, i.e.\ $a>2$. That explains the exception $n=4$.
\writeup Let $n=ab$ with $1<a\le b<n$. If $a<b$, both $a$ and $b$ appear among $1,\dots,n-1$, so $ab=n$ divides $(n-1)!$. If $a=b$, then $n=a^2$ with $a>2$ (as $n>4$), so $a<2a<a^2=n$ and $a\cdot2a=2n$ divides $(n-1)!$. For the equivalence: primes satisfy Wilson; $n=4$ gives $3!\equiv2$; composite $n>4$ gives $0\not\equiv-1$.
\end{solution}

\begin{solution}{ex:hensel}
\idea Hensel's lemma is constructive: at each step, write the candidate $a+tp^k$, expand modulo $p^{k+1}$, and solve the resulting linear equation for $t$ modulo $p$. Start from $2^2\equiv-1\pmod5$.
\writeup Modulo $5$, $2^2=4\equiv-1$; with $f=X^2+1$, $f'(2)=4\not\equiv0$. Lift to $25$: $(2+5t)^2+1\equiv5+20t\pmod{25}$, so $1+4t\equiv0\pmod5$, $t=1$, $x=7$ (and $7^2+1=50\equiv0$). Lift to $125$: $(7+25t)^2+1\equiv50+350t\pmod{125}$, so $2+14t\equiv0\pmod5$, that is $4t\equiv3$, $t=2$, $x=57$. Check: $57^2+1=3250=26\cdot125$.
\end{solution}

\begin{solution}{ex:tower}
\idea Work from the top of the tower down: modulo $100$, a power of $3$ depends on its exponent modulo $40=\varphi(100)$ (Euler); and the exponent $3^{27}$ modulo $40$ depends on $27$ modulo the order of $3$ modulo $40$, which is $4$. Each level reduces the modulus.
\writeup We need $3^E\bmod100$ with $E=3^{27}$. Since $\varphi(100)=40$ and $\gcd(3,100)=1$, Theorem~\ref{thm:eulerthm} shows that $3^E$ depends only on $E\bmod40$. Now $3^4=81\equiv1\pmod{40}$, so $E=3^{27}=(3^4)^6\cdot3^3\equiv27\pmod{40}$. Hence $3^E\equiv3^{27}\pmod{100}$. Finally $3^{10}=59049\equiv49$, $3^{20}\equiv49^2=2401\equiv1$, so $3^{27}\equiv3^7=2187\equiv87\pmod{100}$.
\end{solution}

\begin{solution}{ex:korselt}
\idea To prove a congruence modulo a squarefree $n$, prove it modulo each prime factor $p$ (Chinese remainder theorem). Modulo $p$, Fermat says $a^{p-1}\equiv1$, so $a^{n-1}\equiv1$ as soon as $p-1$ divides $n-1$; that is exactly the hypothesis.
\writeup Let $p\mid n$ and write $n-1=(p-1)m$. If $p\nmid a$, Fermat gives $a^{n-1}=(a^{p-1})^m\equiv1$, so $a^n\equiv a\pmod p$; if $p\mid a$, both sides are $0$ modulo $p$. So every prime factor of $n$ divides $a^n-a$, and since $n$ is squarefree, their product $n$ does (distinct primes are pairwise coprime, apply Gauss' lemma repeatedly). For $561$: $560$ is divisible by $2$, $10$ and $16$.
\end{solution}

\begin{solution}{ex:lucas}
\idea Binomial coefficients are coefficients of $(1+X)^n$, so compute $(1+X)^n$ modulo $p$ cleverly. The ``freshman's dream'' $(1+X)^p=1+X^p$ in characteristic $p$ splits $(1+X)^n$ along the base-$p$ digits of $n$; then reading the coefficient of $X^k$ amounts to writing $k$ in base $p$.
\writeup In $(\Z/p\Z)[X]$, $(1+X)^p=1+X^p$ because $p\mid\binom pj$ for $0<j<p$ (Tome~0, Exercise~1.6); by induction $(1+X)^{p^i}=1+X^{p^i}$. Hence
\[
(1+X)^n=\prod_i\left(1+X^{p^i}\right)^{n_i}=\prod_i\sum_{j=0}^{n_i}\binom{n_i}jX^{jp^i}.
\]
Expanding, each exponent $k$ arises from exactly one choice of digits $j=k_i\le n_i$ (uniqueness of base-$p$ expansion, since all $j\le n_i<p$), with coefficient $\prod_i\binom{n_i}{k_i}$; if some $k_i>n_i$, $X^k$ does not appear. Compare with the coefficient $\binom nk$ of $X^k$. For $p=2$, $\binom{n_i}{k_i}$ is $0$ exactly when $k_i=1$, $n_i=0$.
\end{solution}
