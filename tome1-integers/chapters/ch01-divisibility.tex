\chapter{Divisibility and the Euclidean Algorithm}

\begin{center}\itshape
The oldest non-trivial algorithm still in daily use is Euclid's, from around 300 BC.\\
Its worst case is governed by the Fibonacci numbers, as Lam\'e proved in 1844, in the first analysis of an algorithm's complexity.
\end{center}

\section{Division and ideals of \texorpdfstring{$\Z$}{Z}}

\begin{theorem}[Euclidean division]\label{thm:div}
For $a\in\Z$ and $b\in\N^*$ there is a unique pair $(q,r)\in\Z^2$ with $a=bq+r$ and $0\le r<b$.
\end{theorem}
\begin{proof}
Existence: the set $\{a-bq:q\in\Z\}\cap\N$ is non-empty (take $q=-|a|$), so it has a least element $r=a-bq$. If $r\ge b$ then $r-b=a-b(q+1)$ would be a smaller element. Uniqueness: if $bq+r=bq'+r'$ with $0\le r,r'<b$, then $b\mid r-r'$ and $|r-r'|<b$, so $r=r'$ and $q=q'$.
\end{proof}

\begin{think}
The multiples of $6$ form a subgroup of $(\Z,+)$: sums and differences of multiples of $6$ are multiples of $6$. What about the set $4\Z+6\Z$ of all $4x+6y$? List its elements between $-10$ and $10$. Is it of the form $d\Z$? For which $d$?
\end{think}

\begin{proposition}\label{prop:subgroups}
Every subgroup of $(\Z,+)$ is of the form $d\Z$ for a unique $d\in\N$.
\end{proposition}
\begin{proof}
Let $H\ne\{0\}$ be a subgroup and $d$ its least positive element. Then $d\Z\subset H$. For $a\in H$, divide: $a=dq+r$ with $r=a-dq\in H$ and $0\le r<d$, so $r=0$. Uniqueness: $d$ is the least positive element of $d\Z$.
\end{proof}

\begin{definition}
For $a,b\in\Z$ not both zero, $a\Z+b\Z$ is a subgroup of $\Z$, hence equal to $d\Z$ for a unique $d\ge1$: this $d$ is the \emph{greatest common divisor} $\gcd(a,b)$.
\end{definition}

The name is justified: $d$ divides $a$ and $b$ (they lie in $d\Z$), and any common divisor of $a$ and $b$ divides every element of $a\Z+b\Z$, in particular $d$. Two immediate but fundamental consequences:

\begin{theorem}[B\'ezout]\label{thm:bezout}
There are $u,v\in\Z$ with $au+bv=\gcd(a,b)$. In particular $a$ and $b$ are coprime if and only if $au+bv=1$ for some integers $u,v$.
\end{theorem}
\begin{proof}
$\gcd(a,b)\in d\Z=a\Z+b\Z$. If $au+bv=1$, then $1\in a\Z+b\Z=d\Z$, so $d=1$; the converse is the first statement.
\end{proof}

\begin{theorem}[Gauss' lemma]\label{thm:gauss}
If $a\mid bc$ and $\gcd(a,b)=1$, then $a\mid c$. In particular, if a prime $p$ divides $bc$, it divides $b$ or $c$.
\end{theorem}
\begin{proof}
Write $au+bv=1$. Then $c=acu+bcv$, and $a$ divides both terms. For the second statement, if $p\nmid b$ then $\gcd(p,b)=1$.
\end{proof}

\begin{theorem}[fundamental theorem of arithmetic]\label{thm:fta}
Every integer $n\ge2$ is a product of primes, uniquely up to the order of the factors.
\end{theorem}
\begin{proof}
Existence was proved by minimal counterexample in Tome~0, Chapter~1. Uniqueness, by induction on $n$: if $p_1\cdots p_r=q_1\cdots q_s$, then $p_1$ divides the product of the $q_j$, so by Gauss' lemma (iterated) it divides some $q_j$, hence equals it. Cancel and apply the induction hypothesis.
\end{proof}

For a prime $p$ and $n\ne0$, the \emph{$p$-adic valuation} $v_p(n)$ is the exponent of $p$ in $n$. Uniqueness of factorisation gives $v_p(ab)=v_p(a)+v_p(b)$, and
\[
\gcd(a,b)=\prod_pp^{\min(v_p(a),v_p(b))},\qquad\operatorname{lcm}(a,b)=\prod_pp^{\max(v_p(a),v_p(b))},\qquad\gcd(a,b)\operatorname{lcm}(a,b)=|ab|
\]
for $a,b\ne0$. Indeed, by uniqueness of factorisation, $d\mid n$ iff $v_p(d)\le v_p(n)$ for every $p$ (if $n=dk$ then $v_p(n)=v_p(d)+v_p(k)$; conversely $k=\prod p^{v_p(n)-v_p(d)}$ works). So the common divisors of $a,b$ are the $d$ with $v_p(d)\le\min(v_p(a),v_p(b))$, and the greatest one, which every common divisor divides, is the stated product; likewise for common multiples. The last formula follows from $\min+\max=v_p(a)+v_p(b)$.

\section{The Euclidean algorithm}

Since $\gcd(a,b)=\gcd(b,a-bq)$ (the two pairs generate the same subgroup), we may replace $(a,b)$ by $(b,r)$ where $r$ is the remainder of $a$ divided by $b$. Starting with $r_0=a$, $r_1=b$ and dividing repeatedly,
\[
r_{k-1}=q_kr_k+r_{k+1},\qquad0\le r_{k+1}<r_k,
\]
the remainders decrease strictly, so some $r_{n+1}$ is $0$, and $\gcd(a,b)=r_n$, the last non-zero remainder.

\textbf{Extended version.} Keep track of coefficients with $r_k=au_k+bv_k$: start with $(u_0,v_0)=(1,0)$, $(u_1,v_1)=(0,1)$, and set $u_{k+1}=u_{k-1}-q_ku_k$, $v_{k+1}=v_{k-1}-q_kv_k$. By induction $r_k=au_k+bv_k$ for all $k$ (true for $k=0,1$, and $r_{k+1}=r_{k-1}-q_kr_k$ is matched by the coefficients), so at the end $au_n+bv_n=\gcd(a,b)$.

\begin{example}
For $(a,b)=(240,46)$: $240=5\cdot46+10$, $46=4\cdot10+6$, $10=1\cdot6+4$, $6=1\cdot4+2$, $4=2\cdot2$. So $\gcd=2$, and running the coefficients gives $240\cdot(-9)+46\cdot47=2$.
\end{example}

\begin{theorem}[Lam\'e]\label{thm:lame}
If the algorithm applied to $a>b\ge1$ performs $n$ divisions, then $b\ge F_{n+1}$ and $a\ge F_{n+2}$, where $(F_k)$ is the Fibonacci sequence.
\end{theorem}
\begin{proof}
The divisions produce $r_0=a>r_1=b>\dots>r_{n-1}>r_n\ge1$ and $r_{n+1}=0$. We show $r_k\ge F_{n+2-k}$ by descending induction on $k$. For $k=n$: $r_n\ge1=F_2$. For $k=n-1$: $r_{n-1}>r_n\ge1$ so $r_{n-1}\ge2=F_3$. Then each $q_k\ge1$, so $r_{k-1}=q_kr_k+r_{k+1}\ge r_k+r_{k+1}\ge F_{n+2-k}+F_{n+1-k}=F_{n+3-k}$.
\end{proof}

So the worst inputs are consecutive Fibonacci numbers, and since $F_{k}\approx\varphi^k/\sqrt5$, the number of steps is $O(\log b)$ (Exercise~\ref{ex:lame5}).

\section{Exercises}

\begin{exercise}[$\star$]\label{ex:bez}
Find all $(x,y)\in\Z^2$ with $91x+35y=7$.
\end{exercise}

\begin{exercise}[$\star$]\label{ex:gcdn}
For $n\in\N$, compute $\gcd\left(n^2+1,(n+1)^2+1\right)$.
\end{exercise}

\begin{exercise}[$\star$]\label{ex:zeros}
\emph{Legendre's formula.} Show that $v_p(n!)=\sum_{k\ge1}\left\lfloor\frac n{p^k}\right\rfloor$, and find the number of zeros at the end of $1000!$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:sqrt}
Show that $\sqrt n$ is rational only when $n\in\N$ is a perfect square. More generally, show that if $x\in\Q$ is a root of a monic polynomial with integer coefficients, then $x\in\Z$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:mersgcd}
Show that $\gcd\left(2^a-1,2^b-1\right)=2^{\gcd(a,b)}-1$ for $a,b\ge1$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:lame5}
Deduce from Lam\'e's theorem that the number of divisions is at most $5$ times the number of decimal digits of $b$.
\emph{Use $F_{k+1}\ge\varphi^{k-1}$ and $\varphi^5>10$.}
\end{exercise}

\begin{exercise}[$\star\star$]\label{ex:frob}
\emph{The coin problem.} Let $a,b\ge1$ be coprime. Show that the largest integer that cannot be written $ax+by$ with $x,y\in\N$ is $ab-a-b$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{ex:supercat}
\begin{enumerate}[label=(\alph*)]
\item Show that $\lfloor2x\rfloor+\lfloor2y\rfloor\ge\lfloor x\rfloor+\lfloor y\rfloor+\lfloor x+y\rfloor$ for all real $x,y$.
\item Deduce that $\dfrac{(2m)!\,(2n)!}{m!\,n!\,(m+n)!}$ is an integer for all $m,n\in\N$.
\end{enumerate}
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{ex:n2n}
Show that for every integer $n\ge2$, $n$ does not divide $2^n-1$. \emph{Hint: let $p$ be the smallest prime factor of $n$, and use Exercise~\ref{ex:mersgcd} and Fermat's little theorem.}
\end{exercise}

\begin{beauty}
Every result of this chapter comes from one fact: the subgroups of $\Z$ are the $d\Z$. In Tome~2 this becomes ``$\Z$ is a principal ideal domain'', and the same argument works verbatim for polynomials over a field, Gaussian integers, and every Euclidean ring. B\'ezout's coefficients are how RSA computes private keys (Chapter~3's notebook), and the coin problem with three or more coins, the \emph{Frobenius problem}, has no closed formula and is NP-hard in general.
\end{beauty}

\begin{hint}{ex:bez}

\end{hint}
\begin{hint}{ex:gcdn}
A common divisor $d$ divides every integer combination of $n^2+1$ and $n^2+2n+2$. Find a combination that eliminates $n$.
\end{hint}
\begin{hint}{ex:zeros}
Count, for each $k$, how many of $1,\dots,n$ are multiples of $p^k$; each number $m$ is counted $v_p(m)$ times.
\end{hint}
\begin{hint}{ex:sqrt}
Write $x=\frac pq$ in lowest terms, multiply the equation by $q^n$, and look at what $q$ must divide.
\end{hint}
\begin{hint}{ex:mersgcd}
Show that $2^a-1\equiv2^{a-b}-1\pmod{2^b-1}$: the pair of exponents follows Euclid's algorithm.
\end{hint}
\begin{hint}{ex:lame5}
If there are $n$ divisions, $b\ge F_{n+1}\ge\varphi^{n-1}$. Take logarithms in base $10$.
\end{hint}
\begin{hint}{ex:frob}
Every integer $N$ has a unique representation $N=ax+by$ with $0\le x<b$. It is a non-negative representation exactly when $y\ge0$.
\end{hint}
\begin{hint}{ex:supercat}
For (a), reduce to $0\le x,y<1$ by subtracting integer parts. For (b), compare the $p$-adic valuations of numerator and denominator with Legendre's formula.
\end{hint}
\begin{hint}{ex:n2n}
If $p\mid2^n-1$, then $p\mid\gcd(2^n-1,2^{p-1}-1)$. What is $\gcd(n,p-1)$ when $p$ is the smallest prime factor of $n$?
\end{hint}

\section{Solutions}

\begin{solution}{ex:bez}
\idea A linear equation $ax+by=c$ is solved in two steps: one particular solution from the extended Euclidean algorithm (scaled if $c$ is a multiple of the gcd), then all solutions by subtracting, which gives a homogeneous equation that Gauss' lemma solves after dividing by the gcd.
\writeup Euclid: $91=2\cdot35+21$, $35=21+14$, $21=14+7$, $14=2\cdot7$, so $\gcd(91,35)=7$, and back-substituting,
$7=21-14=2\cdot21-35=2\cdot91-5\cdot35$. So $(2,-5)$ is a solution. If $(x,y)$ is another, $91(x-2)=-35(y+5)$, so $13(x-2)=-5(y+5)$. Since $\gcd(5,13)=1$, Gauss' lemma gives $5\mid x-2$: $x=2+5k$, and then $y=-5-13k$. Conversely these all work: the solutions are $(2+5k,-5-13k)$, $k\in\Z$.
\end{solution}

\begin{solution}{ex:gcdn}
\idea A common divisor of two expressions divides every integer combination of them. So combine them to kill the variable: subtracting removes $n^2$, and then a second combination removes $n$, leaving a constant. The gcd divides that constant; then test which divisors actually occur.
\writeup Let $d$ be the gcd. It divides $(n+1)^2+1-(n^2+1)=2n+1$. Hence $d$ also divides $4(n^2+1)-(2n+1)(2n-1)=5$. So $d\in\{1,5\}$. If $d=5$ then $5\mid2n+1$, so $n\equiv2\pmod5$. Conversely, if $n=5k+2$ then $n^2+1=25k^2+20k+5$ and $(n+1)^2+1=25k^2+30k+10$ are multiples of $5$. So the gcd is $5$ if $n\equiv2\pmod5$ and $1$ otherwise.
\end{solution}

\begin{solution}{ex:zeros}
\idea The valuation of a product is the sum of the valuations, and $v_p(m)$ counts how many of $p,p^2,p^3,\dots$ divide $m$. Exchanging the two sums (``count by powers rather than by numbers'') turns the count into multiples of $p^k$ up to $n$, which is a floor. Trailing zeros count factors $10=2\cdot5$, and $5$ is the scarcer prime.
\writeup $v_p(n!)=\sum_{m=1}^nv_p(m)=\sum_{m=1}^n\sum_{k\ge1}\mathbf 1_{p^k\mid m}=\sum_{k\ge1}\#\{m\le n:p^k\mid m\}=\sum_{k\ge1}\left\lfloor\frac n{p^k}\right\rfloor$, because the multiples of $p^k$ up to $n$ are $p^k,2p^k,\dots,\lfloor n/p^k\rfloor p^k$.
The number of final zeros is $\min(v_2,v_5)=v_5(1000!)=200+40+8+1=249$ (clearly $v_2\ge v_5$ term by term).
\end{solution}

\begin{solution}{ex:sqrt}
\idea Two approaches. Valuations: in $nq^2=p^2$ every prime appears to an even power on the right. For the general statement, clear denominators in $P(p/q)=0$ and look for what $q$ must divide: everything except $p^d$ carries a factor $q$, and $q$ is coprime to $p$.
\writeup If $\sqrt n=\frac pq$ then $nq^2=p^2$, so $v_\ell(n)=2v_\ell(p)-2v_\ell(q)$ is even for every prime $\ell$, and $n$ is a square. In general, let $x=\frac pq$ in lowest terms with $q\ge1$ be a root of $X^d+c_{d-1}X^{d-1}+\dots+c_0$. Multiplying by $q^d$: $p^d=-q\left(c_{d-1}p^{d-1}+\dots+c_0q^{d-1}\right)$, so $q\mid p^d$. As $\gcd(q,p)=1$, Gauss' lemma (iterated) gives $q\mid1$, so $x\in\Z$. The case $X^2-n$ recovers the first statement.
\end{solution}

\begin{solution}{ex:mersgcd}
\idea The formula says that $a\mapsto2^a-1$ transports gcds. Euclid's algorithm computes $\gcd(a,b)$ by repeated subtraction, so it suffices to check that one subtraction step on $(a,b)$ corresponds to one step on $(2^a-1,2^b-1)$: divide $2^a-1$ by $2^b-1$ and look at the remainder.
\writeup For $a\ge b$, $2^a-1=2^{a-b}\left(2^b-1\right)+\left(2^{a-b}-1\right)$, so $\gcd(2^a-1,2^b-1)=\gcd(2^{a-b}-1,2^b-1)$. The map $(a,b)\mapsto\gcd(2^a-1,2^b-1)$ thus follows the subtractive Euclidean algorithm on the exponents, which ends at $(\gcd(a,b),0)$ and gives $\gcd\left(2^{\gcd(a,b)}-1,2^0-1\right)=\gcd\left(2^{\gcd(a,b)}-1,0\right)=2^{\gcd(a,b)}-1$.
\end{solution}

\begin{solution}{ex:lame5}
\idea Lam\'e bounds $b$ below by a Fibonacci number; Fibonacci numbers grow like $\varphi^k$; and $d$ digits means $b<10^d$. Chain the inequalities and compare $\varphi^5$ with $10$.
\writeup By induction, $F_{k+1}\ge\varphi^{k-1}$: true for $k=1,2$ ($F_2=1$, $F_3=2\ge\varphi$), and $F_{k+2}=F_{k+1}+F_k\ge\varphi^{k-1}+\varphi^{k-2}=\varphi^k$ since $\varphi^2=\varphi+1$. If $b$ has $d$ digits and the algorithm takes $n$ steps, then $10^d>b\ge F_{n+1}\ge\varphi^{n-1}$. As $\varphi^5\approx11.09>10$, we get $\varphi^{n-1}<10^d<\varphi^{5d}$, so $n-1<5d$, that is $n\le5d$.
\end{solution}

\begin{solution}{ex:frob}
\idea First understand all integer representations of $m$: they form one family $(x-kb,y+ka)$ (Exercise~\ref{ex:bez}'s method). Among them pick the one with $0\le x<b$; the question ``is there one with $x,y\ge0$?'' is then decided by the sign of its $y$. Compute experimentally with $a=3$, $b=5$ to see that $7=3\cdot5-3-5$ is the last failure, then prove it.
\writeup If $ax+by=ax'+by'$ then $a(x-x')=b(y'-y)$, so $b\mid x-x'$ by Gauss' lemma: all representations of $m$ in $\Z^2$ are $(x-kb,\,y+ka)$, $k\in\Z$. By B\'ezout there is one, so there is exactly one with $0\le x\le b-1$; call it canonical.
If its $y$ is negative, then any representation with $y+ka\ge0$ has $k\ge1$, hence $x-kb<0$. So $m$ is a non-negative combination if and only if its canonical $y$ is $\ge0$.
For $m=ab-a-b=a(b-1)+b\cdot(-1)$ the canonical $y$ is $-1$: not representable. For $m>ab-a-b$ with canonical $(x,y)$, $by=m-ax\ge m-a(b-1)>-b$, so $y>-1$, that is $y\ge0$: representable.
\end{solution}

\begin{solution}{ex:supercat}
\idea (a) Floors of sums depend only on fractional parts, so reduce to $x,y\in[0,1[$, where each floor is $0$ or $1$ and only two cases remain. (b) To prove a fraction is an integer, show every $p$-adic valuation is $\ge0$; Legendre's formula writes it as a sum of exactly the expression of (a).
\writeup (a) Write $x=a+\alpha$, $y=b+\beta$ with $a,b\in\Z$ and $\alpha,\beta\in[0,1[$. The integer parts contribute $2a+2b$ to both sides, and the inequality becomes $\lfloor2\alpha\rfloor+\lfloor2\beta\rfloor\ge\lfloor\alpha+\beta\rfloor$. If $\alpha+\beta<1$ the right side is $0$. Otherwise $\max(\alpha,\beta)\ge\frac12$, so the left side is at least $1=\lfloor\alpha+\beta\rfloor$.

(b) By Legendre's formula, the $p$-adic valuation of the fraction is $\sum_k\left(\left\lfloor\frac{2m}{p^k}\right\rfloor+\left\lfloor\frac{2n}{p^k}\right\rfloor-\left\lfloor\frac m{p^k}\right\rfloor-\left\lfloor\frac n{p^k}\right\rfloor-\left\lfloor\frac{m+n}{p^k}\right\rfloor\right)$, which is $\ge0$ by (a) with $x=\frac m{p^k}$, $y=\frac n{p^k}$. A positive rational with all valuations $\ge0$ is an integer.
\end{solution}

\begin{solution}{ex:n2n}
\idea Suppose $n\mid2^n-1$ and look at one prime $p\mid n$: then $2^n\equiv1$ and also $2^{p-1}\equiv1\pmod p$ (Fermat), so $p$ divides $2^{\gcd(n,p-1)}-1$ by Exercise~\ref{ex:mersgcd}. To get a contradiction we want $\gcd(n,p-1)=1$, which is guaranteed if $p$ is the \emph{smallest} prime factor of $n$: all prime factors of $n$ then exceed $p-1$.
\writeup Suppose $n\mid2^n-1$; then $n$ is odd, so its smallest prime factor $p$ is odd and $p\mid2^n-1$. By Fermat (Tome~0, Exercise~1.6), $p\mid2^{p-1}-1$. By Exercise~\ref{ex:mersgcd}, $p$ divides $\gcd(2^n-1,2^{p-1}-1)=2^{\gcd(n,p-1)}-1$. But every prime factor of $n$ is $\ge p>p-1$, so $\gcd(n,p-1)=1$, and $p\mid2^1-1=1$: contradiction.
\end{solution}
