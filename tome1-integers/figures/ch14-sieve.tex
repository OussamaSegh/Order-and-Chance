% Integers 1 through 30. A teal frame marks gcd(n,30)=1.
\begin{tikzpicture}[x=0.46cm,y=0.52cm,font=\tiny]
  \foreach \n in {1,...,15} {
    \pgfmathtruncatemacro{\x}{\n-1}
    \draw[draw=ocInk!55,fill=ocCream] (\x,0) rectangle ++(0.92,0.86);
    \node at (\x+0.46,0.43) {\n};
  }
  \foreach \n in {16,...,30} {
    \pgfmathtruncatemacro{\x}{\n-16}
    \draw[draw=ocInk!55,fill=ocCream] (\x,-1) rectangle ++(0.92,0.86);
    \node at (\x+0.46,-0.57) {\n};
  }
  % survivors: 1,7,11,13,17,19,23,29
  \foreach \n/\x/\y in {1/0/0, 7/6/0, 11/10/0, 13/12/0, 17/1/-1, 19/3/-1, 23/7/-1, 29/13/-1} {
    \draw[ocTeal,thick] (\x+0.08,\y+0.08) rectangle ++(0.76,0.70);
  }
  % multiples of 5, small olive mark in the corner, except the survivors
  \foreach \n/\x/\y in {5/4/0, 10/9/0, 15/14/0, 20/4/-1, 25/9/-1, 30/14/-1} {
    \fill[ocOlive] (\x+0.72,\y+0.66) circle (1.3pt);
  }
  \node[font=\scriptsize,anchor=west] at (0,-1.7)
    {teal frame: not divisible by $2$, $3$ or $5$\quad olive dot: divisible by $5$};
\end{tikzpicture}
