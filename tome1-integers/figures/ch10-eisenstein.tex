% A patch of the Eisenstein lattice, with the prime 1-omega marked.
\begin{tikzpicture}[scale=1.05,font=\scriptsize]
  \def\ox{-0.5}
  \def\oy{0.8660254}
  \foreach \m in {-1,0,1,2} {
    \foreach \n in {-1,0,1,2} {
      \pgfmathsetmacro{\x}{\m+\n*\ox}
      \pgfmathsetmacro{\y}{\n*\oy}
      \fill[ocInk] (\x,\y) circle (1.15pt);
    }
  }
  \draw[ocTeal,thick] (0,0)--(1,0)--(0.5,0.866)--cycle;
  \draw[ocCoralDeep,thick] (0,0)--(1,0);
  \draw[ocCoralDeep,thick] (0,0)--(-0.5,0.866);
  \node[below] at (0.5,0) {$1$};
  \node[above left] at (-0.5,0.866) {$\omega$};
  \node[ocCoralDeep,right] at (0.55,0.5) {$1-\omega$};
  \draw[ocCoralDeep,dashed] (1,0)--(0.5,0.866);
\end{tikzpicture}
