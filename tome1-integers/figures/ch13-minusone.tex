% The 5-adic expansion of -1: every digit is 4.
\begin{tikzpicture}[x=0.95cm,font=\scriptsize]
  \foreach \k/\dig/\sum in {0/4/4,1/4/24,2/4/124} {
    \draw[draw=ocInk,fill=ocCream] (\k,0.9) rectangle ++(0.8,0.7);
    \node at (\k+0.4,1.25) {\dig};
    \node[below] at (\k+0.4,0.85) {$5^{\k}$};
    \node[ocTeal,below] at (\k+0.4,0.15) {\sum};
  }
  \node[anchor=west] at (3.2,1.25) {$=\cdots444_5=-1$};
  \node[anchor=west,font=\tiny] at (0,-0.55)
    {partial sums $4$, $24$, $124$ are $-1$ modulo $5$, $25$, $125$};
\end{tikzpicture}
