% Primes up to 60, marked by their class modulo 4.
\begin{tikzpicture}[x=0.162cm,font=\scriptsize]
  \draw[ocInk] (2,0)--(61,0);
  \foreach \x in {10,20,30,40,50,60} {
    \draw (\x,0.12)--(\x,-0.12) node[below=2pt] {\x};
  }
  \fill[ocOlive] (2,0) circle (2.4pt);
  \node[ocOlive,above=1pt] at (2,0) {$2$};
  \foreach \p in {5,13,17,29,37,41,53} {
    \draw[ocBlue,thick] (\p,0) circle (2.6pt);
  }
  \foreach \p in {3,7,11,19,23,31,43,47,59} {
    \fill[ocCoralDeep] ([shift={(-1.6pt,-1.6pt)}]\p,0) rectangle ++(3.2pt,3.2pt);
  }
  \node[anchor=west,align=left] at (0,-0.85)
    {\textcolor{ocBlue}{circle $\equiv 1$}\enspace
     \textcolor{ocCoralDeep}{square $\equiv 3$}\enspace
     \textcolor{ocOlive}{dot $2$}};
\end{tikzpicture}
