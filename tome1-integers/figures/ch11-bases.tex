% Two integral bases: Z[sqrt(2)] and Z[(1+sqrt(5))/2].
\begin{tikzpicture}[font=\scriptsize]
  \begin{scope}
    \draw[->,ocInk] (0,0)--(3.3,0);
    \draw[->,ocInk] (0,0)--(0,3.1);
    \foreach \i in {0,1,2} {
      \foreach \j in {0,1,2} {
        \fill[ocBlue] (\i,{1.414*\j}) circle (1.2pt);
      }
    }
    \draw[ocBlue,thick] (0,0)--(1,0)--(1,1.414)--(0,1.414)--cycle;
    \node[below] at (1.5,-0.15) {$\Z[\sqrt2]$};
  \end{scope}
  \begin{scope}[xshift=4.6cm]
    \draw[->,ocInk] (0,0)--(3.4,0);
    \draw[->,ocInk] (0,0)--(0,3.2);
    \foreach \m in {0,1,2} {
      \foreach \n in {0,1,2} {
        \fill[ocCoralDeep] ({\m+0.5*\n},{1.118*\n}) circle (1.2pt);
      }
    }
    \draw[ocCoralDeep,thick] (0,0)--(1,0)--(1.5,1.118)--(0.5,1.118)--cycle;
    \node[below] at (1.6,-0.15) {$\Z[\frac{1+\sqrt5}2]$};
  \end{scope}
\end{tikzpicture}
