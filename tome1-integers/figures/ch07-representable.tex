% Which integers from 1 to 30 are sums of two squares.
\begin{tikzpicture}[x=0.48cm,y=0.48cm,font=\tiny]
  % row-major, five rows of six: n=1..30
  \foreach \n/\ok in {1/1,2/1,3/0,4/1,5/1,6/0,7/0,8/1,9/1,10/1,11/0,12/0,13/1,14/0,15/0,16/1,17/1,18/1,19/0,20/1,21/0,22/0,23/0,24/0,25/1,26/1,27/0,28/0,29/1,30/0} {
    \pgfmathtruncatemacro{\x}{mod(\n-1,6)}
    \pgfmathtruncatemacro{\y}{-(\n-1)/6}
    \ifnum\ok=1
      \draw[draw=ocInk!60,fill=ocTeal!30] (\x,\y) rectangle ++(0.92,0.86);
    \else
      \draw[draw=ocInk!60,fill=ocCoralDeep!12] (\x,\y) rectangle ++(0.92,0.86);
    \fi
    \node at (\x+0.46,\y+0.43) {\n};
  }
\end{tikzpicture}
