% Quadratic residues modulo 11. Squares are filled.
\begin{tikzpicture}[x=0.62cm,font=\scriptsize]
  \foreach \n/\sq in {0/1,1/1,2/0,3/1,4/1,5/1,6/0,7/0,8/0,9/1,10/0} {
    \ifnum\sq=1
      \draw[draw=ocInk,fill=ocTeal!35] (\n,0) rectangle ++(0.9,0.9);
    \else
      \draw[draw=ocInk,fill=ocCream] (\n,0) rectangle ++(0.9,0.9);
    \fi
    \node at (\n+0.45,0.45) {\n};
  }
  \node[anchor=west,font=\scriptsize] at (0,-0.45)
    {teal: quadratic residues modulo $11$, including $0$};
\end{tikzpicture}
